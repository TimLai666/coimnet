from pathlib import Path
import collections, gzip, hashlib, json, statistics, struct

ROOT = Path.cwd()
WORK = ROOT / 'runs/.tmp/chemical-ocr-20261013'
EVIDENCE = ROOT / 'evidence/TSK-11/chemical-quality-20261013'
PROTOCOL = json.loads((EVIDENCE / 'protocol.json').read_text())

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def checked_run(label):
    receipt = json.loads((WORK / 'logs' / (label + '.command.json')).read_text())
    assert receipt['exit_code'] == 0, label
    for channel in ['stdout', 'stderr']:
        assert sha(WORK / 'logs' / (label + '.' + channel)) == receipt[channel + '_sha256'], label
    rows = [json.loads(line) for line in (WORK / 'logs' / (label + '.stdout')).read_text().splitlines()]
    assert rows[0]['event'] == 'selection' and rows[-1]['event'] == 'final', label
    return rows, receipt

# Independently reconstruct the public loader's round-robin indices from raw
# official labels. Validate hashes before using labels to count paired errors.
labels_path = WORK / 'data/kmnist/t10k-labels-idx1-ubyte.gz'
source = next(x for x in json.loads((EVIDENCE / 'data-copy.json').read_text()) if x['path'] == str(labels_path))
assert sha(labels_path) == source['sha256']
raw = gzip.decompress(labels_path.read_bytes())
assert struct.unpack('>II', raw[:8]) == (2049, 10000) and len(raw) == 10008
labels = list(raw[8:])
by_class = [[i for i, value in enumerate(labels) if value == c] for c in range(10)]
assert all(len(indices) == 1000 for indices in by_class)
all_indices = [by_class[c][rank] for rank in range(1000) for c in range(10)]
split_indices = {'after': all_indices[200:2200], 'confirmation': all_indices[8000:10000]}
split_hashes = {name: hashlib.sha256(b''.join(struct.pack('>Q', i) for i in indices)).hexdigest() for name, indices in split_indices.items()}
assert split_hashes['after'] == 'a25ee4d0b8c80fa5dc347e8fa821d90f67eec5e8365dc06fcd20ff83eb734f43'
assert split_hashes['confirmation'] == PROTOCOL['confirmation']['index_sha256_expected']
truth = {name: [labels[i] for i in indices] for name, indices in split_indices.items()}
history = {x['seed']: x['trained'] for x in json.loads((ROOT / 'evidence/TSK-11/quality-20261012/quality-results.json').read_text())['models']}
# The original scorer and SDK objective associate additions differently.
# A frozen-logit probe exactly reproduces both expressions; no tolerance or
# training change is needed for the historical comparison.
probe_label = 'historical-score-probe'
probe_receipt = json.loads((WORK / 'logs' / (probe_label + '.command.json')).read_text())
assert probe_receipt['exit_code'] == 0
for channel in ['stdout', 'stderr']:
    assert sha(WORK / 'logs' / (probe_label + '.' + channel)) == probe_receipt[channel + '_sha256']
probe = {(x['seed'], x['split']): x for x in [json.loads(line) for line in (WORK / 'logs' / (probe_label + '.stdout')).read_text().splitlines()]}
assert len(probe) == 6
checks, records, pairs = [], [], []
for seed in PROTOCOL['seed_list']:
    groups = {}
    for group in PROTOCOL['groups']:
        rows, receipt = checked_run('quality-train-' + str(seed) + '-' + group)
        restored_rows, restore_receipt = checked_run('quality-restore-' + str(seed) + '-' + group)
        selection, final, restored = rows[0], rows[-1], restored_rows[-1]
        assert final['updates'] == PROTOCOL['training'] * PROTOCOL['epochs'] == 120000
        epochs = [x for x in rows if x['event'] == 'epoch']
        assert [x['epoch'] for x in epochs] == list(range(1, 21))
        assert [x['updates'] for x in epochs] == [6000 * i for i in range(1, 21)]
        assert selection == restored_rows[0], (seed, group, 'restored selection')
        restore_fields = ['after', 'confirmation', 'train', 'updates', 'snapshot_sha256', 'next_update_sha256', 'parameters_sha256', 'optimizer_sha256', 'frozen_evaluation']
        assert all(final[k] == restored[k] for k in restore_fields), (seed, group, 'fresh-process continuation')
        assert final['frozen_evaluation']
        assert selection['trainable_parameters'] == 23516 and selection['epochs'] == 20
        assert selection['train_indices_sha256'] == 'bb87f08092d04ad4f4fb16715f44b6e46f44e4d558db6bb5f63be57a3d11434f'
        assert selection['test_indices_sha256'] == split_hashes['after']
        assert selection['confirmation_indices_sha256'] == split_hashes['confirmation']
        if group == 'off':
            old = history[seed]
            assert final['snapshot_sha256'] == old['snapshot_sha256'], (seed, 'historical snapshot')
            for name in ['after', 'train']:
                for field in ['accuracy', 'samples', 'confusion']:
                    assert final[name][field] == old[name][field], (seed, name, field)
                p = probe[(seed, name)]
                assert p['model_unchanged'] and p['snapshot_sha256'] == final['snapshot_sha256']
                assert p['historical_expression_mean'] == old[name]['mean_cross_entropy']
                assert p['sdk_objective_expression_mean'] == final[name]['mean_cross_entropy']
        models = [{'path': str(p.relative_to(ROOT)), 'bytes': p.stat().st_size, 'sha256': sha(p)} for p in sorted(Path(final['model']).iterdir())]
        assert len(models) == 3 and {Path(x['path']).name for x in models} == {'arrays.bin', 'document.json', 'manifest.json'}
        item = {'seed': seed, 'group': group, 'selection': selection, 'final': final, 'epochs': epochs, 'training_seconds': sum(x['train_seconds'] for x in epochs), 'command': receipt, 'restore_command': restore_receipt, 'model_files': models, 'restore_exact': True}
        groups[group] = item
        records.append(item)
        checks.append({'seed': seed, 'group': group, 'restore_exact': True, 'historical_off_snapshot_classification_and_both_score_expressions_exact': group == 'off'})
    off, on = groups['off'], groups['on']
    paired_fields = ['seed', 'initial_parameter_sha256', 'initial_snapshot_sha256', 'task_config', 'epochs', 'updates_expected', 'trainable_parameters', 'program', 'source_files', 'metadata_sha256', 'train_indices_sha256', 'test_indices_sha256', 'confirmation_indices_sha256']
    assert all(off['selection'][k] == on['selection'][k] for k in paired_fields), (seed, 'paired setup')
    off_config, on_config = dict(off['selection']['network_config']), dict(on['selection']['network_config'])
    assert off_config.get('chemistry') is None and on_config.get('chemistry') is not None
    on_config.pop('chemistry'); off_config.pop('chemistry', None)
    assert off_config == on_config
    pair = {'seed': seed, 'training_seconds_off': off['training_seconds'], 'training_seconds_on': on['training_seconds'], 'training_seconds_ratio_on_over_off': on['training_seconds'] / off['training_seconds'], 'splits': {}}
    for name in ['after', 'confirmation']:
        a, b = off['final'][name], on['final'][name]
        assert a['samples'] == b['samples'] == len(truth[name]) == 2000
        counts = collections.Counter((pa == target, pb == target) for pa, pb, target in zip(a['predictions'], b['predictions'], truth[name]))
        assert sum(counts.values()) == 2000
        assert counts[(True, True)] + counts[(True, False)] == a['correct']
        assert counts[(True, True)] + counts[(False, True)] == b['correct']
        pair['splits'][name] = {'accuracy_off': a['accuracy'], 'accuracy_on': b['accuracy'], 'difference_percentage_points': (b['correct'] - a['correct']) / 20.0, 'mean_cross_entropy_off': a['mean_cross_entropy'], 'mean_cross_entropy_on': b['mean_cross_entropy'], 'on_correct_off_wrong': counts[(False, True)], 'off_correct_on_wrong': counts[(True, False)], 'both_correct': counts[(True, True)], 'both_wrong': counts[(False, False)], 'samples': 2000}
    pairs.append(pair)

summary = {}
for name in ['after', 'confirmation']:
    deltas = [x['splits'][name]['difference_percentage_points'] for x in pairs]
    summary[name] = {'mean_accuracy_off': statistics.mean(x['splits'][name]['accuracy_off'] for x in pairs), 'mean_accuracy_on': statistics.mean(x['splits'][name]['accuracy_on'] for x in pairs), 'mean_difference_percentage_points': statistics.mean(deltas), 'minimum_difference_percentage_points': min(deltas), 'maximum_difference_percentage_points': max(deltas), 'positive_pairs': sum(d > 0 for d in deltas), 'negative_pairs': sum(d < 0 for d in deltas), 'zero_pairs': sum(d == 0 for d in deltas)}
# Full predictions/configuration remain in the compressed raw command logs.
# Keep the readable result focused on scores, curves, checks and fingerprints.
compact_records = []
for item in records:
    compact = {k: item[k] for k in ['seed', 'group', 'training_seconds', 'model_files', 'restore_exact']}
    compact['selection'] = {k: item['selection'][k] for k in ['initial_parameter_sha256', 'initial_snapshot_sha256', 'trainable_parameters', 'fixed_chemical_parameters', 'train_indices_sha256', 'test_indices_sha256', 'confirmation_indices_sha256']}
    compact['final'] = {k: item['final'][k] for k in ['updates', 'snapshot_sha256', 'next_update_sha256', 'parameters_sha256', 'optimizer_sha256', 'frozen_evaluation']}
    compact['scores'] = {name: {k: v for k, v in item['final'][name].items() if k != 'predictions'} for name in ['before', 'after', 'confirmation', 'train']}
    compact['epochs'] = [{k: row[k] for k in ['epoch', 'updates', 'mean_training_cross_entropy', 'train_seconds']} | {'development_accuracy': row['development']['accuracy'], 'development_cross_entropy': row['development']['mean_cross_entropy']} for row in item['epochs']]
    compact['train_receipt'] = 'logs/quality-train-' + str(item['seed']) + '-' + item['group'] + '.command.json'
    compact['restore_receipt'] = 'logs/quality-restore-' + str(item['seed']) + '-' + item['group'] + '.command.json'
    compact_records.append(compact)
result = {'protocol_sha256': sha(EVIDENCE / 'protocol.json'), 'paired_initialization_and_setup_equal': True, 'all_six_restore_exact': True, 'all_three_historical_off_snapshots_and_classification_exact': True, 'all_six_historical_score_expressions_reproduced_exact': True, 'independently_reconstructed_split_hashes': split_hashes, 'checks': checks, 'pairs': pairs, 'summary': summary, 'runs': compact_records, 'limits': ['three seeds are insufficient for population confidence', 'primary test split previously observed', 'fixed engineering chemistry, no biological fidelity claim', 'single-character artificial CPU model, no whole-brain or multi-character claim', 'fixed 20 epochs, no best-epoch selection or test-based tuning']}
target = EVIDENCE / 'comparison-results.json'
assert not target.exists()
target.write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({'pairs': pairs, 'summary': summary, 'checks': checks}, indent=2))
