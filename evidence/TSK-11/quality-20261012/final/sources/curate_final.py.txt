def sha(p):return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()
def read(p):return json.loads(pathlib.Path(p).read_text())
def store(p,x):
    b=(json.dumps(x,ensure_ascii=False,indent=2)+'\n').encode()
    if p.exists():assert p.read_bytes()==b,('refuse replacement',p)
    else:p.parent.mkdir(parents=True,exist_ok=True);p.write_bytes(b)
def archive(source,destination,compressed=False):
    b=source.read_bytes();saved=gzip.compress(b,mtime=0) if compressed else b
    if destination.exists():assert destination.read_bytes()==saved,('refuse replacement',destination)
    else:destination.parent.mkdir(parents=True,exist_ok=True);destination.write_bytes(saved)
    assert (gzip.decompress(destination.read_bytes()) if compressed else destination.read_bytes())==b
    return {'path':str(destination.relative_to(repo)),'sha256':sha(destination),'bytes':destination.stat().st_size,'original_sha256':hashlib.sha256(b).hexdigest(),'original_bytes':len(b)}
