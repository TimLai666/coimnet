"""Pure numeric-report controls, using hand-computable finite differences."""
from pathlib import Path
import argparse
import ast
import hashlib
import json
import math
import struct
import sys

e = Path(__file__).resolve().parent
parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--report', required=True, help='new evidence report filename')
args = parser.parse_args()
assert Path(args.report).name == args.report
source = (e / 'analyze.py.txt').read_bytes()
tree = ast.parse(source)
functions = ast.Module(body=[n for n in tree.body if isinstance(n, ast.FunctionDef)
                            and n.name in ['numeric_arrays', 'array_comparison']], type_ignores=[])
ns = {'math': math, 'struct': struct}
exec(compile(functions, 'actual-numeric-report-functions', 'exec'), ns)
checks = []
for name, b, expected in [('ordinary_3_4', [3., 4.], 5.),
                          ('finite_tiny_singleton', [1e-200], 1e-200),
                          ('finite_large_singleton', [1e200], 1e200)]:
    a = [0.] * len(b)
    actual = ns['array_comparison']({'p': a}, {'p': b})['.p']
    checks.append({'case': name, 'expected_l2': expected,
                   'actual_l2': actual['difference_l2'] if math.isfinite(actual['difference_l2']) else str(actual['difference_l2']),
                   'different_bits': actual['different_float64_bits'],
                   'max_absolute_difference': actual['max_absolute_difference'],
                   'passed': actual['difference_l2'] == expected})
zero = ns['array_comparison']({'p': [0.]}, {'p': [-0.]})['.p']
checks.append({'case': 'signed_zero', 'passed': zero['different_float64_bits'] == 1
               and zero['difference_l2'] == zero['max_absolute_difference'] == 0})
try:
    ns['array_comparison']({'p': [0.]}, {'p': [0., 1.]})
    mismatch = False
except AssertionError:
    mismatch = True
checks.append({'case': 'shape_mismatch_rejected', 'passed': mismatch})
report = {'analyzer_sha256': hashlib.sha256(source).hexdigest(), 'checks': checks,
          'passed': sum(x['passed'] for x in checks), 'failed': sum(not x['passed'] for x in checks),
          'all_inputs_finite': True, 'expected_distances_hand_computable': True,
          'pure_functions_only_no_model_or_training_commands': True,
          'scope': 'numeric reporting kernel; not evidence that current model values reach these limits'}
with (e / args.report).open('x') as f:
    json.dump(report, f, indent=2, allow_nan=False)
    f.write('\n')
print(json.dumps(report, allow_nan=False))
sys.exit(0 if all(x['passed'] for x in checks) else 1)
