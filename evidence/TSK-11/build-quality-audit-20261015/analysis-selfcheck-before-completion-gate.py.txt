from pathlib import Path
import ast, hashlib, json, math, struct

r = Path('/Users/timlai/Developer/coimnet')
e = r / 'evidence/TSK-11/build-quality-audit-20261015'
source = (e / 'analyze.py.txt').read_bytes()
assert hashlib.sha256(source).hexdigest() == json.loads((e / 'snapshot-negative-zero-repair.json').read_text())['after_sha256']
names = ['events', 'verify_score', 'numeric_arrays', 'array_comparison', 'score_pair', 'decode_snapshot']
tree = ast.parse(source)
functions = ast.Module(body=[n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names], type_ignores=[])
files = {}
class MemoryFile:
    def __init__(self, name):
        self.name = name
    def read_bytes(self):
        return files[self.name]
    def read_text(self):
        return self.read_bytes().decode()
class MemoryDirectory:
    def __truediv__(self, name):
        return MemoryFile(name)
ns = {'json': json, 'math': math, 'struct': struct, 'e': MemoryDirectory(), 'sha': lambda b: hashlib.sha256(b).hexdigest()}
exec(compile(functions, 'analysis-pure-functions', 'exec'), ns)
receipt = {'exit_code': 0, 'log_sha256': hashlib.sha256(b'{"event":"control"}\n').hexdigest(), 'formal_source_manifest': 'source-before.json', 'formal_sources_checked': 810, 'command': ['control']}
files['control.log'] = b'{"event":"control"}\n'
def set_receipt(value):
    files['control.json'] = json.dumps(value).encode()
set_receipt(receipt)
assert ns['events']('control', ['control']) == [{'event': 'control'}]
def rejects(call):
    try:
        call()
    except AssertionError:
        return True
    raise AssertionError('invalid control was accepted')
checks = {}
bad = dict(receipt, exit_code=1)
set_receipt(bad)
checks['failed_receipt_rejected'] = rejects(lambda: ns['events']('control'))
set_receipt(receipt)
checks['wrong_command_rejected'] = rejects(lambda: ns['events']('control', ['different']))
files['control.log'] = b'{"event":"mutated"}\n'
checks['mutated_log_rejected'] = rejects(lambda: ns['events']('control'))
files['control.log'] = b'{"event":"control"}\n'
set_receipt(dict(receipt, formal_sources_checked=809))
checks['wrong_source_count_rejected'] = rejects(lambda: ns['events']('control'))

truth = [0, 1, 2, 3, 4]
ap, bp = [0, 2, 1, 3, 8], [1, 1, 8, 3, 9]
def independent_score(predictions, cross_entropy):
    confusion = [[0] * 10 for _ in range(10)]
    correct = 0
    for target, prediction in zip(truth, predictions):
        confusion[target][prediction] += 1
        correct += target == prediction
    return {'samples': len(truth), 'predictions': predictions, 'confusion': confusion, 'correct': correct, 'accuracy': correct / len(truth), 'mean_cross_entropy': cross_entropy}
a, b = independent_score(ap, 1.25), independent_score(bp, 1.5)
ns['verify_score'](a, truth)
ns['verify_score'](b, truth)
pair = ns['score_pair'](a, b, truth)
assert pair['different_predictions'] == 4
assert pair['a_correct_b_wrong'] == pair['a_wrong_b_correct'] == 1
assert pair['both_wrong_different_class'] == 2
assert pair['b_minus_a_accuracy_percentage_points'] == 0
assert pair['b_minus_a_cross_entropy'] == 0.25
perfect = ns['score_pair'](a, independent_score(truth, 0), truth)
assert perfect['b_minus_a_accuracy_percentage_points'] == 60
for value in [-1, 10, False]:
    bad = json.loads(json.dumps(a))
    bad['predictions'][0] = value
    checks['invalid_prediction_' + str(value) + '_rejected'] = rejects(lambda: ns['verify_score'](bad, truth))

serialized = (e / 'json-negative-zero-control.log').read_bytes()
assert serialized == b'[-0,0]\n'
decoded = ns['decode_snapshot'](serialized)
assert [struct.pack('>d', v).hex() for v in decoded] == ['8000000000000000', '0000000000000000']
checks['go_serialized_negative_zero_preserved'] = True
numeric = ns['array_comparison']({'p': [0., -0., 2.]}, {'p': [0., 0., 3.]})['.p']
assert numeric['scalars'] == 3 and numeric['different_float64_bits'] == 2
assert numeric['max_absolute_difference'] == numeric['difference_l2'] == 1
checks['shape_mismatch_rejected'] = rejects(lambda: ns['array_comparison']({'p': [0.]}, {'p': [0., 1.]}))
result = {'analyzer_sha256': hashlib.sha256(source).hexdigest(), 'checks': checks, 'paired_control': pair, 'perfect_minus_a_accuracy_percentage_points': 60, 'signed_zero_and_numeric_control': numeric, 'artifact': 'deterministic verifier controls; not model quality measurements', 'pure_functions_only_no_training_or_model_commands': True}
with (e / 'analysis-selfcheck-final.json').open('x') as f:
    json.dump(result, f, indent=2)
    f.write('\n')
print(json.dumps({'analysis_controls': 'passed', 'reject_controls': len(checks), 'paired_different_predictions': 4, 'different_float64_bits': 2}))
