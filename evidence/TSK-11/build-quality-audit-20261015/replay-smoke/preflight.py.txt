from pathlib import Path
from collections import Counter
import gzip, hashlib, json, math, struct

root = Path('/Users/timlai/Developer/coimnet')
e = root / 'runs/.tmp/102-ocr-replay-smoke-20261015/results'
w = root / 'runs/.tmp/102-ocr-replay-smoke-20261015'
sha = lambda b: hashlib.sha256(b).hexdigest()
def save(name, obj):
    with (e / name).open('x') as f:
        json.dump(obj, f, indent=2, ensure_ascii=False)
        f.write('\n')
def events(name):
    return [json.loads(s) for s in (e / name).read_text().splitlines() if s.startswith('{')]

for manifest in ['source-before.json', 'driver-lock.json', 'protected-models.json']:
    for p, h in json.loads((e / manifest).read_text()).items():
        assert sha((root / p).read_bytes()) == h, p
copies = json.loads((e / 'data-copy.json').read_text())
for rec in copies:
    for p in [rec['source'], rec['path']]:
        raw = Path(p).read_bytes()
        assert len(raw) == rec['bytes'] and sha(raw) == rec['sha256'], p

idx = {}
for rec in copies:
    name = Path(rec['path']).name
    if not name.endswith('.gz'):
        continue
    raw = gzip.decompress(Path(rec['path']).read_bytes())
    magic, count = struct.unpack('>II', raw[:8])
    if 'images' in name:
        height, width = struct.unpack('>II', raw[8:16])
        assert (magic, height, width, len(raw)) == (2051, 28, 28, 16 + count * 784)
        payload = raw[16:]
    else:
        assert magic == 2049 and len(raw) == count + 8
        payload = raw[8:]
        assert len(Counter(payload)) == 10
    idx[name] = (count, payload, len(raw))

def selected(labels, offset, size):
    buckets = [[i for i, label in enumerate(labels) if label == c] for c in range(10)]
    return [buckets[c][rank] for rank in range(offset, offset + size) for c in range(10)]
labels = idx['train-labels-idx1-ubyte.gz'][1]
test_labels = idx['t10k-labels-idx1-ubyte.gz'][1]
assert Counter(labels) == {c: 6000 for c in range(10)}
assert Counter(test_labels) == {c: 1000 for c in range(10)}
splits = {
    'train': selected(labels, 0, 600),
    'development': selected(test_labels, 0, 20),
    'primary': selected(test_labels, 20, 200),
    'confirmation': selected(test_labels, 800, 200),
}
expected = {
    'train': 'bb87f08092d04ad4f4fb16715f44b6e46f44e4d558db6bb5f63be57a3d11434f',
    'development': '9b4fb30a75bdb1625bf0e636831ec633c86a7a62e74882d6c6fc2e5bc51943d2',
    'primary': 'a25ee4d0b8c80fa5dc347e8fa821d90f67eec5e8365dc06fcd20ff83eb734f43',
    'confirmation': '9500a51eff8ec4e5ef522459730caae6dd0b3463898ecc86430c4849875937a5',
}
for k, v in splits.items():
    assert sha(b''.join(struct.pack('>Q', i) for i in v)) == expected[k], k
for a, b in [('development', 'primary'), ('development', 'confirmation'), ('primary', 'confirmation')]:
    assert not set(splits[a]) & set(splits[b]), (a, b)

hist = Counter()
images = idx['train-images-idx3-ubyte.gz'][1]
for i in splits['train']:
    hist.update(images[i * 784:(i + 1) * 784])
n = sum(hist.values())
mean = sum(v * c for v, c in hist.items()) / n / 255
std = math.sqrt(sum(c * (v / 255 - mean) ** 2 for v, c in hist.items()) / n)
def quantile(q):
    position = q * (n - 1)
    def at(rank):
        total = 0
        for v in range(256):
            total += hist[v]
            if total > rank:
                return v / 255
        raise AssertionError(rank)
    lo, hi = math.floor(position), math.ceil(position)
    return at(lo) + (position - lo) * (at(hi) - at(lo))

binary = {str((w / p).relative_to(root)): sha((w / p).read_bytes()) for p in ['comparison', 'comparison-race']}
controls = []
for build in ['ordinary', 'race']:
    rows = events('sdk-check-' + build + '.log')
    selection, result = rows
    assert result == {'active_outputs_differ': True, 'event': 'SDK-equivalence', 'exact_logits_objective_optimizer_updates': True, 'neutral_exact': True, 'samples': 10}
    assert selection['train_indices_sha256'] == expected['train']
    assert selection['test_indices_sha256'] == expected['primary']
    assert selection['confirmation_indices_sha256'] == expected['confirmation']
    assert (selection['train_count'], selection['test_count'], selection['confirmation_count'], selection['epochs'], selection['updates_expected'], selection['trainable_parameters']) == (6000, 2000, 2000, 20, 120000, 23516)
    path = 'comparison' if build == 'ordinary' else 'comparison-race'
    assert selection['program']['binary_sha256'] == binary[str((w / path).relative_to(root))]
    for rec in selection['source_files']:
        copy = next(c for c in copies if Path(c['path']).name == rec['name'])
        assert (rec['sha256'], rec['bytes']) == (copy['sha256'], copy['bytes'])
    controls.append(selection)
assert {k: v for k, v in controls[0].items() if k != 'program'} == {k: v for k, v in controls[1].items() if k != 'program'}
assert controls[0]['program']['source_sha256'] == controls[1]['program']['source_sha256']
for stem in ['driver-tests', 'driver-race-tests']:
    lines = (e / (stem + '.log')).read_text().splitlines()
    assert sum(s.startswith('--- PASS:') for s in lines) == 5
    assert not any(s.startswith('--- FAIL:') for s in lines)
    assert json.loads((e / (stem + '.json')).read_text())['exit_code'] == 0
save('binary-lock.json', binary)
save('preflight.json', {
    'root_independent_data_verification': True,
    'data_bytes': sum(c['bytes'] for c in copies),
    'idx_complete_gzip_crc_valid': True,
    'idx': {k: {'count': v[0], 'uncompressed_bytes': v[2]} for k, v in idx.items()},
    'split_index_sha256': expected,
    'split_counts': {k: len(v) for k, v in splits.items()},
    'held_out_splits_disjoint': True,
    'train_pixel_eda': {'pixels': n, 'mean': mean, 'population_stddev': std, 'quartiles': [quantile(q) for q in [.25, .5, .75]], 'range': [min(hist) / 255, max(hist) / 255], 'zero_fraction': hist[0] / n, 'missing': 0, 'nonfinite': 0},
    'sdk_both_builds_verified': True, 'five_control_tests_each_build_passed': True,
    'same_initial_state_and_configuration': True,

})
