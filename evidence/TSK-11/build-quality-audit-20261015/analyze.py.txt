def events(name, expected_command=None):
    receipt = json.loads((e / (name + '.json')).read_text())
    raw = (e / (name + '.log')).read_bytes()
    assert receipt['exit_code'] == 0, (name, 'command failed')
    assert sha(raw) == receipt['log_sha256'], (name, 'log fingerprint changed')
    assert receipt['formal_source_manifest'] == 'source-before.json' and receipt['formal_sources_checked'] == 810
    if expected_command is not None:
        assert receipt['command'] == expected_command, (name, 'command mismatch')
    return [json.loads(s) for s in raw.decode().splitlines() if s.startswith('{')]
def pick(labels, offset, count):
    buckets = [[i for i, v in enumerate(labels) if v == c] for c in range(10)]
    return [labels[buckets[c][rank]] for rank in range(offset, offset + count) for c in range(10)]

def verify_score(score, truth):
    confusion = [[0] * 10 for _ in range(10)]
    assert len(score['predictions']) == len(truth)
    assert all(type(v) is int and 0 <= v < 10 for v in truth + score['predictions'])
    for target, prediction in zip(truth, score['predictions']):
        confusion[target][prediction] += 1
    correct = sum(confusion[i][i] for i in range(10))
    assert confusion == score['confusion'] and correct == score['correct']
    assert score['samples'] == len(truth) and score['accuracy'] == correct / len(truth)
    assert math.isfinite(score['mean_cross_entropy'])

def verify_selection(selection):
    for field, expected in {'train_count': 6000, 'test_count': 2000, 'confirmation_count': 2000, 'epochs': 20, 'updates_expected': 120000, 'trainable_parameters': 23516}.items():
        assert type(selection.get(field)) is int and selection[field] == expected, field
    for field, split in [('train_indices_sha256', 'train'), ('test_indices_sha256', 'primary'), ('confirmation_indices_sha256', 'confirmation')]:
        assert selection[field] == preflight['split_index_sha256'][split], field
    copies = json.loads((e / 'data-copy.json').read_text())
    metadata = next(c for c in copies if Path(c['path']).name == 'metadata.json')
    assert selection['metadata_sha256'] == metadata['sha256']
    assert type(selection['source_files']) is list
    actual = {}
    for source in selection['source_files']:
        assert source['name'] not in actual, source['name']
        actual[source['name']] = (source['bytes'], source['sha256'])
    expected = {Path(c['path']).name: (c['bytes'], c['sha256']) for c in copies if Path(c['path']).name != 'metadata.json'}
    assert actual == expected

def decode_snapshot(raw):
    # Go encodes negative float64 zero as the JSON integer-looking token -0.
    return json.loads(raw, parse_int=lambda text: -0.0 if text == '-0' else int(text))

def numeric_arrays(value, path=''):
    if isinstance(value, dict):
        for k, v in sorted(value.items()):
            yield from numeric_arrays(v, path + '.' + k)
    elif isinstance(value, list):
        if all(type(x) in (int, float) for x in value):
            yield path, value
        else:
            for i, v in enumerate(value):
                yield from numeric_arrays(v, path + f'[{i}]')

def array_comparison(a, b):
    aa, bb = dict(numeric_arrays(a)), dict(numeric_arrays(b))
    assert aa.keys() == bb.keys()
    result = {}
    for key in aa:
        x, y = aa[key], bb[key]
        assert len(x) == len(y)
        assert all(math.isfinite(v) for v in x + y)
        unequal = [i for i, (u, v) in enumerate(zip(x, y)) if struct.pack('>d', u) != struct.pack('>d', v)]
        diff = [v - u for u, v in zip(x, y)]
        result[key] = {'scalars': len(x), 'different_float64_bits': len(unequal), 'max_absolute_difference': max(map(abs, diff), default=0), 'difference_l2': math.hypot(*diff), 'first_differences': [{'index': i, 'ordinary': x[i], 'race': y[i]} for i in unequal[:5]]}
    return result

def score_pair(a, b, truth):
    pa, pb = a['predictions'], b['predictions']
    return {'a_accuracy': a['accuracy'], 'b_accuracy': b['accuracy'], 'b_minus_a_accuracy_percentage_points': (b['correct'] - a['correct']) * 100 / len(truth), 'b_minus_a_cross_entropy': b['mean_cross_entropy'] - a['mean_cross_entropy'], 'different_predictions': sum(u != v for u, v in zip(pa, pb)), 'a_correct_b_wrong': sum(u == t and v != t for u, v, t in zip(pa, pb, truth)), 'a_wrong_b_correct': sum(u != t and v == t for u, v, t in zip(pa, pb, truth)), 'both_wrong_different_class': sum(u != v and u != t and v != t for u, v, t in zip(pa, pb, truth))}
