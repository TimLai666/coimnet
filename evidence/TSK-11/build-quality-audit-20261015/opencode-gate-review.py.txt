command = ['opencode', 'run', '--model', 'opencode/big-pickle', '--agent', 'plan', '--format', 'json', prompt]
started = datetime.datetime.now(datetime.timezone.utc)
record = {'command': command, 'started_at': started.isoformat(), 'timeout_seconds': 55,
          'model_verified_available_by_opencode_models': True,
          'host': 'Codex', 'report': 'runs/.tmp/build-quality-audit-20261015/gate-review.md'}
with (e / 'opencode-gate-review.log').open('xb') as f:
    try:
        result = subprocess.run(command, cwd=r, stdout=f, stderr=subprocess.STDOUT, timeout=55)
        record['exit_code'] = result.returncode
    except subprocess.TimeoutExpired:
        record['exit_code'] = None
        record['actual_error'] = 'subprocess.TimeoutExpired after 55 seconds; direct child killed and reaped'
raw = (e / 'opencode-gate-review.log').read_bytes()
record['log_sha256'] = hashlib.sha256(raw).hexdigest()
record['duration_seconds'] = (datetime.datetime.now(datetime.timezone.utc) - started).total_seconds()
record['report_exists'] = (r / record['report']).exists()
with (e / 'opencode-gate-review.json').open('x') as f:
    json.dump(record, f, indent=2)
    f.write('\n')
print(json.dumps({k: v for k, v in record.items() if k != 'command'}))
