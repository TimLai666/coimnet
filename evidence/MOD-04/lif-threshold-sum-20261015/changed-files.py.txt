EXPLICIT_PATHS = (

FOREIGN_PREFIXES = (
    ("evidence", "LRN-06"),
    ("evidence", "LRN-09"),
)


def is_excluded(relative: Path) -> bool:
    if "AGENTS.md" in relative.parts:
        return True
    return relative.parts[:2] in FOREIGN_PREFIXES


def collect_evidence_files() -> list[Path]:
    result: list[Path] = []
    for base, dirs, files in os.walk(EVIDENCE, topdown=True, followlinks=False):
        base_path = Path(base)
        dirs[:] = [name for name in dirs if not (base_path / name).is_symlink()]
        for name in files:
            path = base_path / name
            if path.is_symlink() or not stat.S_ISREG(path.stat().st_mode):
                continue
            relative = path.relative_to(ROOT)
            if not is_excluded(relative):
                result.append(path)
    return sorted(result, key=lambda path: str(path.resolve()))


def collect_files() -> list[Path]:
    explicit = [ROOT / relative for relative in EXPLICIT_PATHS]
    for path in explicit:
        if path.is_symlink() or not path.is_file():
            raise SystemExit(f"required ticket file is missing or non-regular: {path}")

    files: list[Path] = []
    seen: set[Path] = set()
    for path in explicit + collect_evidence_files():
        resolved = path.resolve()
        if resolved not in seen:
            seen.add(resolved)
            files.append(resolved)
    return files


def artifact_kind(path: Path) -> str:
    name = path.name
    if name.endswith(".log.gz"):
        return "壓縮 log"
    if name.endswith(".log"):
        return "log"
    if name.endswith(".json"):
        return "receipt"
    if name.endswith(".md"):
        return "文件"
    if name.endswith(".go.txt"):
        return "來源快照"
    if name.endswith(".py.txt"):
        return "腳本來源"
    if name.endswith(".gz"):
        return "壓縮證據"
    return "驗收證據"

def render(files: list[Path]) -> str:
    rows = ["# Ticket 112 實際檔案", "", "| 檔案 | 變更摘要 |", "| --- | --- |"]
    for path in files:
        absolute = path.as_posix()
        rows.append(f"| [{absolute}]({absolute}) | {summary(path)} |")
    rows.append("")
    return "\n".join(rows)


def main() -> None:
    if OUTPUT.is_symlink() or (OUTPUT.exists() and not OUTPUT.is_file()):
        raise SystemExit("refusing non-regular inventory target")
    before = OUTPUT.read_bytes() if OUTPUT.exists() else None
    files = collect_files()
    if OUTPUT.resolve() not in files:
        files.append(OUTPUT.resolve())
        files[len(EXPLICIT_PATHS):] = sorted(files[len(EXPLICIT_PATHS):], key=lambda path: str(path.resolve()))
    content = render(files).encode("utf-8")
    if before is None:
        with OUTPUT.open("xb") as handle:
            handle.write(content)
    else:
        if OUTPUT.is_symlink() or OUTPUT.read_bytes() != before:
            raise SystemExit("inventory target changed before replacement")
        OUTPUT.write_bytes(content)
    if [path.as_posix() for path in collect_files()] != [path.as_posix() for path in files]:
        raise SystemExit("file inventory changed while rendering")


if __name__ == "__main__":
    main()
