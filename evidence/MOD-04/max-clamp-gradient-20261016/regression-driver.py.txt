"""Run the final frozen permanent tests and existing effect-rule checks."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys

if not __debug__:
    raise SystemExit("optimized Python cannot skip gates")
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
source = json.loads((E / "formal-source.json").read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()

def gate():
    paths = {p for p in subprocess.check_output(["git", "ls-files", "-co", "--exclude-standard", "-z"], cwd=R).decode().split("\0") if p and (p.endswith(".go") or p in ("go.mod", "go.sum"))}
    assert len(source) == 835 and paths == set(source)
    assert all(sha(R / p) == h for p, h in source.items())

for mode, flags in [("ordinary", []), ("race", ["-race"])]:
    for label, pattern, package in [("regression-r2-", "^TestChemicalMax", "./learning"), ("modulation-rules-", "^TestEffectGradient", "./modulation")]:
        gate()
        subprocess.run([sys.executable, str(E / "run-command.py.txt"), label + mode, "go", "test", *flags, "-json", "-count=1", "-run", pattern, package], cwd=R, check=True)
        gate()
