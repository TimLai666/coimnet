"""Publish bounded acceptance metadata only after actual full gate success."""
import datetime
import gzip
import hashlib
import json
from pathlib import Path

if not __debug__:
    raise SystemExit("optimized Python cannot skip gates")
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
source = read(E / "formal-source.json")
assert len(source) == 835 and all(sha(R / p) == h for p, h in source.items())
engineering = read(E / "engineering-verification.json")
diagnostic = read(E / "diagnostic-verification.json")
regression = read(E / "regression-verification.json")
mutation = read(E / "mutation-verification.json")
assert engineering["all_passed"] and engineering["source_count"] == 835
assert diagnostic["all_passed"] and regression["all_passed"] and mutation["all_rejected"]
for item in engineering["steps"]:
    assert item["exit_code"] == 0 and not item["acceptance_errors"] and sha(E / item["log"]) == item["log_sha256"]
adoption = {}
for mode in ["unit", "race"]:
    item = next(v for v in engineering["steps"] if v["name"] == mode)
    raw = gzip.decompress((E / item["log"]).read_bytes())
    events = [json.loads(line) for line in raw.splitlines() if line.startswith(b"{")]
    selected = {v["Package"] + ":" + v["Test"]: v["Action"] for v in events if v.get("Action") in ["pass", "fail", "skip"] and v.get("Test", "").startswith("TestNeutralGain")}
    assert len(selected) == 21 and set(selected.values()) == {"pass"}
    adoption[mode] = selected
with (E / "adoption-current.json").open("x") as handle:
    json.dump({"all_passed": True, "effective_gain_bounds_include_one": True, "current_effects_sha256": source["modulation/effects.go"], "construction_and_restore_refuse_excluded_ranges": True, "current_full_suite_events": adoption, "original_adoption": "../neutral-gain-adoption-20261014/verification.json"}, handle, indent=2)
    handle.write("\n")
verification = {
    "requirement": ["LRN-02", "MOD-04"],
    "verified_at": datetime.datetime.now(datetime.timezone.utc).isoformat(),
    "engineering_passed": True, "diagnostic_passed": True,
    "production_changes": False, "new_permanent_test_files": 1,
    "confirmed_new_production_defects": 0,
    "source_count": 835,
    "reproduction_command": "audit-driver.py.txt; verify-diagnostic.py.txt; regression-driver.py.txt; verify-regression.py.txt; mutation-check.py.txt; engineering-r2.py.txt",
    "environment": read(E / "environment.json"),
    "input_fingerprints": {"base_commit": read(E / "baseline.json")["head"], "formal_source": "formal-source.json", "formal_source_sha256": sha(E / "formal-source.json"), "diagnostic": "frozen-diagnostic.json", "diagnostic_sha256": sha(E / "frozen-diagnostic.json"), "command_environment": "command-environment.json"},
    "observed_result": {"diagnostic": diagnostic["builds"], "permanent_tests": regression["builds"], "mutations": "mutation-verification.json", "engineering": "engineering-verification.json", "adopted_neutral_gain_rule": "adoption-current.json"},
    "retained_failed_diagnostics": {"empty_slots": ["skeleton-empty.json", "permanent-empty.json"], "diagnostic_unused_import": "skeleton-build.json", "incorrect_new_test_shape_oracle": "test-oracle-correction.json", "sandbox_loopback_denial": "sandbox-limitation.json"},
    "test_log": ["audit-ordinary.log", "audit-race.log", "regression-r2-ordinary.log", "regression-r2-race.log", "engineering-r2-unit.log.gz", "engineering-r2-race.log.gz"],
    "root_review": "root-review.md", "readonly_review": "readonly-review.md",
    "limits": ["Bounded artificial CPU scalar fixtures, not whole-brain or real-task quality evidence", "No new independent full LIF surrogate gradient reference or hard-spike finite difference claim", "First-step chemistry proofs only; no claim every later row has the same clamp state", "No cross-build bitwise or checkpoint-bundle migration claim", "No biological calibration or absolute zero-bug proof"],
    "remaining_delivery": ["final governance", "preserve necessary evidence", "owned temporary cleanup", "commit/push/remote readback"],
    "acceptance_complete": False,
}
with (E / "verification.json").open("x") as handle:
    json.dump(verification, handle, indent=2)
    handle.write("\n")
steps = {item["name"]: item for item in engineering["steps"]}
unit, race = steps["unit"]["test_events"], steps["race"]["test_events"]
report = f"""# 最大受體作用與夾限的訓練驗證

負係數、最大受體作用與作用上下限的指定組合通過驗證，這輪沒有確認新的正式程式缺陷。已修正 ticket21 的數學說明，並加入永久整合測試。正式模型計算、API、保存格式及依賴保持原樣。

格式、build、vet、固定 Insyra v0.3.4、模組校驗與工程治理檢查通過。原有各6,477個終端測試事件保持，只新增各65個通過。完整結果見[驗收](verification.json)、[工程結果](engineering-verification.json)、[診斷](diagnostic-verification.json)、[永久測試](regression-verification.json)與[錯誤植入驗證](mutation-verification.json)。

"""
with (E / "README.md").open("x") as handle:
    handle.write(report)
p = R / "docs/requirements-status.json"
original = p.read_bytes()
document = json.loads(original)
relative = "../" + str(E.relative_to(R)) + "/verification.json"
seen = set()
for requirement in document["requirements"]:
    if requirement["id"] in verification["requirement"]:
        assert requirement["status"] == "passed"
        assert relative not in requirement["evidence"]
        requirement["evidence"].append(relative)
        seen.add(requirement["id"])
assert seen == set(verification["requirement"])
assert p.read_bytes() == original
p.write_text(json.dumps(document, ensure_ascii=False, indent=2) + "\n")
print(json.dumps({"engineering_passed": True, "diagnostic_passed": True, "requirements_evidence_added": sorted(seen), "acceptance_complete": False}), flush=True)
