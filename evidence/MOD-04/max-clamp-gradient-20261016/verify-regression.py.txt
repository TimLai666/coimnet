"""Check all permanent cases ran, including finite difference coverage logs."""
import hashlib
import itertools
import json
from pathlib import Path
import re

if not __debug__:
    raise SystemExit("optimized Python cannot skip checks")
E = Path(__file__).resolve().parent
common = ["open-positive", "open-negative", "gain-upper", "gain-lower", "offset-upper", "offset-lower"]
training = "TestChemicalMaxRecomputeSignedAndClampedTraining"
reference = "TestChemicalMaxContinuousGradientReference"
expected = {"TestChemicalMaxNegativeScaleUsesRawOccupancy", training, reference}
for core in ["continuous", "lif"]:
    for profile in common + (["threshold-upper", "threshold-lower"] if core == "lif" else []):
        for filled, segment in itertools.product([1, 3], [1, 5]):
            expected.add(f"{training}/{core}/{profile}/filled{filled}/segment{segment}")
expected.update(reference + "/" + profile for profile in common)
assert len(expected) == 65
builds = {}
for mode in ["ordinary", "race"]:
    raw = (E / ("regression-r2-" + mode + ".log")).read_bytes()
    receipt = json.loads((E / ("regression-r2-" + mode + ".json")).read_text())
    assert receipt["exit_code"] == 0 and hashlib.sha256(raw).hexdigest() == receipt["log_sha256"]
    rows = [json.loads(line) for line in raw.splitlines()]
    terminal = {r["Test"]: r["Action"] for r in rows if r.get("Test") and r.get("Action") in ["pass", "fail", "skip"]}
    assert set(terminal) == expected and all(v == "pass" for v in terminal.values())
    counts = {}
    for row in rows:
        text = row.get("Output", "")
        match = re.search(r"profile=([^ ]+) checked_coordinates=(\d+) informative=(\d+)", text)
        if match:
            profile, checked, informative = match.groups()
            assert profile not in counts and int(checked) == 18 and int(informative) > 0
            counts[profile] = {"checked": int(checked), "informative": int(informative)}
    assert set(counts) == set(common)
    direct = (E / ("modulation-rules-" + mode + ".log")).read_text()
    rules = [json.loads(line) for line in direct.splitlines()]
    finals = [r for r in rules if r.get("Test") and r.get("Action") in ["pass", "fail", "skip"]]
    assert finals and all(r["Action"] == "pass" for r in finals)
    tops = {r["Test"] for r in finals if "/" not in r["Test"]}
    assert {"TestEffectGradientClampsHaveZeroSlopeAtAndOutsideBounds", "TestEffectGradientMaxChoosesFirstPositiveMaximum", "TestEffectGradientFiniteDifferenceAwayFromClampsAndTies"} <= tops
    builds[mode] = {"permanent_test_pass_events": len(terminal), "training_leaf_cases": 56, "finite_difference_coordinates": sum(v["checked"] for v in counts.values()), "reference_profiles": counts, "direct_effect_rule_pass_events": len(finals)}
with (E / "regression-verification.json").open("x") as f:
    json.dump({"all_passed": True, "builds": builds}, f, indent=2)
    f.write("\n")
print(json.dumps(builds), flush=True)
