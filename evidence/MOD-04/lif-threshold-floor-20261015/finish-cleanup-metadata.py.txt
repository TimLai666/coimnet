import json

p = e / "verification.json"
old = p.read_bytes()

d["test_log"].append("final-governance.log")
d["cleanup_receipt"] = "cleanup.json"
d["final_governance_receipt"] = "final-governance.json"
assert p.read_bytes() == old
p.write_text(json.dumps(d, indent=2) + "\n")
