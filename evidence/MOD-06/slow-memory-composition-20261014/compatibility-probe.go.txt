package learning_test

import (
	"context"
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"testing"

	"github.com/TimLai666/coimnet/learning"
)

func TestCompositionCompatibilityReceipt(t *testing.T) {
	for _, profile := range []string{"continuous", "lif", "mixed"} {
		for _, sign := range []int8{0, 1, -1} {
			for _, rate := range []float64{.25, 1} {
				for _, retain := range []float64{0, .5, 1} {
					a := compositionIndividual(t, profile, sign, true, .75, .125, .25)
					r, err := a.Consolidate(context.Background(), learning.ConsolidationTrigger{Receptor: 0, Threshold: 0, Rate: rate, Retain: retain})
					if err != nil {
						t.Fatal(err)
					}
					y, err := a.Advance(context.Background(), [][]float64{{1.25}, {1.25}, {1.25}})
					if err != nil {
						t.Fatal(err)
					}
					bytes, err := json.Marshal(map[string]any{"report": r, "output": y, "snapshot": a.Snapshot()})
					if err != nil {
						t.Fatal(err)
					}
					fmt.Printf("COMPOSITION_COMPAT %s/%d/%g/%g %x\n", profile, sign, rate, retain, sha256.Sum256(bytes))
				}
			}
		}
	}
}
