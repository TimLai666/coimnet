"""Assemble this audit from completed, hash-checked command receipts."""
import datetime,gzip,hashlib,json
from pathlib import Path
R=Path(__file__).resolve().parents[3];E=Path(__file__).resolve().parent
read=lambda p:json.loads(p.read_text())
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
engineering=read(E/'engineering-verification.json');diagnostic=read(E/'diagnostic-verification.json')
assert engineering['all_passed'] and diagnostic['all_passed']
source=read(E/'formal-source.json');assert len(source)==839
for p,h in source.items():assert sha(R/p)==h,p
neutral={}
for mode in ['unit','race']:
 step=next(x for x in engineering['steps'] if x['name']==mode)
 assert step['exit_code']==0 and not step['acceptance_errors'] and not step['changed_events'] and step['prior_event_count']==6912
 assert len(step['added_events'])==253 and all(x['action']=='pass' for x in step['added_events'])
 packed=E/step['log'];assert sha(packed)==step['log_sha256'];raw=gzip.decompress(packed.read_bytes());assert hashlib.sha256(raw).hexdigest()==step['raw_sha256']
 events=[json.loads(line) for line in raw.decode().splitlines() if line.startswith('{')]
 neutral[mode]={x['Package']+':'+x['Test']:x['Action'] for x in events if x.get('Test','').startswith('TestNeutralGain') and x.get('Action') in ['pass','fail','skip']}
 assert len(neutral[mode])==21 and set(neutral[mode].values())=={'pass'}
with (E/'adoption-current.json').open('x') as f:
 json.dump({'all_passed':True,'effective_gain_bounds_include_one':True,'construction_and_restore_refuse_excluded_ranges':True,'current_effects_sha256':sha(R/'modulation/effects.go'),'current_full_suite_events':neutral},f,indent=2);f.write('\n')
counts={x['name']:{k:x[k] for k in ['test_events','package_events','prior_event_count'] if k in x} for x in engineering['steps'] if x['name'] in ['unit','race','governance']}
result={'requirement':'MOD-02','related_requirements':['MOD-02','COR-04','LRN-02'],'verified_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'engineering_passed':True,'diagnostic_passed':True,'production_changes':False,'source_count':839,'confirmed_new_production_defects':0,'reproduction_command':{'ordinary':read(E/'reference-final-ordinary.json')['command'],'race':read(E/'reference-final-race.json')['command'],'engineering':['python3','evidence/MOD-02/homeostasis-chemical-gradient-20261016/engineering.py.txt']},'environment':{**read(E/'environment.json'),'go':(E/'go-version.log').read_text().strip(),'command_environment':'command-environment.json'},'input_fingerprints':{'formal_source':'formal-source.json','formal_source_sha256':sha(E/'formal-source.json'),'frozen_checks':'frozen-checks.json','frozen_checks_sha256':sha(E/'frozen-checks.json'),'original_source':'source-before.json'},'observed_result':{'reference':diagnostic['modes'],'red_and_wrong_algorithms':'diagnostic-verification.json','engineering':counts,'approved_gain_policy':'adoption-current.json'},'test_log':['red-hand-corrected.log','reference-final-ordinary.log','reference-final-race.log','wrong-tangent-ordinary.log','wrong-tangent-race.log','missing-refractory-ordinary.log','missing-refractory-race.log','engineering-unit.log.gz','engineering-race.log.gz','engineering-governance.log'],'retained_corrections':['test-correction.json'],'root_review':'root-review.md','limits':diagnostic['limits']+['No absolute zero-bug proof'],'remaining_delivery':['final metadata governance','necessary work preservation','owned temporary cleanup','Git commit/push/readback'],'acceptance_complete':False}
with (E/'verification.json').open('x') as f:json.dump(result,f,indent=2);f.write('\n')
print(json.dumps({'engineering':counts,'public_reference_cases_per_mode':80,'trainer_routes_per_mode':160,'source_count':839,'remaining_delivery':result['remaining_delivery']}))
