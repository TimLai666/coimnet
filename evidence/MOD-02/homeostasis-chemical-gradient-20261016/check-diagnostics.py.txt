"""Verify frozen checks, exact legacy records and isolated algorithm rejection."""
import collections,hashlib,itertools,json,math,platform,re
from pathlib import Path
R=Path(__file__).resolve().parents[3];E=Path(__file__).resolve().parent
read=lambda p:json.loads(p.read_text());sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
for p,h in read(E/'formal-source.json').items():assert sha(R/p)==h,p
frozen=read(E/'frozen-checks.json')
for p,h in frozen['tests'].items():assert sha(R/p)==h,p

def command(label,code):
 receipt=read(E/(label+'.json'));p=E/receipt['log'];assert receipt['exit_code']==code and sha(p)==receipt['log_sha256'],label
 events=[]
 for line in p.read_text().splitlines():
  try:events.append(json.loads(line))
  except json.JSONDecodeError:pass
 return receipt,events

def terminal(events):
 pairs=[(x['Test'],x['Action']) for x in events if x.get('Test') and x.get('Action') in ['pass','fail','skip']]
 assert len(pairs)==len(dict(pairs));return dict(pairs)

def legacy(events):
 return [x['Output'].split('legacy-record:',1)[1].rstrip('\r\n') for x in events if 'legacy-record:' in x.get('Output','')]

red,red_events=command('red-hand-corrected',1)
assert 'reference fixture LIF homeostasis is unsupported' in (E/red['log']).read_text()
prefix='TestChemicalHomeostasisPublicReference/'
expected_cases={prefix+'/'.join([core,source,mix,profile,'implicit'+implicit]) for core,source,mix,profile,implicit in itertools.product(['mixed','lif'],['controller','neural'],['sum','max'],['open-positive','open-negative','gain-upper','threshold-upper','floor'],['false','true'])}
expected_routes={name+'/train'+str(seg) for name in expected_cases for seg in ([0,1,3] if name.startswith(prefix+'lif/') else [0])}
assert len(expected_cases)==80 and len(expected_routes)==160
modes={}
for mode in ['ordinary','race']:
 receipt,events=command('reference-final-'+mode,0);ts=terminal(events)
 assert len(ts)==254 and set(ts.values())=={'pass'},collections.Counter(ts.values())
 new={k:v for k,v in ts.items() if k.startswith('TestChemicalHomeostasis')};assert len(new)==253
 assert {n for n in ts if n.startswith(prefix) and '/train' not in n}==expected_cases
 assert {n for n in ts if n.startswith(prefix) and '/train' in n}==expected_routes
 assert {n for n in ts if n.startswith('TestChemicalHomeostasisHandReference/')}=={'TestChemicalHomeostasisHandReference/refractory0','TestChemicalHomeostasisHandReference/refractory1'}
 assert len([n for n in ts if n.startswith('TestChemicalHomeostasisDisabledReference/')])==8
 controls=[float(m.group(1)) for x in events if (m:=re.search(r'homeostasis gradient control max_abs=([0-9.eE+-]+)',x.get('Output','')))]
 assert len(controls)==16 and all(math.isfinite(v) and v>1e-9 for v in controls)
 _,before=command('legacy-before-'+mode+'-corrected',0);old=legacy(before);now=legacy(events)
 assert len(old)==7 and old==now
 errors={key:max(float(m.group(1)) for x in events if x.get('Test','').startswith('TestChemicalHomeostasis') and 'reference=' in x.get('Output','') and (m:=re.search(key+r'=([0-9.eE+-]+)',x['Output']))) for key in ['max_gradient_abs','max_gradient_relative','max_prediction_abs']}
 mutations={}
 for name,test,pattern,want in [('wrong-tangent','refractory0',r'hand tangent\[0\]\[0\]=([0-9.eE+-]+) want0.0625',17/1296),('missing-refractory','refractory1',r'rate\[2\]\[1\]=([0-9.eE+-]+) want0.25',0.0)]:
  mr,me=command(name+'-'+mode,1);mt=terminal(me);leaf='TestChemicalHomeostasisHandReference/'+test
  assert mt.get(leaf)=='fail' and len(mt)==2 and set(mt.values())=={'fail'},mt
  matches=[re.search(pattern,x.get('Output','')) for x in me];values=[float(m.group(1)) for m in matches if m]
  assert len(values)==1 and math.isclose(values[0],want,rel_tol=0,abs_tol=1e-15),(name,values)
  mutations[name]={'rejected':True,'actual_value':values[0],'receipt':name+'-'+mode+'.json','leaf':leaf}
 modes[mode]={'terminal_events':len(new),'public_leaf_cases':len(expected_cases),'trainer_routes':len(expected_routes),'paired_controls':len(controls),'hand_cases':2,'disabled_cases':8,'legacy_records':7,'legacy_records_sha256':hashlib.sha256('\n'.join(now).encode()).hexdigest(),'legacy_records_preserved_within_build':True,'control_min_abs':min(controls),'maximum_reference_error':errors,'mutations':mutations,'command_receipt':'reference-final-'+mode+'.json'}
result={'all_passed':True,'modes':modes,'red':{'receipt':'red-hand-corrected.json','scope':'private reference rejects enabled homeostasis; not a production defect'},'checks_frozen':True,'production_changes':False,'existing_source_changed':['learning/chemical_mixed_reference_test.go'],'new_source':['learning/chemical_homeostasis_reference_test.go'],'scope_review':'scope-review.json','limits':['Four steps, two nodes, four edges, one chemistry source/channel, ordinary finite fixtures','Independent full-window gradients; pure LIF only for recompute segments 1 and 3','First Adam update with accumulate=2 and no clipping, schedule, masks, sharing or weight decay','No persistent neural-state, positive truncation window, true-task quality or biological calibration conclusion']}
with (E/'diagnostic-verification.json').open('x') as f:json.dump(result,f,indent=2);f.write('\n')
with (E/'environment.json').open('x') as f:json.dump({'platform':platform.platform(),'machine':platform.machine(),'baseline':read(E/'baseline.json')['head'],'command_settings':'command-environment.json'},f,indent=2);f.write('\n')
print(json.dumps(result))
