import collections
import datetime
import gzip
import hashlib
import json
import pathlib
import re
import subprocess

root = pathlib.Path.cwd()
evidence = root / 'evidence/MOD-02/chemical-boundaries-20261013'

def read(name):
    return json.loads((evidence / (name + '.json')).read_text())

def write(name, value):
    with (evidence / (name + '.json')).open('x') as out:
        json.dump(value, out, indent=2, ensure_ascii=False)
        out.write('\n')

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def raw_log(receipt):
    packed = (root / receipt['log']).read_bytes()
    assert sha(packed) == receipt['log_sha256'], receipt['log']
    raw = gzip.decompress(packed)
    assert sha(raw) == receipt['raw_log_sha256'], receipt['log']
    assert len(raw) == receipt['raw_log_bytes'], receipt['log']
    return raw

source = read('source-final')
assert len(source) == 782
assert all(sha((root / path).read_bytes()) == digest for path, digest in source.items())
frozen = read('frozen-tests-final')
assert len(frozen) == 7
assert all(source[path] == digest for path, digest in frozen.items())
baseline = read('baseline')
for path in ('go.mod', 'go.sum'):
    old = subprocess.check_output(['git', 'show', baseline['head'] + ':' + path])
    assert old == (root / path).read_bytes(), path

passed_commands = ['final-build', 'final-vet', 'dependencies-native-cache',
                   'green-targeted', 'green-next-all', 'compatibility-before',
                   'compatibility-after', 'compatibility-final', 'cli-build-final',

expected_red = ['red-chemistry-source', 'red-expression', 'red-next-chemical',

for name in passed_commands:

assert b'imported and not used' in raw_log(read('red-next-retry'))
assert read('full-race')['exit_code'] != 0  # superseded, interrupted run

tests = sorted(name for path in frozen for name in re.findall(
    r'^func (Test\w+)\(t \*testing\.T\)', (root / path).read_text(), re.M))
assert len(tests) == 22
summary = {}
for kind in ('unit', 'race'):
    receipt = read('full-final-' + kind)
    raw = raw_log(receipt)
    assert b'WARNING: DATA RACE' not in raw
    events = [json.loads(line) for line in raw.splitlines() if line.startswith(b'{')]
    counts = collections.Counter((e['Action'], 'test' if 'Test' in e else 'package') for e in events)
    assert counts['fail', 'test'] == counts['fail', 'package'] == 0
    assert counts['pass', 'package'] == 58
    passed = {e['Test'] for e in events if e['Action'] == 'pass' and 'Test' in e}
    assert all(name in passed for name in tests)
    skips = [e for e in events if e['Action'] == 'skip']
    for skipped in skips:
        skipped['output'] = [e['Output'].strip() for e in events
                             if e['Action'] == 'output'
                             and e.get('Package') == skipped.get('Package')
                             and e.get('Test') == skipped.get('Test')]
    summary[kind] = dict(passed_packages=counts['pass', 'package'],
                         passed_test_events=counts['pass', 'test'],
                         skipped_test_events=counts['skip', 'test'],
                         skipped_packages=counts['skip', 'package'],
                         skips=skips, log=receipt['log'])
write('suite-summary-final', summary)
