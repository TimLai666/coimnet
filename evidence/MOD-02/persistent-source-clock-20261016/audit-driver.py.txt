"""Execute the reviewed frozen public diagnostic; no formal source mutations."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys

if not __debug__:
    raise SystemExit("optimized Python cannot skip audit gates")
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
W = R / "runs/.tmp/persistent-source-clock-20261016"
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
source = json.loads((E / "source-before.json").read_text())
frozen = json.loads((E / "frozen-diagnostic.json").read_text())["files"]

def gate():
    paths = {p for p in subprocess.check_output(["git", "ls-files", "-co", "--exclude-standard", "-z"], cwd=R).decode().split("\0") if p and (p.endswith(".go") or p in ("go.mod", "go.sum"))}
    assert paths == set(source), "formal source path set changed"
    diagnostics = {str(p.relative_to(R)) for p in (W / "probe").glob("*.go")}
    assert diagnostics == set(frozen), "diagnostic source path set changed"
    assert W.stat().st_ino == json.loads((E / "baseline.json").read_text())["work_inode"]
    assert all(sha(R / p) == h for p, h in {**source, **frozen}.items())
    assert all(sha(Path(p)) == h for p, h in json.loads((E / "foreign-before.json").read_text()).items())

def run(label, command):
    gate()
    subprocess.run([sys.executable, str(E / "run-command.py.txt"), label, *command], cwd=R, check=True)
    gate()

files = sorted(str(p.relative_to(R)) for p in (W / "probe").glob("*.go"))
run("diagnostic-format", ["gofmt", "-l", *files])
assert not (E / "diagnostic-format.log").read_bytes().strip()
for mode, flags in [("ordinary", []), ("race", ["-race"])]:
    binary = W / ("probe-" + mode)
    run("build-" + mode, ["go", "build", *flags, "-o", str(binary), "./" + str((W / "probe").relative_to(R))])
    resume = W / ("resume-" + mode)
    run("audit-" + mode, [str(binary), "audit", str(resume)])
    report = json.loads((E / ("audit-" + mode + ".log")).read_text())
    assert report["all_passed"] and len(report["matrix"]) == 48 and report["routes"] == 1920 and report["atomic_cases"] == 12 and report["atomic_rejections"] == 48
    inputs = report["resume_inputs"]
    assert len(inputs) == 48 and len(set(inputs)) == 48
    for i, name in enumerate(inputs):
        p = Path(name)
        if not p.is_absolute():
            p = R / p
        assert p.is_file() and p.resolve().is_relative_to(resume.resolve())
        run("resume-" + mode + "-" + str(i), [str(binary), "resume", str(p)])
run("framework-build", ["go", "build", "./..."])
run("framework-vet", ["go", "vet", "./..."])
run("diagnostic-vet", ["go", "vet", "./" + str((W / "probe").relative_to(R))])
run("module-verify", ["go", "mod", "verify"])
run("module-selection", ["go", "list", "-m", "github.com/HazelnutParadise/insyra"])
assert (E / "module-selection.log").read_text().strip() == "github.com/HazelnutParadise/insyra v0.3.4"
pattern = "^Test(NoTelemetry|NoTelemetryProjectTempBoundary|NoTelemetrySourceErrorsRemainVisible|FormatIndex|FormatIndexControls|FormatIndexGoLiteralsAndScopes|FormatIndexNoGitAndReadErrors|GovernanceRequirementIDsMatchHandoff|GovernanceStatusValuesAreDeclared|GovernancePassedRequirementsHaveEvidence|GovernanceTicketsHaveDatedRootDecisions)$"
run("governance-before-metadata", ["go", "test", "-json", "-count=1", "-run", pattern, "."])
events = [json.loads(x) for x in (E / "governance-before-metadata.log").read_bytes().splitlines()]
assert sum(x.get("Test") is not None and x.get("Action") == "pass" for x in events) == 31
assert not any(x.get("Action") == "fail" for x in events)
run("closure-binary", ["go", "test", "-c", "-o", str(W / "final-root.test"), "."])
gate()
print(json.dumps({"all_commands_passed": True, "formal_source_entries": len(source), "ordinary_race_cases_each": 48, "ordinary_race_routes_each": 1920, "new_process_resumes_each": 48}))
