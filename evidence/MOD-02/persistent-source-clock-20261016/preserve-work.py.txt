import hashlib,json,tarfile
from pathlib import Path
R=Path(__file__).resolve().parents[3];E=Path(__file__).resolve().parent;W=R/'runs/.tmp/persistent-source-clock-20261016'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
assert W.stat().st_ino==57486054 and not W.is_symlink()
v=json.loads((E/'verification.json').read_text());assert v['engineering_passed'] and v['diagnostic_passed']
assert json.loads((E/'final-governance-v2.json').read_text())['passed_events']==31

files={};binaries={}
for p in sorted(W.rglob('*')):

archive=E/'preserved-work.tar.gz';assert not archive.exists()
with tarfile.open(archive,'w:gz') as h:

 for name,item in files.items():
  b=h.extractfile(name).read();assert len(b)==item['bytes'] and hashlib.sha256(b).hexdigest()==item['sha256']
for name,value in [('work-preservation.json',{'archive':archive.name,'archive_bytes':archive.stat().st_size,'archive_sha256':sha(archive),'files':files}),('discarded-binaries.json',binaries)]:
 (E/name).open('x').write(json.dumps(value,indent=2)+'\n')
print(json.dumps({'necessary_files':len(files),'archive_bytes':archive.stat().st_size,'discarded_reproducible_binaries':len(binaries)}))
