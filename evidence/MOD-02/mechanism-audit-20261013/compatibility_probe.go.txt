package main
import("context";"encoding/json";"math";"os";"github.com/TimLai666/coimnet/tasks/ocr";"github.com/TimLai666/coimnet/tasks/ocr/glyphs";"github.com/TimLai666/coimnet/tasks/asr/audio")
func main(){
 result:=map[string]interface{}{}
 for _,profile:=range []string{"columns","aligned-columns","single-character"}{
 c:=ocr.RecognizerConfig{Height:2,Hidden:4,Alphabet:[]rune{'a','b'},LearningRate:.03,Seed:91,InputProfile:profile};if profile=="single-character"{c.CharacterWidth=3}
 r,err:=ocr.NewRecognizer(c);if err!=nil{panic(err)}
 image:=glyphs.Image{Width:3,Height:2,Pixels:[]float64{0,1,.25,.5,1,0}}
 inputs,err:=r.Inputs(image);if err!=nil{panic(err)}
 report,err:=r.TrainLine(context.Background(),image,"a");if err!=nil{panic(err)}
 result[profile]=map[string]interface{}{"inputs":inputs,"report":report,"snapshot":r.Snapshot()}
 }
 c:=audio.FrontEndConfig{Window:512,Hop:128,Mels:20,FMin:0,FMax:8000,Floor:1e-10}
 samples:=make([]float64,768);for i:=range samples{samples[i]=.25*math.Sin(2*math.Pi*440*float64(i)/16000)}
 stft,err:=audio.STFT(samples,c);if err!=nil{panic(err)}
 mel,report,err:=audio.LogMel(samples,16000,c);if err!=nil{panic(err)}
 result["audio"]=map[string]interface{}{"stft":stft,"mel":mel,"report":report}
 if err:=json.NewEncoder(os.Stdout).Encode(result);err!=nil{panic(err)}
}
