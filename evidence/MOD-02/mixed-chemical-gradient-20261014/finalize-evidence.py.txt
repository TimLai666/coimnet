def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def save(name,value):
    p=evidence/name
    assert not p.exists(),str(p)
    p.write_text(json.dumps(value,ensure_ascii=False,indent=2)+"\n")
def archive(path,name):
    out=evidence/(name+".gz")
    assert not out.exists()
    raw=path.read_bytes()
    out.write_bytes(gzip.compress(raw,mtime=0))
    assert gzip.decompress(out.read_bytes())==raw
    return {"log":out.name,"log_sha256":hashlib.sha256(raw).hexdigest(),"archive_sha256":sha(out)}
def summary(path):
    events=[]
    for line in path.read_text().splitlines():
        if line.startswith("{"): events.append(json.loads(line))
    passes=[x for x in events if x.get("Action")=="pass"]
    return {"package_pass_count":sum(not x.get("Test") for x in passes),"test_pass_events":sum(bool(x.get("Test")) for x in passes),"top_level_pass_count":sum(bool(x.get("Test")) and "/" not in x["Test"] for x in passes),"skipped_tests":[{"package":x["Package"],"test":x.get("Test")} for x in events if x.get("Action")=="skip"],"failure_events":sum(x.get("Action")=="fail" for x in events),"log_sha256":sha(path)}
