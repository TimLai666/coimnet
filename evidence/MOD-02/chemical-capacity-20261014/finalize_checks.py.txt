import gzip
import hashlib
import json
from pathlib import Path
import re
import sys

root = Path(__file__).resolve().parents[3]
work = Path(__file__).resolve().parent
evidence = root / "evidence/MOD-02/chemical-capacity-20261014"


def sha(data):
    return hashlib.sha256(data).hexdigest()


def fresh(path, value):
    data = (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode()
    if path.exists():
        assert path.read_bytes() == data, str(path)
    else:
        with path.open("xb") as output:
            output.write(data)


def events(name):
    result = []
    for line in (work / (name + ".log")).read_text().splitlines():
        try:
            result.append(json.loads(line))
        except json.JSONDecodeError:
            continue
    return result


def tests(name):
    receipt = json.loads((work / (name + ".json")).read_text())
    assert receipt["exit_code"] == 0 and not receipt.get("timed_out"), name
    rows = events(name)
    assert not any(row.get("Action") == "fail" for row in rows), name
    passed = [row for row in rows if row.get("Action") == "pass"]
    return {
        "package_pass_count": sum(not row.get("Test") for row in passed),
        "test_pass_events": sum(bool(row.get("Test")) for row in passed),
        "top_level_pass_count": sum(bool(row.get("Test")) and "/" not in row["Test"] for row in passed),
        "skipped_tests": [{"package": row["Package"], "test": row["Test"]}
                          for row in rows if row.get("Action") == "skip" and row.get("Test")],
        "failure_events": 0,
        "log_sha256": sha((work / (name + ".log")).read_bytes()),
    }


def retain(name):
    receipt_path = work / (name + ".json")
    receipt = json.loads(receipt_path.read_text())
    raw = (work / (name + ".log")).read_bytes()
    assert sha(raw) == receipt["log_sha256"], name
    archive = evidence / (name + ".log.gz")
    if archive.exists():
        assert gzip.decompress(archive.read_bytes()) == raw, name
    else:
        with archive.open("xb") as output:
            output.write(gzip.compress(raw, mtime=0))
    receipt["retained_log"] = archive.name
    receipt["retained_log_sha256"] = sha(archive.read_bytes())
    fresh(evidence / (name + ".json"), receipt)
    return {"receipt": name + ".json", "exit_code": receipt["exit_code"],
            "duration_seconds": receipt["duration_seconds"],
            "log": archive.name, "log_sha256": sha(raw),
            "archive_sha256": sha(archive.read_bytes())}


def cases(name):
    rows = {}
    for line in (work / name).read_text().splitlines():
        match = re.fullmatch(r"case=(\S+) status=(\S+) err=(.*?) output=(.*)", line)
        if match:
            case, status, error, output = match.groups()
            rows[case] = {"status": status, "error": error,
                          "output": json.loads(output), "line": line}
    return rows

assert phase in ("full", "governance")

modules = []

    modules.append(value)
