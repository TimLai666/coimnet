import json

import sys

evidence = root / "evidence/MOD-02/chemical-capacity-20261014"

assert phase in ("initial", "governance")
summary = json.loads((evidence / "engineering-summary.json").read_text())
assert summary["unit"]["package_pass_count"] == summary["race"]["package_pass_count"] == 58


def replace(path, old, new):
    assert path.read_bytes() == old, str(path)
    path.write_bytes(new)


def encode(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode()


verification = evidence / "verification.json"
if phase == "initial":
    assert not verification.exists()
    receipts = []
    for name in summary["commands"]:

    value = {

        "observed_result": {
            "repaired_classes": ["source release vector", "controller activity grid", "chemical concentration grid", "effect node vectors"],
            "capacity": summary["capacity"], "capacity_green": summary["capacity_green"],

            "activity": summary["activity"], "compatibility": summary["compatibility"], "cli": summary["cli"],
            "source_integrity": summary["source_integrity"], "frozen_tests": summary["frozen_tests"],
            "dependencies": summary["dependencies"],
            "normal_mechanism_confirmed_design_mismatches": 0,
            "normal_mechanism_audit_scope": "declared small chemical fixtures and current design; complete mixed continuous/LIF reverse composition not independently verified",
            "normal_model_quality_improvement": "not measured",
        },
        "test_log": [receipt["log"] for receipt in summary["commands"].values()],
        "reviews": ["root-review.md", "adversarial-review.md", "mechanism-design-review.md"],
        "engineering_passed": True, "governance": {"state": "pending"},
        "cleanup": {"state": "pending"}, "delivery": {"state": "pending"},

            "Representability checks do not guarantee enough physical memory for every large legal declaration.",

    status = root / "docs/requirements-status.json"

    assert hashlib.sha256(old).hexdigest() == "a379fa5a452000d63cb6229e45e7eb8f5fb78a46813cf4ad36f0aadc2986d2a5"
    registry = json.loads(old)
    reference = "../evidence/MOD-02/chemical-capacity-20261014/verification.json"
    affected = 0
    for requirement in registry["requirements"]:
        if requirement["id"] in value["requirement"]:
            assert requirement["status"] == "passed" and reference not in requirement["evidence"]
            requirement["evidence"].append(reference)
            affected += 1
    assert affected == 3 and len(registry["requirements"]) == 91
    replace(status, old, encode(registry))

else:

    governance = json.loads((evidence / "governance-summary.json").read_text())

    assert governance["top_level_pass_count"] == 11 and governance["test_pass_events"] == 31

    value["test_log"].append("governance.log.gz")
    receipt = json.loads((evidence / "governance.json").read_text())
    value["reproduction_command"].append({"receipt": "governance.json", "command": receipt["command"], "exit_code": receipt["exit_code"]})
    value["ticket_acceptance"] = {"completed": 4, "total": 5}

    completed = 4
