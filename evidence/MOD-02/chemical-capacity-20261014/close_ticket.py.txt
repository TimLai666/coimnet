remote_revision = remote.stdout.split()[0]

sources = json.loads((evidence / "source-final.json").read_text())
queries = [primary + ":" + path for path in sources]
assert all("\n" not in query for query in queries)
result = subprocess.run(["git", "cat-file", "--batch"], cwd=root,
                        input=("\n".join(queries) + "\n").encode(),
                        stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=True)
position = 0
for path, fingerprint in sources.items():
    end = result.stdout.index(b"\n", position)
    header = result.stdout[position:end].split()
    assert len(header) == 3 and header[1] == b"blob", path
    size = int(header[2])
    position = end + 1
    data = result.stdout[position:position + size]
    assert hashlib.sha256(data).hexdigest() == fingerprint, path
    position += size
    assert result.stdout[position:position + 1] == b"\n"
    position += 1
assert position == len(result.stdout)


def create_json(path, value):
    assert not path.exists(), str(path)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def replace(path, old, text):
    assert path.read_bytes() == old, str(path)
    path.write_text(text)
