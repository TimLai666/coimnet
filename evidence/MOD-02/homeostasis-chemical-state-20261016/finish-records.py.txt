"""Append completed state-audit evidence, then run final metadata governance."""
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import time

R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
W = R / 'runs/.tmp/homeostasis-chemical-state-20261016'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
v = read(E / 'verification.json')
assert v['engineering_passed'] and v['diagnostic_passed']
assert v['observed_result']['engineering']['unit']['test_events'] == {'pass': 7915, 'skip': 23}
assert v['observed_result']['engineering']['race']['test_events'] == {'pass': 7914, 'skip': 24}
for p, h in read(E / 'formal-source.json').items():
    assert sha(R / p) == h, p
p = R / 'docs/requirements-status.json'
b = p.read_bytes()
assert hashlib.sha256(b).hexdigest() == read(E / 'original-records.json')[str(p)]['sha256']
status = json.loads(b)
assert len(status['requirements']) == 91
for item in status['requirements']:
    assert item['status'] == 'passed'
    if item['id'] in ['COR-04', 'MOD-02', 'STA-03']:
        link = '../evidence/MOD-02/homeostasis-chemical-state-20261016/verification.json'
        assert link not in item['evidence']
        item['evidence'].append(link)
assert p.read_bytes() == b
p.write_text(json.dumps(status, ensure_ascii=False, indent=2) + '\n')
p = R / 'docs/tickets/125-chemical-homeostasis-state.md'
b = p.read_bytes()
s = b.decode()
assert s.count('- [ ]') == 2
s = s.replace('- [ ]', '- [x]', 1)
s += '\n840 份來源的完整一般／競態各 58 套件通過，7,915／7,914 pass、23／24 skip，原有各 7,165 個終端事件保持，新增各 772 個通過及一個父程序入口 skip。格式、build、vet、固定 Insyra v0.3.4、模組與 31 項治理通過。沒有確認新正式運算缺陷，見[驗收](../../evidence/MOD-02/homeostasis-chemical-state-20261016/verification.json)。清理與交付完成後才勾選最後一項。\n'
assert p.read_bytes() == b
p.write_text(s)
p = R / 'delivery-status.md'
b = p.read_bytes()
s = b.decode()
start = s.index('[ticket 125]')
end = s.index('\n\n', start)
assert '本票 2／4' in s[start:end]
new = '[ticket 125](docs/tickets/125-chemical-homeostasis-state.md) 已通過活動化學與慢速穩定並用的獨立持續狀態驗收。一般／競態各 640 條分段/恢復路徑、16 次新程序及 40 次 fresh neural 拒絕通過，兩種刻意錯誤被拒絕。840 份來源的完整一般／競態各 58 套件通過，7,915／7,914 pass、23／24 skip，原有各 7,165 個終端事件保持，新增各 772 個通過及一個明示的父程序入口 skip。格式、build、vet、固定依賴與 31 項治理通過，工程追蹤保持 91／91，見[驗收](evidence/MOD-02/homeostasis-chemical-state-20261016/verification.json)。本票 3／4，暫存清理及交付尚待完成。沒有確認新正式運算缺陷，整體機制稽核持續。'
s = s[:start] + new + s[end:]
assert p.read_bytes() == b
p.write_text(s)
env = dict(os.environ)
env.update(read(E / 'command-environment.json'))
pattern = '^Test(NoTelemetry|NoTelemetryProjectTempBoundary|NoTelemetrySourceErrorsRemainVisible|FormatIndex|FormatIndexControls|FormatIndexGoLiteralsAndScopes|FormatIndexNoGitAndReadErrors|GovernanceRequirementIDsMatchHandoff|GovernanceStatusValuesAreDeclared|GovernancePassedRequirementsHaveEvidence|GovernanceTicketsHaveDatedRootDecisions)$'
command = [str(W / 'final-root.test'), '-test.v', '-test.run', pattern]
log = E / 'final-governance.log'
start = time.monotonic()
with log.open('xb') as f:
    code = subprocess.run(command, cwd=R, env=env, stdout=f, stderr=subprocess.STDOUT).returncode
passed = len(re.findall(r'^\s*--- PASS:', log.read_text(), re.M))
assert code == 0 and passed == 31, (code, passed)
result = {'command': command, 'exit_code': code, 'passed_events': passed, 'seconds': time.monotonic() - start, 'environment_file': 'command-environment.json', 'log': log.name, 'log_sha256': sha(log), 'source_count': 840}
with (E / 'final-governance.json').open('x') as f:
    json.dump(result, f, indent=2)
    f.write('\n')
p = E / 'verification.json'
b = p.read_bytes()
v = json.loads(b)
v['observed_result']['final_governance'] = {'receipt': 'final-governance.json', 'passed_events': 31}
v['remaining_delivery'].remove('final metadata governance')
assert p.read_bytes() == b
p.write_text(json.dumps(v, indent=2) + '\n')
print(json.dumps(result))
