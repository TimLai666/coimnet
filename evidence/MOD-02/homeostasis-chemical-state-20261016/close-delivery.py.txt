"""Close the verified milestone and publish only its delivery metadata."""
import datetime
import hashlib
import json
from pathlib import Path
import subprocess

R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
read = lambda p: json.loads(p.read_text())

def git(*args):
    return subprocess.check_output(['git', *args], cwd=R, text=True).strip()

def update(p, function):
    before = p.read_bytes()
    after = function(before)
    assert p.read_bytes() == before
    p.write_bytes(after)

delivery = read(E / 'delivery.json')
primary = delivery['primary_commit']
assert git('rev-parse', 'HEAD') == primary
assert git('ls-remote', 'origin', 'refs/heads/codex/receptor-chemistry-training').split()[0] == primary
assert not git('diff', '--cached', '--name-only')
assert delivery['all_content_verified'] and read(E / 'cleanup.json')['directory_removed']
for p, h in read(E / 'committed-content.json')['files'].items():
    assert sha(R / p) == h, p
v = read(E / 'verification.json')
assert v['engineering_passed'] and v['diagnostic_passed']
v['observed_result']['own_cleanup'] = {'receipt': 'cleanup.json', 'removed_bytes': read(E / 'cleanup.json')['removed_bytes']}
v['observed_result']['delivery'] = {'receipt': 'delivery.json', 'primary_commit': primary, 'remote_readback': delivery['remote_readback'], 'source_entries': 840}
v['remaining_delivery'] = []
v['acceptance_complete'] = True
v['completed_at'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
update(E / 'verification.json', lambda b: (json.dumps(v, indent=2) + '\n').encode())

def ticket(b):
    s = b.decode()
    assert s.count('- [ ]') == 1
    s = s.replace('**Status:** in_progress', '**Status:** complete').replace('- [ ]', '- [x]')
    old = '清理與交付完成後才勾選最後一項。'
    assert s.count(old) == 1
    new = f"本輪 {read(E / 'cleanup.json')['removed_bytes']:,} bytes 暫存已清除，18 個原模型、六份必要資料、原始模組與其他工作保持，見[清理](../../evidence/MOD-02/homeostasis-chemical-state-20261016/cleanup.json)。主要提交 `{primary[:7]}` 已正常推送且遠端讀回相同，840 份來源與提交內容核對完成，見[交付](../../evidence/MOD-02/homeostasis-chemical-state-20261016/delivery.json)。本票完成 4／4。"
    return s.replace(old, new).encode()

update(R / 'docs/tickets/125-chemical-homeostasis-state.md', ticket)

def status(b):
    s = b.decode()
    start = s.index('[ticket 125]')
    end = s.index('\n\n', start)
    part = s[start:end]
    old = '本票 3／4，本輪暫存已清除，提交與遠端讀回尚待完成。'
    assert part.count(old) == 1
    new = f"本票 4／4，最後 31 項治理通過，{read(E / 'cleanup.json')['removed_bytes']:,} bytes 自有暫存已清除，原模型、資料、原始模組與其他工作保持，見[清理](evidence/MOD-02/homeostasis-chemical-state-20261016/cleanup.json)。主要提交 `{primary[:7]}` 已推送且遠端讀回相同，840 份提交來源核對完成，見[交付](evidence/MOD-02/homeostasis-chemical-state-20261016/delivery.json)。"
    part = part.replace(old, new)
    part += '下一個可驗證候選為[持續化學來源的絕對時間對照](evidence/MOD-02/homeostasis-chemical-state-20261016/next-audit.md)，尚未確認缺陷。'
    return (s[:start] + part + s[end:]).encode()

update(R / 'delivery-status.md', status)
base = read(E / 'baseline.json')['head']
paths = set(git('diff', '--name-only', base, primary).splitlines())
paths |= {str(p.relative_to(R)) for p in E.rglob('*') if p.is_file()}
paths.add(str((E / 'changed-files.md').relative_to(R)))

def summary(p):
    if p == 'learning/chemical_mixed_reference_test.go':
        return '獨立參考回傳持續狀態、明示未調節暖機，沿用原四列結果'
    if p == 'learning/chemical_homeostasis_state_reference_test.go':
        return '640 條持續路徑、16 次新程序、手算、首步拒絕及原子性對照'
    if p in ['ENG.md', 'delivery-status.md', 'docs/requirements-status.json'] or p.startswith('docs/tickets/'):
        return '更新本票契約、驗收與交付狀態'
    name = Path(p).name
    descriptions = {'plan.md': '固定算式、責任、驗收判準與契約核對修正', 'root-review.md': 'Root 的數學、完整差異與限制審查', 'next-audit.md': '下一個尚未確認缺陷的驗證候選', 'changed-files.md': '完整列出本票變更檔案', 'preserved-work.tar.gz': '封存原始模組清單與隔離錯誤來源、映射', 'cleanup-inventory.json.gz': '清理前逐項內容指紋與位元組數'}
    if name in descriptions:
        return descriptions[name]
    if name.endswith('.go.txt'):
        return '保留判準修正或分工前的私有參考與測試原件'
    if name.endswith('.py.txt'):
        return '保存 ' + name[:-7] + ' 的執行腳本'
    if name.endswith(('.log', '.log.gz')):
        return '保存 ' + name.split('.log')[0] + ' 的原始命令日誌'
    if name.endswith('.json'):
        return '保存 ' + name[:-5] + ' 的設定、驗證結果或內容指紋'
    return '本票驗收證據'

table = '# 本票變更檔案\n\n| 檔案 | 變更摘要 |\n| --- | --- |\n'
table += ''.join(f'| [{p}](../../../{p}) | {summary(p)} |\n' for p in sorted(paths))
with (E / 'changed-files.md').open('x') as f:
    f.write(table)
changed = ['delivery-status.md', 'docs/tickets/125-chemical-homeostasis-state.md', str((E / 'verification.json').relative_to(R)), str((E / 'delivery.json').relative_to(R)), str((E / 'committed-content.json').relative_to(R)), str((E / 'changed-files.md').relative_to(R))]
hashes = {p: sha(R / p) for p in paths}
assert set(git('diff', '--name-only').splitlines()) == {'AGENTS.md', *changed[:2], changed[2]}
for p, h in read(E / 'foreign-before.json').items():
    assert sha(Path(p)) == h, p
subprocess.run(['git', 'add', '--', *changed], cwd=R, check=True)
assert set(git('diff', '--cached', '--name-only').splitlines()) == set(changed)
subprocess.run(['git', 'diff', '--cached', '--check'], cwd=R, check=True)
assert all(sha(R / p) == h for p, h in hashes.items())
subprocess.run(['git', 'commit', '--quiet','-m', 'docs: close persistent chemical state validation'], cwd=R, check=True)
head = git('rev-parse', 'HEAD')
assert set(git('diff-tree', '--no-commit-id', '--name-only', '-r', head).splitlines()) == set(changed)
for p, h in {**read(E / 'formal-source.json'), **hashes}.items():
    assert hashlib.sha256(subprocess.check_output(['git', 'show', head + ':' + p], cwd=R)).hexdigest() == h, p
subprocess.run(['git', 'push', 'origin', 'HEAD:refs/heads/codex/receptor-chemistry-training'], cwd=R, check=True)
remote = git('ls-remote', 'origin', 'refs/heads/codex/receptor-chemistry-training').split()[0]
assert remote == head
assert git('diff', '--name-only') == 'AGENTS.md'
for p, h in read(E / 'foreign-before.json').items():
    assert sha(Path(p)) == h, p
print(json.dumps({'completed': True, 'primary_commit': primary, 'final_commit': head, 'remote_readback': remote, 'source_entries': 840, 'changed_file_count': len(paths), 'final_content_files': len(set(read(E / 'formal-source.json')) | paths), 'remaining_foreign_work_preserved': True}))
