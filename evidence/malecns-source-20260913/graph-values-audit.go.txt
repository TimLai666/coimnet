func addUint64(dst *uint64, value uint64) error {
	if maxUint64-*dst < value {
		return fmt.Errorf("uint64 counter overflow: current=%d add=%d", *dst, value)
	}
	*dst += value
	return nil
}

func inc(dst *uint64) error { return addUint64(dst, 1) }

func addCategory(m map[string]uint64, value string) error {
	count, ok := m[value]
	if !ok {
		if len(m) >= maxCategoryKeys {
			return fmt.Errorf("category map limit exceeded: %d keys", maxCategoryKeys)
		}
	}
	if err := inc(&count); err != nil {
		return err
	}
	m[value] = count
	return nil
}

type endpointSpiller struct {
	dir  string
	ids  map[uint64]struct{}
	peak uint64
	runs []string
}

func newEndpointSpiller() (*endpointSpiller, error) {
	dir, err := os.MkdirTemp("", "coimnet-endpoint-spill-")
	if err != nil {
		return nil, fmt.Errorf("create endpoint spill directory: %w", err)
	}
	return &endpointSpiller{dir: dir, ids: make(map[uint64]struct{}, 1<<20)}, nil
}

func (s *endpointSpiller) add(value int64) error {
	if value < 0 {
		return nil
	}
	id := uint64(value)
	if _, ok := s.ids[id]; ok {
		return nil
	}
	if len(s.ids) >= maxEndpointKeys {
		if err := s.flush(); err != nil {
			return err
		}
	}
	s.ids[id] = struct{}{}
	if keys := uint64(len(s.ids)); keys > s.peak {
		s.peak = keys
	}
	return nil
}

func (s *endpointSpiller) flush() error {
	if len(s.ids) == 0 {
		return nil
	}
	if len(s.runs) >= maxEndpointSpillRuns {
		return fmt.Errorf("endpoint spill run limit exceeded: %d runs", maxEndpointSpillRuns)
	}
	values := make([]uint64, 0, len(s.ids))
	for id := range s.ids {
		values = append(values, id)
	}
	sort.Slice(values, func(i, j int) bool { return values[i] < values[j] })
	file, err := os.CreateTemp(s.dir, "run-*.bin")
	if err != nil {
		return fmt.Errorf("create endpoint spill run: %w", err)
	}
	path := file.Name()
	keep := false
	defer func() {
		if !keep {
			_ = file.Close()
			_ = os.Remove(path)
		}
	}()
	writer := bufio.NewWriterSize(file, 1<<20)
	var encoded [8]byte
	for _, value := range values {
		binary.LittleEndian.PutUint64(encoded[:], value)
		if _, err := writer.Write(encoded[:]); err != nil {
			return fmt.Errorf("write endpoint spill run: %w", err)
		}
	}
	if err := writer.Flush(); err != nil {
		return fmt.Errorf("flush endpoint spill run: %w", err)
	}
	if err := file.Sync(); err != nil {
		return fmt.Errorf("sync endpoint spill run: %w", err)
	}
	if err := file.Close(); err != nil {
		return fmt.Errorf("close endpoint spill run: %w", err)
	}
	s.runs = append(s.runs, path)
	s.ids = make(map[uint64]struct{}, 1<<20)
	keep = true
	return nil
}

type endpointRunReader struct {
	file  *os.File
	value uint64
	ok    bool
	buf   [8]byte
}

func (r *endpointRunReader) next() error {
	n, err := io.ReadFull(r.file, r.buf[:])
	switch err {
	case nil:
		r.value = binary.LittleEndian.Uint64(r.buf[:])
		r.ok = true
		return nil
	case io.EOF:
		if n != 0 {
			return errors.New("truncated endpoint spill run")
		}
		r.ok = false
		return nil
	case io.ErrUnexpectedEOF:
		return errors.New("truncated endpoint spill run")
	default:
		return fmt.Errorf("read endpoint spill run: %w", err)
	}
}

type endpointHeapItem struct {
	run   int
	value uint64
}

type endpointHeap []endpointHeapItem

func (h endpointHeap) Len() int           { return len(h) }
func (h endpointHeap) Less(i, j int) bool { return h[i].value < h[j].value }
func (h endpointHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *endpointHeap) Push(value any)    { *h = append(*h, value.(endpointHeapItem)) }
func (h *endpointHeap) Pop() any {
	old := *h
	n := len(old)
	value := old[n-1]
	*h = old[:n-1]
	return value
}

func (s *endpointSpiller) finish(annotations map[uint64]annotationInfo) (unique, unannotated, runs uint64, err error) {
	if err := s.flush(); err != nil {
		return 0, 0, 0, err
	}
	runs = uint64(len(s.runs))
	readers := make([]endpointRunReader, len(s.runs))
	defer func() {
		for i := range readers {
			if readers[i].file == nil {
				continue
			}
			if closeErr := readers[i].file.Close(); err == nil && closeErr != nil {
				err = fmt.Errorf("close endpoint spill run: %w", closeErr)
			}
		}
	}()
	queue := &endpointHeap{}
	heap.Init(queue)
	for i, path := range s.runs {
		readers[i].file, err = os.Open(path)
		if err != nil {
			return 0, 0, runs, fmt.Errorf("open endpoint spill run: %w", err)
		}
		if err := readers[i].next(); err != nil {
			return 0, 0, runs, err
		}
		if readers[i].ok {
			heap.Push(queue, endpointHeapItem{run: i, value: readers[i].value})
		}
	}
	var last uint64
	haveLast := false
	for queue.Len() > 0 {
		item := heap.Pop(queue).(endpointHeapItem)
		if !haveLast || item.value != last {
			if err := inc(&unique); err != nil {
				return 0, 0, runs, err
			}
			if _, ok := annotations[item.value]; !ok {
				if err := inc(&unannotated); err != nil {
					return 0, 0, runs, err
				}
			}
			last, haveLast = item.value, true
		}
		reader := &readers[item.run]
		if err := reader.next(); err != nil {
			return 0, 0, runs, err
		}
		if reader.ok {
			heap.Push(queue, endpointHeapItem{run: item.run, value: reader.value})
		}
	}
	return unique, unannotated, runs, nil
}

func (s *endpointSpiller) cleanup() error {
	if s == nil || s.dir == "" {
		return nil
	}
	return os.RemoveAll(s.dir)
}

func field(record arrow.Record, name string) (arrow.Array, error) {
	indices := record.Schema().FieldIndices(name)
	if len(indices) != 1 {
		return nil, fmt.Errorf("field %q has %d matches", name, len(indices))
	}
	return record.Column(indices[0]), nil
}

func int64Column(record arrow.Record, name string) (*array.Int64, error) {
	column, err := field(record, name)
	if err != nil {
		return nil, err
	}
	values, ok := column.(*array.Int64)
	if !ok {
		return nil, fmt.Errorf("field %q has type %T, want *array.Int64", name, column)
	}
	return values, nil
}

func stringColumn(record arrow.Record, name string) (*array.String, error) {
	column, err := field(record, name)
	if err != nil {
		return nil, err
	}
	values, ok := column.(*array.String)
	if !ok {
		return nil, fmt.Errorf("field %q has type %T, want *array.String", name, column)
	}
	return values, nil
}

func dictionaryColumn(record arrow.Record, name string) (*array.Dictionary, error) {
	column, err := field(record, name)
	if err != nil {
		return nil, err
	}
	values, ok := column.(*array.Dictionary)
	if !ok {
		return nil, fmt.Errorf("field %q has type %T, want *array.Dictionary", name, column)
	}
	return values, nil
}

func decodeReceipt(path string) (receipt, error) {
	var out receipt
	file, err := os.Open(path)
	if err != nil {
		return out, err
	}
	defer file.Close()
	decoder := json.NewDecoder(file)
	decoder.DisallowUnknownFields()
	if err := decoder.Decode(&out); err != nil {
		return out, fmt.Errorf("decode receipt %s: %w", path, err)
	}
	var extra any
	if err := decoder.Decode(&extra); err != io.EOF {
		return out, fmt.Errorf("receipt %s has trailing JSON", path)
	}
	if out.SchemaVersion != "coimnet-download-receipt/v1" || out.HashStatus != "upstream_verified" || out.Bytes <= 0 || len(out.SHA256) != sha256.Size*2 {
		return out, fmt.Errorf("invalid receipt %s", path)
	}
	return out, nil
}

type fileHash struct {
	bytes int64
	sha   string
}

type sourceFingerprint struct {
	Path   string `json:"path"`
	Bytes  int64  `json:"bytes"`
	SHA256 string `json:"sha256"`
}

func hashFile(ctx context.Context, path string) (fileHash, error) {
	var out fileHash
	if err := ctx.Err(); err != nil {
		return out, err
	}
	info, err := os.Lstat(path)
	if err != nil {
		return out, err
	}
	if !info.Mode().IsRegular() {
		return out, fmt.Errorf("%s is not a regular file", path)
	}
	file, err := os.Open(path)
	if err != nil {
		return out, err
	}
	defer file.Close()
	stat, err := file.Stat()
	if err != nil {
		return out, err
	}
	if !stat.Mode().IsRegular() || stat.Size() < 0 {
		return out, fmt.Errorf("%s changed to non-regular file", path)
	}
	h := sha256.New()
	buffer := make([]byte, 1<<20)
	for {
		if err := ctx.Err(); err != nil {
			return out, err
		}
		n, readErr := file.Read(buffer)
		if n > 0 {
			if int64(n) > math.MaxInt64-out.bytes {
				return out, errors.New("file byte counter overflow")
			}
			out.bytes += int64(n)
			if _, err := h.Write(buffer[:n]); err != nil {
				return out, err
			}
		}
		if readErr == io.EOF {
			break
		}
		if readErr != nil {
			return out, readErr
		}
	}
	out.sha = hex.EncodeToString(h.Sum(nil))
	return out, nil
}

func scanOptions(maxFileBytes, maxRows int64) feather.Options {
	return feather.Options{MaxFileBytes: maxFileBytes, MaxFooterBytes: maxFooterBytes, MaxArrowBytes: maxArrowBytes, MaxRows: maxRows}
}

func scanAnnotations(ctx context.Context, spec sourceSpec, rec receipt, source *sourceReport) (map[uint64]annotationInfo, annotationStats, error) {
	ids := make(map[uint64]annotationInfo)
	stats := annotationStats{StatusValueCounts: map[string]uint64{}, StatusLabelValueCounts: map[string]uint64{}}
	report, err := feather.Scan(ctx, spec.Path, scanOptions(rec.Bytes, spec.MaxRows), func(record arrow.Record) error {
		body, err := int64Column(record, "bodyId")
		if err != nil {
			return err
		}
		status, err := stringColumn(record, "status")
		if err != nil {
			return err
		}
		statusLabel, err := dictionaryColumn(record, "statusLabel")
		if err != nil {
			return err
		}
		rows := record.NumRows()
		if rows < 0 {
			return errors.New("negative annotation batch rows")
		}
		for i := 0; i < int(rows); i++ {
			if err := inc(&stats.Rows); err != nil {
				return err
			}
			if status.IsNull(i) {
				if err := inc(&stats.StatusNullRows); err != nil {
					return err
				}
			} else {
				value := status.Value(i)
				if err := addCategory(stats.StatusValueCounts, value); err != nil {
					return err
				}
				if value == "Traced" {
					if err := inc(&stats.TracedStatusRows); err != nil {
						return err
					}
				}
			}
			if statusLabel.IsNull(i) {
				if err := inc(&stats.StatusLabelNullRows); err != nil {
					return err
				}
			} else if err := addCategory(stats.StatusLabelValueCounts, statusLabel.ValueStr(i)); err != nil {
				return err
			}
			if body.IsNull(i) {
				if err := inc(&stats.BodyIDNullRows); err != nil {
					return err
				}
				continue
			}
			value := body.Value(i)
			if value < 0 {
				if err := inc(&stats.BodyIDNegativeRows); err != nil {
					return err
				}
				continue
			}
			id := uint64(value)
			traced := !status.IsNull(i) && status.Value(i) == "Traced"
			previous, exists := ids[id]
			if exists {
				if err := inc(&stats.BodyIDDuplicateRows); err != nil {
					return err
				}
				if previous.traced != traced {
					if err := inc(&stats.DuplicateStatusConflictRows); err != nil {
						return err
					}
				}
				continue
			}
			if len(ids) >= maxEndpointKeys {
				return fmt.Errorf("annotation ID map limit exceeded: %d keys", maxEndpointKeys)
			}
			ids[id] = annotationInfo{traced: traced}
			if err := inc(&stats.BodyIDUniqueNonNull); err != nil {
				return err
			}
			if traced {
				if err := inc(&stats.TracedSelectedUniqueNodes); err != nil {
					return err
				}
			}
		}
		return nil
	})
	if err != nil {
		return nil, stats, err
	}
	if !report.Complete {
		return nil, stats, errors.New("annotation scan incomplete")
	}
	stats.AnnotationMapKeys = uint64(len(ids))
	source.Report = report
	return ids, stats, nil
}

func updateColumnStats(stats *columnStats, value int64, valid bool) error {
	if !valid {
		return inc(&stats.NullRows)
	}
	if value < 0 {
		return inc(&stats.NegativeRows)
	}
	if value == 0 {
		return inc(&stats.ZeroRows)
	}
	return nil
}

func scanWeights(ctx context.Context, spec sourceSpec, rec receipt, annotations map[uint64]annotationInfo, source *sourceReport) (stats weightsStats, err error) {
	stats = weightsStats{RawContactTotalValid: true, TracedBothContactTotalValid: true}
	spiller, err := newEndpointSpiller()
	if err != nil {
		return stats, err
	}
	defer func() {
		if cleanupErr := spiller.cleanup(); err == nil && cleanupErr != nil {
			err = fmt.Errorf("cleanup endpoint spill: %w", cleanupErr)
		}
	}()
	report, err := feather.Scan(ctx, spec.Path, scanOptions(rec.Bytes, spec.MaxRows), func(record arrow.Record) error {
		pre, err := int64Column(record, "body_pre")
		if err != nil {
			return err
		}
		post, err := int64Column(record, "body_post")
		if err != nil {
			return err
		}
		weight, err := int64Column(record, "weight")
		if err != nil {
			return err
		}
		rows := record.NumRows()
		if rows < 0 {
			return errors.New("negative weight batch rows")
		}
		for i := 0; i < int(rows); i++ {
			if err := inc(&stats.RawEdgeRows); err != nil {
				return err
			}
			preValid, postValid, weightValid := !pre.IsNull(i), !post.IsNull(i), !weight.IsNull(i)
			if err := updateColumnStats(&stats.BodyPre, pre.Value(i), preValid); err != nil {
				return err
			}
			if err := updateColumnStats(&stats.BodyPost, post.Value(i), postValid); err != nil {
				return err
			}
			if err := updateColumnStats(&stats.Weight, weight.Value(i), weightValid); err != nil {
				return err
			}
			if weightValid {
				w := weight.Value(i)
				if w < 0 {
					stats.RawContactTotalValid = false
				} else {
					if err := addUint64(&stats.RawContactTotal, uint64(w)); err != nil {
						return fmt.Errorf("raw contact total: %w", err)
					}
					if err := inc(&stats.RawContactRowsIncluded); err != nil {
						return err
					}
				}
			}
			if preValid && postValid {
				pv, qv := pre.Value(i), post.Value(i)
				if pv == qv {
					if err := inc(&stats.SelfEdgeRows); err != nil {
						return err
					}
					if weightValid && weight.Value(i) >= 0 {
						if err := addUint64(&stats.SelfEdgeContactTotal, uint64(weight.Value(i))); err != nil {
							return fmt.Errorf("self contact total: %w", err)
						}
					}
				}
				if err := spiller.add(pv); err != nil {
					return err
				}
				if err := spiller.add(qv); err != nil {
					return err
				}
				if pv >= 0 && qv >= 0 {
					left, leftOK := annotations[uint64(pv)]
					right, rightOK := annotations[uint64(qv)]
					if leftOK && rightOK && left.traced && right.traced {
						if err := inc(&stats.TracedBothEdgeRows); err != nil {
							return err
						}
						if weightValid && weight.Value(i) < 0 {
							stats.TracedBothContactTotalValid = false
						} else if weightValid {
							if err := addUint64(&stats.TracedBothContactTotal, uint64(weight.Value(i))); err != nil {
								return fmt.Errorf("traced contact total: %w", err)
							}
						}
					}
				}
			} else {
				if preValid {
					if err := spiller.add(pre.Value(i)); err != nil {
						return err
					}
				}
				if postValid {
					if err := spiller.add(post.Value(i)); err != nil {
						return err
					}
				}
			}
		}
		return nil
	})
	if err != nil {
		return stats, err
	}
	if !report.Complete {
		return stats, errors.New("weights scan incomplete")
	}
	unique, unannotated, runs, err := spiller.finish(annotations)
	if err != nil {
		return stats, fmt.Errorf("merge endpoint spill runs: %w", err)
	}
	stats.IndependentEndpointCount = unique
	stats.UnannotatedUniqueEndpointCount = unannotated
	stats.EndpointMapPeakKeys = spiller.peak
	stats.EndpointSpillRuns = runs
	source.Report = report
	return stats, nil
}

func unknownConsensus(value string) bool {
	switch strings.ToLower(strings.TrimSpace(value)) {
	case "", "unknown", "unclear", "none", "null", "na", "n/a":
		return true
	default:
		return false
	}
}

func scanNeurotransmitters(ctx context.Context, spec sourceSpec, rec receipt, source *sourceReport) (ntStats, error) {
	stats := ntStats{ConsensusValueCounts: map[string]uint64{}, ConsensusUnknownValueCounts: map[string]uint64{}}
	ids := make(map[uint64]struct{})
	report, err := feather.Scan(ctx, spec.Path, scanOptions(rec.Bytes, spec.MaxRows), func(record arrow.Record) error {
		body, err := int64Column(record, "body")
		if err != nil {
			return err
		}
		consensus, err := stringColumn(record, "consensus_nt")
		if err != nil {
			return err
		}
		rows := record.NumRows()
		if rows < 0 {
			return errors.New("negative neurotransmitter batch rows")
		}
		for i := 0; i < int(rows); i++ {
			if err := inc(&stats.Rows); err != nil {
				return err
			}
			if body.IsNull(i) {
				if err := inc(&stats.BodyNullRows); err != nil {
					return err
				}
			} else {
				value := body.Value(i)
				if value < 0 {
					if err := inc(&stats.BodyNegativeRows); err != nil {
						return err
					}
				} else {
					id := uint64(value)
					if _, exists := ids[id]; exists {
						if err := inc(&stats.BodyDuplicateRows); err != nil {
							return err
						}
					} else {
						if len(ids) >= maxNTKeys {
							return fmt.Errorf("neurotransmitter body map limit exceeded: %d keys", maxNTKeys)
						}
						ids[id] = struct{}{}
						if err := inc(&stats.BodyUniqueNonNull); err != nil {
							return err
						}
					}
				}
			}
			if consensus.IsNull(i) {
				if err := inc(&stats.ConsensusNullRows); err != nil {
					return err
				}
				continue
			}
			value := consensus.Value(i)
			if err := addCategory(stats.ConsensusValueCounts, value); err != nil {
				return err
			}
			if unknownConsensus(value) {
				if err := inc(&stats.ConsensusUnknownRows); err != nil {
					return err
				}
				if err := addCategory(stats.ConsensusUnknownValueCounts, value); err != nil {
					return err
				}
			}
		}
		return nil
	})
	if err != nil {
		return stats, err
	}
	if !report.Complete {
		return stats, errors.New("neurotransmitter scan incomplete")
	}
	stats.NTMapKeys = uint64(len(ids))
	if stats.Rows != 0 {
		stats.ConsensusUnknownRatio = float64(stats.ConsensusUnknownRows) / float64(stats.Rows)
	}
	source.Report = report
	return stats, nil
}

func verifySource(ctx context.Context, spec sourceSpec) (sourceReport, receipt, fileHash, error) {
	var source sourceReport
	rec, err := decodeReceipt(spec.ReceiptPath)
	if err != nil {
		return source, rec, fileHash{}, err
	}
	before, err := hashFile(ctx, spec.Path)
	if err != nil {
		return source, rec, fileHash{}, fmt.Errorf("hash before %s: %w", spec.Name, err)
	}
	source = sourceReport{Name: spec.Name, Role: spec.Role, Path: spec.Path, ReceiptPath: spec.ReceiptPath, Receipt: rec, BytesBefore: before.bytes, SHA256Before: before.sha, BeforeMatchesReceipt: before.bytes == rec.Bytes && strings.EqualFold(before.sha, rec.SHA256)}
	if !source.BeforeMatchesReceipt {
		return source, rec, before, fmt.Errorf("%s before hash does not match receipt", spec.Name)
	}
	return source, rec, before, nil
}

func finalizeSource(ctx context.Context, source *sourceReport, before fileHash) error {
	after, err := hashFile(ctx, source.Path)
	if err != nil {
		return fmt.Errorf("hash after %s: %w", source.Name, err)
	}
	source.BytesAfter, source.SHA256After = after.bytes, after.sha
	source.AfterMatchesReceipt = after.bytes == source.Receipt.Bytes && strings.EqualFold(after.sha, source.Receipt.SHA256)
	source.HashStable = source.BeforeMatchesReceipt && source.AfterMatchesReceipt && before.bytes == after.bytes && before.sha == after.sha
	if !source.HashStable {
		return fmt.Errorf("%s hash changed during audit", source.Name)
	}
	return nil
}
