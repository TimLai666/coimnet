bits = lambda x: struct.unpack('>Q', struct.pack('>d', x))[0]
maximum = float.fromhex('0x1.fffffffffffffp1023')
minimum = float.fromhex('0x0.0000000000001p-1022')
cases = []
def add(name, logits, target=0, factor=1.0):
    cases.append(dict(name=name, logits=logits, target=target, factor=factor))
for i in range(3):

    add('winner_' + str(i), row, i)

add('minimum_factor_extreme_gap', [maximum, -maximum], 1, minimum)

with localcontext() as ctx:
    ctx.prec = 1250
    cache = {}
    for row in cases:
        values = [Decimal.from_float(v) for v in row['logits']]
        largest = max(values)
        total = Decimal(0)
        for value in values:
            gap = value - largest
            # These selected cases have <=259 classes and factor<=MaxFloat64.
            # Terms beyond this cutoff cannot affect their float64 rounding.
            if gap < -10000:
                term = Decimal(0)
            else:
                if gap not in cache:
                    cache[gap] = gap.exp()
                term = cache[gap]
            total += term
        expected = (largest - values[row['target']] + total.ln()) * Decimal.from_float(row['factor'])
        rounded = float(expected)
        row.update(expected_decimal=str(expected), expected_error=math.isinf(rounded),
                   expected_bits=None if math.isinf(rounded) else f'{bits(rounded):016x}',
                   logits_bits=[f'{bits(v):016x}' for v in row['logits']], factor_bits=f'{bits(row["factor"]):016x}')
out = {'precision_decimal_digits':1250, 'formula':'factor * (max(logits)-target + ln(sum(exp(logit-max))))', 'float_inputs':'Decimal.from_float; exact input binary values', 'omitted_exp_gap_below':-10000, 'cases':cases}
with (E/'reference-v2.json').open('x') as handle:
    json.dump(out,handle,indent=2,allow_nan=False)
    handle.write('\n')
