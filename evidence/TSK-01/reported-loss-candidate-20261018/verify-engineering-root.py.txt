"""Independently verify completed ticket148 engineering and old-model receipts."""
from pathlib import Path
import argparse, collections, datetime, gzip, hashlib, json

p = argparse.ArgumentParser(description=__doc__)
p.add_argument('--run', action='store_true', help='Verify all completed receipts and publish new Root evidence without overwriting files.')
p.add_argument('--terminal-tool-chunk', help='Root-observed successful terminal tool chunk; required with --run')
a = p.parse_args()
if not a.run:
    p.print_help()
    raise SystemExit(0)
if not a.terminal_tool_chunk:
    p.error('--terminal-tool-chunk is required with --run')
if not __debug__:
    raise SystemExit('Optimized Python cannot skip validation')
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
B = R / 'evidence/TSK-03/categorical-gradient-adoption-20261018'
W = R / 'runs/.tmp/reported-loss-candidate-20261018'
sha = lambda raw: hashlib.sha256(raw).hexdigest()
read = lambda path: json.loads(path.read_bytes())
source_raw = (E / 'formal-source.json').read_bytes()
source = json.loads(source_raw)
assert len(source) == 901 and sha(source_raw) == read(E / 'formal-tests-frozen.json')['source_manifest_sha256']
for name, digest in source.items():
    assert sha((R / name).read_bytes()) == digest, name

def log_bytes(directory, receipt):
    assert Path(receipt['log']).name == receipt['log']
    raw = (directory / receipt['log']).read_bytes()
    assert len(raw) == receipt['log_bytes'] and sha(raw) == receipt['log_sha256']
    if receipt['log'].endswith('.gz'):
        raw = gzip.decompress(raw)
        assert len(raw) == receipt['raw_bytes'] and sha(raw) == receipt['raw_sha256']
    return raw

def outcomes(raw):
    tests, packages = {}, {}
    for line in raw.splitlines():
        try:
            row = json.loads(line)
        except ValueError:
            continue
        if row.get('Action') not in ['pass', 'skip', 'fail']:
            continue
        key = (row['Package'], row['Test']) if row.get('Test') else row['Package']
        target = tests if row.get('Test') else packages
        assert key not in target
        target[key] = row['Action']
    return tests, packages

pattern = '^Test(NoTelemetry|NoTelemetryProjectTempBoundary|NoTelemetrySourceErrorsRemainVisible|FormatIndex|FormatIndexControls|FormatIndexGoLiteralsAndScopes|FormatIndexNoGitAndReadErrors|GovernanceRequirementIDsMatchHandoff|GovernanceStatusValuesAreDeclared|GovernancePassedRequirementsHaveEvidence|GovernanceTicketsHaveDatedRootDecisions)$'
expected = [
    ('format', ['gofmt', '-l', *[n for n in source if n.endswith('.go') and not n.startswith('evidence/')]]),
    ('build', ['go', 'build', './...']),
    ('unit', ['go', 'test', '-json', '-count=1', '-timeout=60m', './...']),
    ('race', ['go', 'test', '-race', '-json', '-count=1', '-timeout=60m', './...']),
    ('vet', ['go', 'vet', './...']),
    ('module-verify', ['go', 'mod', 'verify']),
    ('module-selection', ['go', 'list', '-m', 'github.com/HazelnutParadise/insyra']),
    ('governance', ['go', 'test', '-json', '-count=1', '-run', pattern, '.']),
    ('closure-binary', ['go', 'test', '-c', '-o', str(W / 'final-root.test'), '.']),
]
engineering = read(E / 'engineering-verification.json')
assert engineering['all_passed'] and engineering['source_count'] == 901 and len(engineering['steps']) == 9
frozen = read(E / 'formal-tests-frozen.json')
proof = []
for embedded, (name, command) in zip(engineering['steps'], expected):
    receipt = read(E / ('engineering-' + name + '.json'))
    assert embedded == receipt and receipt['command'] == command
    assert receipt['exit_code'] == 0 and receipt['source_count'] == 901 and receipt['source_manifest'] == 'formal-source.json' and receipt['acceptance_errors'] == []
    raw = log_bytes(E, receipt)
    item = {'step': name, 'receipt': 'engineering-' + name + '.json', 'receipt_sha256': sha((E / ('engineering-' + name + '.json')).read_bytes()), 'command': command, 'log': receipt['log'], 'log_sha256': receipt['log_sha256'], 'exit_code': 0}
    if name in ['unit', 'race', 'governance']:
        tests, packages = outcomes(raw)
        assert tests and set(tests.values()) <= {'pass', 'skip'} and set(packages.values()) <= {'pass', 'skip'}
        assert dict(collections.Counter(tests.values())) == receipt['test_events']
        assert dict(collections.Counter(packages.values())) == receipt['package_events']
        if name in ['unit', 'race']:
            old, old_packages = outcomes(log_bytes(B, read(B / ('engineering-' + name + '.json'))))
            assert len(old) == 8940 and all(tests.get(k) == v for k, v in old.items())
            rows = frozen['expected_new_events'][name]
            new = {(r['package'], r['test']): r['action'] for r in rows}
            assert len(new) == len(rows) == 77 and set(new.values()) == {'pass'}
            assert all(k[1].split('/')[0] in frozen['parents'] for k in new)
            assert {k: v for k, v in tests.items() if k not in old} == new
            assert packages == old_packages
            assert receipt['test_events'] == ({'pass': 8994, 'skip': 23} if name == 'unit' else {'pass': 8993, 'skip': 24})
            item.update(prior_events_preserved=8940, added_pass_events=77)
        else:
            assert receipt['test_events'] == {'pass': 31}
        item.update(test_events=receipt['test_events'], package_events=receipt['package_events'])
    if name in ['format', 'build', 'vet', 'closure-binary']:
        assert not raw.strip()
    if name == 'module-verify':
        assert raw.decode().strip() == 'all modules verified'
    if name == 'module-selection':
        assert raw.decode().strip() == 'github.com/HazelnutParadise/insyra v0.3.4'
    if name == 'race':
        assert b'WARNING: DATA RACE' not in raw
    proof.append(item)
