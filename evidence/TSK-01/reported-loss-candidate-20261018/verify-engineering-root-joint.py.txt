def log_bytes(directory, receipt):
    assert Path(receipt['log']).name == receipt['log']
    raw = (directory / receipt['log']).read_bytes()
    assert len(raw) == receipt['log_bytes'] and sha(raw) == receipt['log_sha256']
    if receipt['log'].endswith('.gz'):
        raw = gzip.decompress(raw)
        assert len(raw) == receipt['raw_bytes'] and sha(raw) == receipt['raw_sha256']
    return raw

def outcomes(raw):
    tests, packages = {}, {}
    for line in raw.splitlines():
        try:
            row = json.loads(line)
        except ValueError:
            continue
        if row.get('Action') not in ['pass', 'skip', 'fail']:
            continue
        key = (row['Package'], row['Test']) if row.get('Test') else row['Package']
        target = tests if row.get('Test') else packages
        assert key not in target
        target[key] = row['Action']
    return tests, packages

        assert tests and set(tests.values()) <= {'pass', 'skip'} and set(packages.values()) <= {'pass', 'skip'}

            assert prior_receipt['log'] == 'engineering-' + name + '.log.gz'

            assert receipt['test_events'] == {'pass': 31}
        item.update(test_events=receipt['test_events'], package_events=receipt['package_events'])
    if name in ['format', 'build', 'vet', 'closure-binary']:
