package categorical

import "math"

// nearHalfMinimum selects a precision fallback; it never changes a value.
func nearHalfMinimum(mantissa float64, exponent int) bool {
	scaled := math.Ldexp(math.Abs(mantissa), exponent+1075)
	const guard = 32 * 0x1p-52
	return scaled >= 1-guard && scaled <= 1+guard
}
