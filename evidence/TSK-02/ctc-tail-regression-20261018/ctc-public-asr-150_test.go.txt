const (
	asrCTCTailOneFrameLoss150 = 0x1.39792499b1a24p-58
	asrCTCTailTwoFrameLoss150 = 0x1.7fd974d372e44p-116
)

// TestASRCTCTailTraining150 exercises the public whole-file ASR CTC path with
// exactly one or two log-mel frames and retains a normal ln(2) control.
func TestASRCTCTailTraining150(t *testing.T) {
	cases := []struct {
		name     string
		frames   int
		wantLoss float64
		normal   bool
	}{
		{name: "one-frame", frames: 1, wantLoss: asrCTCTailOneFrameLoss150},
		{name: "two-frame", frames: 2, wantLoss: asrCTCTailTwoFrameLoss150},
		{name: "normal", frames: 1, wantLoss: math.Ln2, normal: true},
	}

	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			ctx := context.Background()
			config := asrCTCTailConfig150()
			model, err := asr.NewRecognizer(config)
			if err != nil {
				t.Fatalf("NewRecognizer: %v", err)
			}
			parameters := asrCTCTailParameters150(t, model, tc.normal)
			model, err = asrCTCTailRestore150(t, config, model.Snapshot(), parameters)
			if err != nil {
				t.Fatalf("LoadRecognizer: %v", err)
			}

			samples := make([]float64, tc.frames*config.FrontEnd.Window)
			beforeSamples := append([]float64(nil), samples...)
			before := model.Snapshot()
			if before.Options.WeightDecay != 0 {
				t.Fatalf("weight decay=%g, want 0", before.Options.WeightDecay)
			}
			if before.Parameters.Readout[1] != 0 {
				t.Fatalf("initial target readout=%g, want exact zero", before.Parameters.Readout[1])
			}
			if tc.normal {
				for index, value := range before.Parameters.Readout {
					if value != 0 {
						t.Fatalf("normal readout[%d]=%g, want exact zero", index, value)
					}
				}
				readoutNode := before.Config.ReadoutNodes[0]
				if !(before.Parameters.Core.Bias[readoutNode] > 0) || math.IsInf(before.Parameters.Core.Bias[readoutNode], 0) {
					t.Fatalf("normal hidden bias[%d]=%g, want finite positive activity source", readoutNode, before.Parameters.Core.Bias[readoutNode])
				}
			}

			asrCTCTailAssertLogits150(t, model, samples, tc.frames, tc.normal)
			report, err := model.TrainUtterance(ctx, samples, "a")
			if err != nil {
				t.Fatalf("TrainUtterance: %v", err)
			}
			after := model.Snapshot()
			t.Logf("frames=%d labels=%d impossible=%t loss=%.17g target_readout_before=%.17g target_readout_after=%.17g", report.Frames, report.Labels, report.Impossible, report.Loss, before.Parameters.Readout[1], after.Parameters.Readout[1])
			if report.Frames != tc.frames || report.Labels != 1 || report.Impossible {
				t.Errorf("TrainUtterance report=%+v, want frames=%d labels=1 impossible=false", report, tc.frames)
			}
			if !sameFloatBits150(report.Loss, tc.wantLoss) && ulpDistance150(report.Loss, tc.wantLoss) > 4 {
				t.Errorf("TrainUtterance loss=%.17g, want %.17g within 4 ULP (distance=%d)", report.Loss, tc.wantLoss, ulpDistance150(report.Loss, tc.wantLoss))
			}
			if !(report.Loss > 0) || math.IsInf(report.Loss, 0) {
				t.Errorf("TrainUtterance erased finite positive loss: %.17g", report.Loss)
			}
			if after.Updates != before.Updates+1 {
				t.Errorf("updates=%d, want %d", after.Updates, before.Updates+1)
			}
			if !sameFloatSlice150(samples, beforeSamples) {
				t.Error("TrainUtterance mutated input samples")
			}
			// The independent oracle gives [+q,-q] per row. With a positive
			// hidden activity, the target readout coefficient must therefore move
			// strictly positive from its exact zero.
			if !(after.Parameters.Readout[1] > 0) || math.IsInf(after.Parameters.Readout[1], 0) {
				t.Errorf("target readout=%g, want strict finite positive update from zero", after.Parameters.Readout[1])
			}
		})
	}
}

func asrCTCTailConfig150() asr.RecognizerConfig {
	return asr.RecognizerConfig{
		SampleRate: 16000,
		FrontEnd: audio.FrontEndConfig{
			Window: 16,
			Hop:    16,
			Mels:   1,
			FMin:   0,
			FMax:   8000,
			Floor:  1e-10,
		},
		Hidden:       2,
		Alphabet:     []rune{'a'},
		LearningRate: 0.01,
		Seed:         150,
	}
}

func asrCTCTailParameters150(t *testing.T, model *asr.Recognizer, normal bool) learning.Parameters {
	t.Helper()
	snapshot := model.Snapshot()
	parameters := snapshot.Parameters
	parameters.Core.Weights = make([]float64, len(parameters.Core.Weights))
	parameters.Core.Bias = make([]float64, len(parameters.Core.Bias))
	parameters.Core.LogTau = make([]float64, len(parameters.Core.LogTau))
	for i := range parameters.Core.LogTau {
		parameters.Core.LogTau[i] = math.Log(2)
	}
	parameters.Encoder = make([]float64, len(parameters.Encoder))
	parameters.Readout = make([]float64, len(parameters.Readout))
	if normal {
		// Keep the readout weights at [0,0] but retain the same constant positive
		// hidden activity as the tiny fixture. This makes the ln(2) control's
		// target coefficient update independently observable.
		parameters.Core.Bias[snapshot.Config.ReadoutNodes[0]] = 100
		return parameters
	}

	// The original public SDK has no separate ReadoutBias field. To create the
	// same constant logits using only public Parameters, saturate the first
	// readout node at exactly one and use its blank readout coefficient -40;
	// the target coefficient starts at zero. All encoder and core edge weights
	// are zero, and the effective readout rows are exactly [-40,0].
	readoutNode := snapshot.Config.ReadoutNodes[0]
	parameters.Core.Bias[readoutNode] = 100
	if len(parameters.Readout) < 2 {
		t.Fatalf("readout length=%d, want at least two", len(parameters.Readout))
	}
	parameters.Readout[0] = -40
	return parameters
}

func asrCTCTailRestore150(t *testing.T, config asr.RecognizerConfig, snapshot learning.TrainingSnapshot, parameters learning.Parameters) (*asr.Recognizer, error) {
	t.Helper()
	snapshot.Parameters = parameters
	configJSON, err := json.Marshal(config)
	if err != nil {
		return nil, err
	}
	path := filepath.Join(t.TempDir(), "asr-checkpoint")
	if err := checkpoint.SaveTaskTrainingBundle(context.Background(), path, "coimnet-asr-checkpoint/v1", configJSON, snapshot); err != nil {
		return nil, err
	}
	return asr.LoadRecognizer(context.Background(), path)
}

func asrCTCTailAssertLogits150(t *testing.T, model *asr.Recognizer, samples []float64, wantFrames int, normal bool) {
	t.Helper()
	snapshot := model.Snapshot()
	features, front, err := model.Features(samples)
	if err != nil {
		t.Fatalf("Features: %v", err)
	}
	if front.Frames != wantFrames || len(features) != wantFrames {
		t.Fatalf("feature frames report=%d matrix=%d, want %d", front.Frames, len(features), wantFrames)
	}
	network, err := learning.NewNetwork(snapshot.Config)
	if err != nil {
		t.Fatalf("learning.NewNetwork: %v", err)
	}
	logits, err := network.PredictAll(context.Background(), snapshot.Parameters, features)
	if err != nil {
		t.Fatalf("PredictAll: %v", err)
	}
	if len(logits) != wantFrames {
		t.Fatalf("logits frames=%d, want %d", len(logits), wantFrames)
	}
	for frame, row := range logits {
		if len(row) != 2 {
			t.Fatalf("logits frame %d width=%d, want 2", frame, len(row))
		}
		if normal {
			if row[0] != 0 || row[1] != 0 {
				t.Fatalf("normal logits frame %d=%v, want exact [0,0]", frame, row)
			}
			continue
		}
		if row[0] != -40 || row[1] != 0 {
			t.Fatalf("tail logits frame %d=%v, want exact [-40,0]", frame, row)
		}
	}
}

func sameFloatBits150(a, b float64) bool {
	return math.Float64bits(a) == math.Float64bits(b)
}

func sameFloatSlice150(a, b []float64) bool {
	if len(a) != len(b) {
		return false
	}
	for i := range a {
		if !sameFloatBits150(a[i], b[i]) {
			return false
		}
	}
	return true
}

func ulpDistance150(a, b float64) uint64 {
	ua, ub := math.Float64bits(a), math.Float64bits(b)
	if ua > ub {
		return ua - ub
	}
	return ub - ua
}
