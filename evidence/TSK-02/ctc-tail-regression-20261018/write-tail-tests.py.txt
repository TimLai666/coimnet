def row(values):
    return "{" + ", ".join(float(value).hex() for value in values) + "}"


def table(values):
    return "[][]float64{" + ", ".join(row(values_row) for values_row in values) + "}"


source = r'''package ctc_test

import (
    "math"
    "reflect"
    "testing"

// References enumerate all paths independently at 900 and 1400 Decimal digits.

// Four ULP is supplemented by strict nonzero, sign and actual-underflow checks.
func TestTailRepresentability150(t *testing.T) {
    cases := []struct {
        name string
        logits [][]float64
        target []int
        loss float64
        gradient [][]float64
    }{
'''
for case in reference["cases"]:
    target = "[]int{" + ", ".join(str(value) for value in case["target"]) + "}"
    source += "        {" + json.dumps(case["name"]) + ", " + table(case["logits"]) + ", " + target + ", " + float(case["loss"]).hex() + ", " + table(case["gradient"]) + "},\n"
source += r'''    }
    for _, tc := range cases {
        t.Run(tc.name, func(t *testing.T) {
            logits := tail150Copy(tc.logits)
            before := tail150Copy(logits)
            target := append([]int{}, tc.target...)
            beforeTarget := append([]int{}, target...)
            loss, gradient, report, err := ctc.LossWith(logits, target, ctc.Options{})
            if err != nil {

            }

            tail150Gradient(t, gradient, tc.gradient)
            if !reflect.DeepEqual(logits, before) || !reflect.DeepEqual(target, beforeTarget) {
                t.Fatal("LossWith changed caller inputs")
            }

                t.Fatalf("Loss wrapper refused valid input: %v", wrapperErr)

            tail150Gradient(t, wrapperGradient, tc.gradient)

                t.Fatal("Loss changed caller inputs")
            }
        })
    }
}

func tail150Copy(rows [][]float64) [][]float64 {
    if rows == nil {
        return nil
    }
    copied := make([][]float64, len(rows))
    for i := range rows {
        if rows[i] != nil {
            copied[i] = append([]float64{}, rows[i]...)
        }
    }
    return copied
}

func tail150Close(t *testing.T, label string, got, want float64) {
    t.Helper()
    if math.IsNaN(got) || math.IsInf(got, 0) {
        t.Errorf("%s non-finite: got=%g want=%g", label, got, want)
        return
    }
    if want == 0 {
        if got != 0 {
            t.Errorf("%s true rounded zero: got=%g", label, got)
        }
        return
    }
    if got == 0 || math.Signbit(got) != math.Signbit(want) {
        t.Errorf("%s lost nonzero/sign: got=%g want=%g", label, got, want)
        return
    }
    a, b := math.Float64bits(got), math.Float64bits(want)
    if a < b {
        a, b = b, a
    }
    if a-b > 4 {
        t.Errorf("%s outside four ULP: got=%g want=%g distance=%d", label, got, want, a-b)
    }
}

func tail150Gradient(t *testing.T, got, want [][]float64) {
    t.Helper()
    if len(got) != len(want) {
        t.Fatalf("gradient rows=%d want=%d", len(got), len(want))
    }
    for i := range want {
        if len(got[i]) != len(want[i]) {
            t.Fatalf("gradient row %d columns=%d want=%d", i, len(got[i]), len(want[i]))
        }
        for k := range want[i] {
            tail150Close(t, "gradient", got[i][k], want[i][k])
        }
    }
}
