package main

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"os"
	"strconv"

	"github.com/TimLai666/coimnet/tasks/asr"
	"github.com/TimLai666/coimnet/tasks/asr/audio"
)

type observation struct {
	Name             string `json:"name"`
	Bits             int    `json:"int_bits"`
	Mels             int    `json:"mels"`
	Hidden           string `json:"hidden"`
	WrappedNodes     string `json:"wrapped_nodes"`
	WrappedEdges     string `json:"wrapped_edges"`
	Invalid          bool   `json:"intended_invalid_dimensions"`
	ValidateError    string `json:"validate_error"`
	ConstructorError string `json:"constructor_error"`
	ModelNil         bool   `json:"model_nil"`
	Panic            string `json:"panic"`
	SnapshotSHA256   string `json:"snapshot_sha256,omitempty"`
}

func errText(err error) string {
	if err == nil {
		return ""
	}
	return err.Error()
}
func probe(name string, mels, hidden int, invalid bool) (out observation) {
	nodes, edges := mels+hidden, mels*hidden+hidden*hidden
	if invalid && edges >= 0 {
		panic("preflight failed: first capacity is not negative")
	}
	if !invalid && (nodes < 1 || nodes > 16 || edges < 1 || edges > 256) {
		panic("preflight failed: control is not tiny")
	}
	out = observation{Name: name, Bits: strconv.IntSize, Mels: mels, Hidden: strconv.Itoa(hidden), WrappedNodes: strconv.Itoa(nodes), WrappedEdges: strconv.Itoa(edges), Invalid: invalid, ModelNil: true}
	config := asr.RecognizerConfig{SampleRate: 16000, FrontEnd: audio.FrontEndConfig{Window: 16, Hop: 8, Mels: mels, FMin: 0, FMax: 8000, Floor: 1e-10}, Hidden: hidden, Alphabet: []rune{'A'}, LearningRate: .01, Seed: 7}
	out.ValidateError = errText(config.Validate())
	defer func() {
		if p := recover(); p != nil {
			out.Panic = fmt.Sprint(p)
		}
	}()
	model, err := asr.NewRecognizer(config)
	out.ModelNil = model == nil
	out.ConstructorError = errText(err)
	if model != nil {
		data, err := json.Marshal(model.Snapshot())
		if err != nil {
			panic(err)
		}
		sum := sha256.Sum256(data)
		out.SnapshotSHA256 = hex.EncodeToString(sum[:])
	}
	return
}
func main() {
	maxInt := int(^uint(0) >> 1)
	rows := []observation{probe("one-mel-max-hidden", 1, maxInt, true), probe("two-mel-max-hidden", 2, maxInt, true), probe("four-mel-max-hidden", 4, maxInt, true), probe("eight-mel-max-hidden", 8, maxInt, true), probe("two-mel-half-boundary", 2, maxInt/2+1, true), probe("control-one-mel", 1, 2, false), probe("control-two-mel", 2, 2, false), probe("control-four-mel", 4, 2, false)}
	for _, row := range rows {
		if err := json.NewEncoder(os.Stdout).Encode(row); err != nil {
			panic(err)
		}
	}
}
