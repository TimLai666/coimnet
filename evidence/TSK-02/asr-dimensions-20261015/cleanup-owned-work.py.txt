sha = lambda raw: hashlib.sha256(raw).hexdigest()
read = lambda path: json.loads(path.read_text())


def verify(path, item):
    assert path.is_file() and not path.is_symlink(), path
    raw = path.read_bytes()
    assert len(raw) == item['bytes'] and sha(raw) == item['sha256'], path


def gate():

    assert len(sources) == 820
    for path, digest in sources.items():
        assert sha((root / path).read_bytes()) == digest, path

def inventory(directory):
    result = {}
    for current, dirs, files in os.walk(directory, followlinks=False):
        for name in dirs + files:
            path = Path(current) / name
            key = str(path.relative_to(directory))
            if path.is_symlink():
                result[key] = {'symlink': os.readlink(path)}
            elif path.is_file():
                raw = path.read_bytes()
                result[key] = {'bytes': len(raw), 'sha256': sha(raw)}
    return result


def preserve_check(key):
    name, inode, evidence = jobs[key]
    work = root / 'runs/.tmp' / name
    assert not work.is_symlink() and work.stat().st_ino == inode
    evidence = root / 'evidence' / evidence
    receipt = read(evidence / 'work-preservation.json')
    accounted = set()
    if key == '104':
        for item in receipt['remaining_noncached_files']:
            verify(work / item['path'], item)
            verify(evidence / item['archive'], item)
            accounted.add(item['path'])
        binaries = receipt['regenerable_binaries_not_archived']
        for archive in read(evidence / 'model-preservation.json')['archives']:
            packed = evidence / archive['archive']
            assert packed.stat().st_size == archive['compressed_bytes']
            assert sha(packed.read_bytes()) == archive['sha256']
            with tarfile.open(packed, 'r:gz') as handle:
                assert sorted(handle.getnames()) == sorted(archive['members'])
                for path, item in archive['members'].items():
                    raw = handle.extractfile(path).read()
                    assert len(raw) == item['bytes'] and sha(raw) == item['sha256']
                    verify(work / 'models' / path, item)
                    accounted.add('models/' + path)
        for path, item in read(evidence / 'data-preservation.json')['files'].items():
            relative = 'data/kmnist/' + Path(path).name
            verify(work / relative, item)
            accounted.add(relative)
    else:
        for item in receipt['files']:
            verify(work / item['source'], item)
            verify(root / item['archive'], item)
            accounted.add(item['source'])
        binaries = receipt['regenerable_executables']
    for item in binaries:
        path = item.get('path', item.get('source'))
        verify(work / path, item)
        accounted.add(path)
    actual = inventory(work)
    for path in actual:
        if path.split('/')[0] not in ['go-cache', 'go-mod-cache', 'go-tmp']:
            assert path in accounted, ('unpreserved work', key, path)
    return work, evidence, actual


def remove_readonly(function, path, error):
    target = Path(path)
    assert not target.is_symlink() and root / 'runs/.tmp' in target.parents
    os.chmod(target, target.stat().st_mode | 0o700)
    function(path)
