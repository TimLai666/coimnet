def git(*args):
    return subprocess.check_output(['git', *args], cwd=R, text=True).strip()


def update(p, transform):
    raw = p.read_bytes()
    after = transform(raw.decode())
    assert p.read_bytes() == raw, p
    p.write_text(after)

def status(text):
    a = text.index('[ticket130]')
    z = text.index('\n\n', a)
    part = text[a:z]
    old = '本票3／4，待清理與交付，'
    assert part.count(old) == 1
    part = part.replace(old, f'本票4／4。最後31項治理通過，{removed:,} bytes自有暫存已清除，原模型、資料、原始模組及其他工作保持。主要提交`{primary[:7]}`已推送且遠端讀回相同，見[清理](evidence/LRN-06/online-receptor-policy-20261016/cleanup.json)與[交付](evidence/LRN-06/online-receptor-policy-20261016/delivery.json)。')
    text = text[:a] + part + text[z:]
    old = '| in_progress | 3／4。完整工程與LRN-06驗收通過，待清理及交付 |'
    assert text.count(old) == 1
    text = text.replace(old, '| completed | 4／4。完整工程、清理與遠端提交核對完成，LRN-06 passed |')
    old = '先完成ticket130的清理與交付。之後的機制稽核先以公開API核對疑慮，重現缺陷後才修正。'
    assert text.count(old) == 1
    return text.replace(old, 'ticket130清理與交付已完成。後續機制稽核先以公開API核對疑慮，重現缺陷後才修正。')

def describe(name):
    if name in descriptions:
        return descriptions[name]
    short = Path(name).name
    if short.endswith('.log.gz'):
        return '保存無損壓縮完整工程日誌'
    if short.endswith('.log'):
        return '保存原始驗證命令日誌'
    if short.endswith('.py.txt'):
        return '保存驗證、清理或交付腳本'
    if short.endswith('.go.txt'):
        return '保存原公開缺陷診斷程式'
    if short.endswith('.json'):
        return '保存來源指紋、命令回條或驗證結果'
    if short == 'preserved-work.tar.gz':
        return '封存必要診斷、歷史輸入及隔離對照'
    if short == 'cleanup-inventory.json.gz':
        return '保存清理前完整內容指紋與大小'
    return '保存本票範圍、審查與驗收依據'
