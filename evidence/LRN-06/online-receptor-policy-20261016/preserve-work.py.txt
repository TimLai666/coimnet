"""Preserve all non-cache sources and inputs before removing owned work."""
import hashlib
import json
import tarfile
from pathlib import Path

assert __debug__
E = Path(__file__).resolve().parent
R = E.parents[2]
W = R / 'runs/.tmp/online-receptor-policy-20261016'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert W.stat().st_ino == 57786233 and not W.is_symlink()
assert json.loads((E / 'engineering-verification.json').read_text())['all_passed']
allowed = {'module-originals.json', 'public-baseline/main.go', 'public-output/main.go', 'old-chemical/main.go'}
allowed |= {str(Path(n).relative_to(W.relative_to(R))) for n in json.loads((E / 'old-chemical-inputs.json').read_text())['inputs']}
for name in ['act-native-gate', 'update-native-gate']:
    allowed |= {'mutations/' + name + '.go', 'mutations/' + name + '.json'}
files, binaries = {}, {}
for p in sorted(W.rglob('*')):
    if not p.is_file() or p.relative_to(W).parts[0] in ['go-cache', 'go-mod-cache', 'go-tmp']:
        continue
    name = str(p.relative_to(W))
    assert not p.is_symlink()
    item = {'bytes': p.stat().st_size, 'sha256': sha(p)}
    if name == 'final-root.test':
        binaries[name] = item
    else:
        assert name in allowed, name
        files[name] = item
assert set(files) == allowed and len(files) == 20
assert set(binaries) == {'final-root.test'}
archive = E / 'preserved-work.tar.gz'
assert not archive.exists()
with tarfile.open(archive, 'w:gz') as handle:
    for name in files:
        handle.add(W / name, arcname=name, recursive=False)
with tarfile.open(archive, 'r:gz') as handle:
    assert set(handle.getnames()) == set(files)
    for name, item in files.items():
        raw = handle.extractfile(name).read()
        assert len(raw) == item['bytes'] and hashlib.sha256(raw).hexdigest() == item['sha256']
for name, value in [('work-preservation.json', {'archive': archive.name, 'archive_bytes': archive.stat().st_size, 'archive_sha256': sha(archive), 'files': files}), ('discarded-binaries.json', binaries)]:
    with (E / name).open('x') as handle:
        json.dump(value, handle, indent=2)
        handle.write('\n')
print(json.dumps({'necessary_files': len(files), 'archive_bytes': archive.stat().st_size, 'reproducible_binaries': len(binaries)}))
