def once(text, before, after):
    assert text.count(before) == 1, before
    return text.replace(before, after)


pending = {}
def prepare(name, transform):
    p = R / name
    raw = p.read_bytes()
    pending[p] = (raw, transform(raw.decode()).encode())


def requirements(text):
    current = json.loads(text)
    baseline = json.loads(subprocess.check_output(['git', 'show', read(E / 'baseline.json')['head'] + ':docs/requirements-status.json'], cwd=R))
    assert current == baseline
    item = next(x for x in current['requirements'] if x['id'] == 'LRN-06')
    assert item['status'] == 'specified'
    item['status'] = 'passed'
    item['evidence'].append('../evidence/LRN-06/online-receptor-policy-20261016/verification.json')
    assert len(current['requirements']) == 91 and all(x['status'] == 'passed' for x in current['requirements'])
    return json.dumps(current, ensure_ascii=False, indent=2) + '\n'


prepare('docs/requirements-status.json', requirements)
