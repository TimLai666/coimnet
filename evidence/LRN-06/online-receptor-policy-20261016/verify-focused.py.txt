"""Verify focused receipts, independent controls, source and historical inputs."""
import collections
import datetime
import hashlib
import json
from pathlib import Path

R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()


def receipt(name, code=0):
    result = json.loads((E / (name + '.json')).read_text())
    assert result['exit_code'] == code, name
    assert sha(E / result['log']) == result['log_sha256'], name
    return result


def events(name, code=0):
    result = receipt(name, code)
    tests, packages = {}, {}
    for line in (E / result['log']).read_text().splitlines():
        try:
            x = json.loads(line)
        except ValueError:
            continue
        if x.get('Action') not in ['pass', 'fail', 'skip']:
            continue
        if x.get('Test'):
            key = (x['Package'], x['Test'])
            assert key not in tests, key
            tests[key] = x['Action']
        else:
            packages[x['Package']] = x['Action']
    assert tests, name
    return dict(collections.Counter(tests.values())), dict(collections.Counter(packages.values()))


results = {}
assert events('red-online', 1)[0] == {'pass': 112, 'fail': 56}
for mode in ['ordinary', 'race']:
    tests, packages = events('green-online-' + mode)
    assert tests == {'pass': 168} and packages == {'pass': 1}
    results['green-' + mode] = tests
    tests, packages = events('related-' + mode)
    assert tests == ({'pass': 3743, 'skip': 16} if mode == 'ordinary' else {'pass': 3742, 'skip': 17})
    assert packages == {'pass': 4}
    results['related-' + mode] = tests
    for name, expected in [('act-native-gate', {'pass': 130, 'fail': 38}), ('update-native-gate', {'pass': 137, 'fail': 31})]:
        tests, packages = events('mutation-' + name + '-' + mode, 1)
        assert tests == expected and packages == {'fail': 1}
        results['mutation-' + name + '-' + mode] = tests
    for family in ['public', 'output']:
        receipt(family + '-green-' + mode)
        receipt(family + '-red-' + mode)
        old_lines = (E / (family + '-red-' + mode + '.log')).read_text().splitlines()
        new_lines = (E / (family + '-green-' + mode + '.log')).read_text().splitlines()
        assert len(old_lines) == len(new_lines) == 16
        unchanged = 0
        for before, after in zip(old_lines, new_lines):
            old, new = json.loads(before), json.loads(after)
            assert (old['policy'], old['receptor_gate'], old['pulse']) == (new['policy'], new['receptor_gate'], new['pulse'])
            assert not new['act_plastic_learned'] and not new['act_base_changed']
            assert new['act_plastic_after'] == new['act_plastic_before'] == [0]
            if new['policy'] in ['plastic', 'mixed']:
                assert new['update_error'] == '' and new['update_report']['plastic_steps'] == 1
            if not new['receptor_gate']:
                assert before == after
                unchanged += 1
        assert unchanged == 8
        results[family + '-' + mode] = {'rows': 16, 'unchanged_no_receptor_controls': unchanged}
    receipt('old-chemical-baseline-' + mode)
    receipt('old-chemical-green-' + mode)
    before = (E / ('old-chemical-baseline-' + mode + '.log')).read_bytes()
    after = (E / ('old-chemical-green-' + mode + '.log')).read_bytes()
    assert before == after and len(json.loads(after)) == 6
    results['historical-' + mode] = {'snapshots': 6, 'continuation_byte_identical': True}

source = json.loads((E / 'formal-source.json').read_text())
assert len(source) == 846
for name, digest in source.items():
    assert sha(R / name) == digest, name
for name, digest in json.loads((E / 'foreign-before.json').read_text()).items():
    assert sha(Path(name)) == digest, name
for name, digest in json.loads((R / 'evidence/MOD-04/lif-threshold-floor-20261015/protected-models.json').read_text()).items():
    assert sha(R / name) == digest, name
for name, item in json.loads((R / 'evidence/MOD-04/lif-threshold-floor-20261015/persistent-data-preservation.json').read_text())['persistent_data_files'].items():
    assert sha(R / name) == item['sha256'], name
for name, digest in json.loads((E / 'old-chemical-inputs.json').read_text())['inputs'].items():
    assert sha(R / name) == digest, name
results.update(all_passed=True, verified_at=datetime.datetime.now(datetime.timezone.utc).isoformat(), source_count=len(source))
with (E / 'diagnostic-verification.json').open('x') as handle:
    json.dump(results, handle, indent=2)
    handle.write('\n')
print(json.dumps(results))
