def summarize(slug):
    d = json.loads((evidence / (slug + '.json')).read_text())
    raw = gzip.decompress((root / d['log']).read_bytes())
    events = [json.loads(line) for line in raw.splitlines()]
    failed = [v for v in events if v.get('Action') == 'fail']
    assert not failed
    assert b'WARNING: DATA RACE' not in raw
    result = dict(top_pass_events=sum(v.get('Action') == 'pass' and bool(v.get('Test')) and '/' not in v['Test'] for v in events), all_test_pass_events=sum(v.get('Action') == 'pass' and bool(v.get('Test')) for v in events), packages_pass=sum(v.get('Action') == 'pass' and not v.get('Test') for v in events), failure_events=failed, skipped_test_events=[v for v in events if v.get('Action') == 'skip' and v.get('Test')], race_warnings=0)
    assert result['packages_pass'] == 58
    summary = evidence / (slug + '-summary.json')
    if summary.exists():
        assert json.loads(summary.read_text()) == result
    else:
        with summary.open('x') as out:
            json.dump(result, out, indent=2)
            out.write('\n')
    return result
