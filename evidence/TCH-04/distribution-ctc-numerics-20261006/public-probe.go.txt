package main
import("fmt"; "github.com/TimLai666/coimnet/distill"; "github.com/TimLai666/coimnet/tasks/ocr/ctc")
func main(){
 d:=distill.DistributionDistiller{Temperature:1,Scale:1,Mix:1,Alignment:distill.Alignment{Rule:distill.AlignmentIdentical,TeacherVocabHash:"same",StudentVocabHash:"same"}}
 for _,shift:=range []float64{0,1e16}{
  l,g,_,e:=d.Loss(distill.TeacherDistribution{Probabilities:[]float64{.5,.5}},[]float64{shift,shift},-1)
  fmt.Printf("distill shift=%g loss=%.17g grad=%v error=%v\n",shift,l,g,e)
  l,cg,e:=ctc.Loss([][]float64{{shift,shift}},[]int{1})
  fmt.Printf("ctc shift=%g loss=%.17g grad=%v error=%v\n",shift,l,cg,e)
 }
}
