# -*- coding: utf-8 -*-
"""Record owned cleanup while Git delivery remains pending."""
import hashlib
import json
from pathlib import Path

assert __debug__
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
cleanup = read(E / "cleanup.json")
assert cleanup["directory_removed"] and not (R / cleanup["owned_directory"]).exists()
assert cleanup["source_entries_preserved"] == 844 and cleanup["original_model_files_preserved"] == 18 and cleanup["persistent_data_files_preserved"] == 6
for name, digest in read(E / "formal-source-v2.json").items():
    assert sha(R / name) == digest, name
for name, digest in read(E / "foreign-before.json").items():
    assert sha(Path(name)) == digest, name
assert read(E / "engineering-v2-verification.json")["all_passed"]
path = E / "verification.json"
before = path.read_bytes()
value = json.loads(before)
assert value["engineering_passed"] and value["diagnostic_passed"]
value["remaining_delivery"].remove("owned cleanup")
value["observed_result"]["own_cleanup"] = {"receipt": "cleanup.json", "removed_bytes": cleanup["removed_bytes"]}
assert path.read_bytes() == before
path.write_text(json.dumps(value, indent=2) + "\n")
path = R / "delivery-status.md"
before = path.read_bytes()
text = before.decode()
old = "本票 3／4，暫存清理及交付尚待完成。"
assert text.count(old) == 1
text = text.replace(old, "本票 3／4，本輪暫存已清除，提交與遠端讀回尚待完成。")
assert path.read_bytes() == before
path.write_text(text)
path = E / "root-review.md"
before = path.read_bytes()
text = before.decode()
old = "完整競態與後續工程仍在執行。"
assert text.count(old) == 1
text = text.replace(old, "844 份來源的完整一般／競態各 58 套件通過，8,104／8,103 pass、23／24 skip，原有各 8,028 個終端事件保持，只新增各 99 pass。格式、build、vet、固定 Insyra v0.3.4、模組、31 項治理與最後 metadata 的 31 項治理通過。")
text += "\n本輪自有暫存已清除，28 個必要輸入、診斷與隔離回退來源已封存並重新比對。18 個原模型、六份必要資料、15,046 個原始模組檔、32 個其他工作檔案與兩個既有暫存根經清理前後核對保持。實際移除量見 cleanup.json，提交與遠端讀回以 delivery.json 為準。\n"
assert path.read_bytes() == before
path.write_text(text)
print(json.dumps({"owned_temp_removed": True, "requirements_passed": "90/91", "remaining_delivery": value["remaining_delivery"]}))
