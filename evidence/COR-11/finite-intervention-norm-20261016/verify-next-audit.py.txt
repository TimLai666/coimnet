"""Verify a separate public consolidation report defect without changing frozen source."""
import hashlib
import json
from pathlib import Path

assert __debug__
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
source = read(E / "formal-source.json")
assert len(source) == 843 and all(sha(R / p) == h for p, h in source.items())
results = {}
for mode in ["ordinary", "race"]:
    receipt = read(E / ("next-consolidation-layers-" + mode + ".json"))
    assert receipt["exit_code"] == 0 and sha(E / receipt["log"]) == receipt["log_sha256"]
    rows = [json.loads(x) for x in (E / receipt["log"]).read_text().splitlines()]
    assert len(rows) == 9
    assert {(x["case"], x["layer"]) for x in rows} == {(c, k) for c in ["upper_finite", "upper_overflow", "obvious_overflow"] for k in ["base", "slow", "plastic"]}
    for row in rows:
        if row["case"] == "upper_finite":
            assert row["applied"] and not row["error"] and row["used_after"] == 1 and row["state_changed"]
            assert not row["expected_unrepresentable"] and row["reported_norm"] == row["expected_norm"] == "1.7976931348623157e+308"
        elif row["case"] == "upper_overflow":
            assert row["applied"] and not row["error"] and row["used_after"] == 1 and row["state_changed"]
            assert row["expected_unrepresentable"] and row["expected_norm"] == "+Inf" and row["reported_norm"] == "1.7976931348623157e+308"
        else:
            assert not row["applied"] and row["error"] == "consolidation report norm is not finite"
            assert not row["state_changed"] and row["used_after"] == 0 and row["expected_unrepresentable"]
    results[mode] = {"receipt": "next-consolidation-layers-" + mode + ".json", "rows": rows}
assert results["ordinary"]["rows"] == results["race"]["rows"]
result = {"requirement": "MOD-06", "severity": "P2", "confirmed": True, "fixed": False, "reproduction_command": "go run [-race] ./runs/.tmp/intervention-finite-norm-20261016/next-consolidation-layers", "environment": "command-environment.json", "input_fingerprints": {"probe": "next-consolidation-layers.go.txt", "probe_sha256": sha(E / "next-consolidation-layers.go.txt"), "formal_source": "formal-source.json", "formal_source_sha256": sha(E / "formal-source.json")}, "observed_result": results, "test_log": ["next-consolidation-layers-ordinary.log", "next-consolidation-layers-race.log"], "scope": "Separate existing Consolidate defect. Ticket128 production source remains frozen. Valid finite state is accepted, exact 8192-bit norm is unrepresentable, sequential Hypot misreports MaxFloat64 and consumes a write slot. Base, slow and plastic report paths all reproduced."}
with (E / "next-audit-verification.json").open("x") as handle:
    json.dump(result, handle, indent=2)
    handle.write("\n")
print(json.dumps({"confirmed": True, "fixed": False, "public_cases_each_build": 9, "affected_report_layers": 3, "source_unchanged": True}))
