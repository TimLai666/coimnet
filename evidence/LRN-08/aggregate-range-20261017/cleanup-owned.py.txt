sha = lambda b: hashlib.sha256(b).hexdigest()

def gate():
    source = read(E / "formal-source.json")
    assert len(source) == 855
    paths = {p for p in subprocess.check_output(["git", "ls-files", "-co", "--exclude-standard", "-z"], cwd=R).decode().split("\0") if p and (p.endswith(".go") or p in ("go.mod", "go.sum"))}
    assert paths == set(source), "formal source paths changed"
    for name, digest in source.items():
        assert sha((R / name).read_bytes()) == digest, name
    models = read(R / "evidence/MOD-04/lif-threshold-floor-20261015/protected-models.json")
    assert len(models) == 18
    for name, digest in models.items():
        assert sha((R / name).read_bytes()) == digest, name
    data = read(R / "evidence/MOD-04/lif-threshold-floor-20261015/persistent-data-preservation.json")["persistent_data_files"]
    assert len(data) == 6
    for name, item in data.items():
        b = (R / name).read_bytes()
        assert len(b) == item["bytes"] and sha(b) == item["sha256"], name
    foreign = read(E / "foreign-before.json")
    assert len(foreign) == 32
    for name, digest in foreign.items():
        assert sha(pathlib.Path(name).read_bytes()) == digest, name
    for name, inode in PROTECTED.items():
        p = R / "runs/.tmp" / name
        assert not p.is_symlink() and p.stat().st_ino == inode


def inventory(root):
    result = {}
    for base, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            p = pathlib.Path(base) / name
            key = str(p.relative_to(root))
            if p.is_symlink():
                result[key] = {"symlink": os.readlink(p)}
            elif p.is_dir():
                result[key] = {"directory": True, "inode": p.stat().st_ino}
            else:
                assert p.is_file(), p
                b = p.read_bytes()
                result[key] = {"bytes": len(b), "sha256": sha(b)}
    return result
