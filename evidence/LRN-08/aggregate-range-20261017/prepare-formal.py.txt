"""Create one owned, project-local work root from verified pinned modules."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import tarfile

if not __debug__:
    raise SystemExit("optimized Python cannot skip preparation checks")
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
P = R / "evidence/LRN-10/chemical-evaluation-reset-20261017"
W = R / "runs/.tmp/aggregate-range-20261017"
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=R, text=True).strip()
assert head == "3442f8afb98674dba84f5e76d3fdd9ccd1738a1b"
source = json.loads((P / "formal-source.json").read_text())
for p, digest in source.items():
    assert sha(R / p) == digest, p
foreign = json.loads((P / "foreign-before.json").read_text())
for p, digest in foreign.items():
    assert sha(Path(p)) == digest, p
assert not W.exists() and not W.is_symlink()
for name in ["command-environment.json", "baseline.json", "formal-baseline-source.json", "module-manifest-anchor.json", "run-command.py.txt"]:
    assert not (E / name).exists(), name
W.mkdir()
for name in ["go-cache", "go-mod-cache", "go-tmp"]:
    (W / name).mkdir()
preserved = json.loads((P / "work-preservation.json").read_text())
archive = P / preserved["archive"]
assert sha(archive) == preserved["archive_sha256"]
with tarfile.open(archive) as handle:
    raw = handle.extractfile("module-originals.json").read()
assert hashlib.sha256(raw).hexdigest() == preserved["payload"]["module-originals.json"]
originals = json.loads(raw)
assert len(originals) == 15046
global_modules = Path("/Users/timlai/go/pkg/mod")
roots = set()
for name, digest in originals.items():
    p = Path(name)
    assert p.is_file() and not p.is_symlink() and sha(p) == digest, name
    relative = p.relative_to(global_modules)
    index = next(i for i, part in enumerate(relative.parts) if "@" in part)
    roots.add(global_modules.joinpath(*relative.parts[:index + 1]))
assert len(roots) == 45
metadata_files = 0
for root in sorted(roots):
    relative = root.relative_to(global_modules)
    shutil.copytree(root, W / "go-mod-cache" / relative, copy_function=shutil.copy2)
    module, version = str(relative).rsplit("@", 1)
    for name in [version + ".mod", version + ".info", version + ".ziphash", version + ".lock", "list"]:
        src = global_modules / "cache/download" / module / "@v" / name
        if src.is_file():
            dst = W / "go-mod-cache/cache/download" / module / "@v" / name
            dst.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(src, dst)
            metadata_files += 1
for name, digest in originals.items():
    p = W / "go-mod-cache" / Path(name).relative_to(global_modules)
    assert sha(p) == digest and p.stat().st_ino != Path(name).stat().st_ino, name
(W / "module-originals.json").write_bytes(raw)
settings = {
    "GOMAXPROCS": "2", "GOFLAGS": "-mod=readonly -p=2",
    "GOMEMLIMIT": "8GiB", "GOCACHE": str(W / "go-cache"),
    "GOMODCACHE": str(W / "go-mod-cache"), "GOTMPDIR": str(W / "go-tmp"),
    "TMPDIR": str(W / "go-tmp"), "GOTOOLCHAIN": "local", "GOSUMDB": "off",
    "GOTELEMETRY": "off", "GOPROXY": "file:///Users/timlai/go/pkg/mod/cache/download",
}
with (E / "command-environment.json").open("x") as handle:
    json.dump(settings, handle, indent=2)
    handle.write("\n")
for name in ["foreign-before.json", "protected-models.json", "persistent-data.json"]:
    b = (P / name).read_bytes()
    with (E / name).open("xb") as handle:
        handle.write(b)
with (E / "formal-baseline-source.json").open("x") as handle:
    json.dump(source, handle, indent=2)
    handle.write("\n")
with (E / "module-manifest-anchor.json").open("x") as handle:
    json.dump({"path": str(W / "module-originals.json"), "sha256": hashlib.sha256(raw).hexdigest(), "entries": len(originals)}, handle, indent=2)
    handle.write("\n")
with (E / "baseline.json").open("x") as handle:
    json.dump({"head": head, "work_directory": str(W.relative_to(R)), "work_inode": W.stat().st_ino, "source_count": len(source), "module_files": len(originals), "modules": len(roots), "metadata_files": metadata_files, "at": datetime.datetime.now(datetime.timezone.utc).isoformat(), "previous_goal_turn": {"classification": "progress", "evidence": "Ticket135 delivered at3442f8a, full58-package ordinary/race, owned2.15GB cleaned; current source and cleanup receipts rechecked."}}, handle, indent=2)
    handle.write("\n")
with tarfile.open(archive) as handle:
    wrapper = handle.extractfile("run.py").read()
assert hashlib.sha256(wrapper).hexdigest() == preserved["payload"]["run.py"]
with (E / "run-command.py.txt").open("xb") as handle:
    handle.write(wrapper)
with (W / "run.py").open("xb") as handle:
    handle.write(wrapper)
with (W / "environment.json").open("x") as handle:
    json.dump(settings, handle, indent=2)
    handle.write("\n")
for name in ["README.md", "ENG.md", "delivery-status.md", "docs/requirements-status.json"]:
    with (W / (name.replace("/", "__") + ".before")).open("xb") as handle:
        handle.write((R / name).read_bytes())
with (W / "original-continual.go.txt").open("xb") as handle:
    handle.write((R / "experiment/continual_math.go").read_bytes())
print(json.dumps({"source_count": len(source), "modules": len(roots), "module_files": len(originals), "work_inode": W.stat().st_ino}), flush=True)
