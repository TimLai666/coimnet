type rootRequestObservation struct {
 Case string `json:"case"`
 Loss float64 `json:"loss"`
 Expected float64 `json:"expected_uniform_loss,omitempty"`
 Error string `json:"error,omitempty"`
 Panic string `json:"panic,omitempty"`
 Gradient []float64 `json:"gradient,omitempty"`
 Mismatch bool `json:"uniform_loss_mismatch,omitempty"`
}

func rootObserveRequestHead(shortShapes bool) (out rootRequestObservation) {
 old:=ImageShapes
 defer func(){ImageShapes=old}()
 defer func(){if v:=recover();v!=nil{out.Panic=fmt.Sprint(v)}}()
 out.Case="default_public_step"
 if shortShapes { out.Case="short_shapes_public_step";ImageShapes=append([]string(nil),old[:2]...) }
 c:=DefaultGeneratorConfig(ModalityImage);c.Embed=4;c.Hidden=8;c.Settle=1;c.Seed=17
 h,err:=NewRequestHead(c)
 if err!=nil{out.Error=err.Error();return}
 _,out.Loss,err=h.Step(context.Background(),"red square")
 if err!=nil{out.Error=err.Error()}
 return
}
