import datetime, hashlib, json, os, pathlib, subprocess, time
R=pathlib.Path.cwd()
E=R/"evidence/TSK-04/report-config-ownership-20261017"
W=R/"runs/.tmp/report-config-ownership-20261017"
assert W.stat().st_ino==59884782
source=(E/"probe-valid.go.txt").read_bytes()
assert source==(W/"main-valid.go").read_bytes()
env=dict(os.environ)
env.update(json.loads((E/"environment.json").read_text()))
