def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def result(name):
    receipt=json.loads((f/(name+'.json')).read_text())
    assert receipt['exit_code']==0,name
    assert sha(f/receipt['log'])==receipt['log_sha256'],name
    return receipt
def counts(name):
    result(name)
    rows=[json.loads(x)for x in(f/(name+'.log')).read_text().splitlines()]
    tests=collections.Counter(x['Action']for x in rows if x.get('Test')and x.get('Action')in['pass','fail','skip'])
    packages=collections.Counter(x['Action']for x in rows if not x.get('Test')and x.get('Action')in['pass','fail'])
    assert not tests['fail']and not packages['fail']
    assert packages['pass']==58,(name,packages)
    return {'test_events':dict(tests),'package_events':dict(packages)}
