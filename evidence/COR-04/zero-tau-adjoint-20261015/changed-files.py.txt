"""List every delivered path without staging unrelated work."""
import json
from pathlib import Path
import subprocess

R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
target = E / "changed-files.md"
previous = target.read_bytes() if target.exists() else None
summaries = {
    "dynamics/continuous.go": "連續核心保留原累加式，只恢復非有限的膜時間導數。",
    "dynamics/lif.go": "LIF 核心恢復不應期內原本應為零的膜時間導數。",
    "dynamics/mixed.go": "混合核心沿用相同的例外恢復與既有梯度檢查。",
    "dynamics/recompute.go": "連續核心的分段重算補上相同恢復。",
    "dynamics/lif_recompute.go": "LIF 分段重算補上相同恢復。",
    "dynamics/coupled_reverse.go": "連續與 LIF 的化學耦合反向保留正常值，另算例外恢復值。",
    "dynamics/mixed_coupled.go": "混合化學耦合反向保留正常捨入與累加有限值檢查。",
    "dynamics/vector_core.go": "逐向量分量保留原累加式並恢復可表示導數。",
    "dynamics/gpu_adapter.go": "GPU 主機端反向補上相同恢復，裝置限制保持。",
    "dynamics/tau_gradient.go": "新增有界的精確差乘積恢復，原始非有限值保持拒絕。",
    "dynamics/zero_tau_adjoint_test.go": "凍結公開回歸，涵蓋核心、重算、化學耦合、向量與 Metal。",
    "ENG.md": "記錄膜時間導數的正常算式與例外恢復契約。",
    "delivery-status.md": "記錄本票驗收、清理與交付狀態。",
    "docs/requirements-status.json": "連結受影響需求的實際重現與驗收證據。",
    "docs/tickets/113-membrane-tau-derivative.md": "記錄根因、驗收條件及五項完成狀態。",
}
files = set(summaries)
files.update(str(p.relative_to(R)) for p in E.rglob("*") if p.is_file())
files.add(str((E / "changed-files.md").relative_to(R)))
foreign = {str(Path(p).relative_to(R)) for p in json.loads((E / "foreign-before.json").read_text())}
changed = set(subprocess.check_output(["git", "diff", "--name-only", "HEAD", "-z"], cwd=R).decode().split("\0"))
changed.update(subprocess.check_output(["git", "ls-files", "--others", "--exclude-standard", "-z"], cwd=R).decode().split("\0"))
changed.discard("")
assert not (changed - files - foreign), changed - files - foreign
assert not (files & foreign)
rows = ["# 本輪檔案變更", "", "本表逐檔列出 ticket 113 交付內容。既有 AGENTS.md 與兩組其他工作證據不在交付範圍。", "", "| 檔案 | 變更摘要 |", "|---|---|"]
for name in sorted(files):
    p = R / name
    summary = summaries.get(name)
    if summary is None:
        if name.endswith(".log") or name.endswith(".log.gz"):
            summary = "保留 " + p.name + " 的原始命令輸出。"
        elif name.endswith(".py.txt"):
            summary = "保存 " + p.name + " 的驗收、證據或清理程式。"
        elif name.endswith(".go.txt"):
            summary = "保存 " + p.name + " 的公開重現、凍結測試或相容性程式。"
        elif p.name == "preserved-work.tar.gz":
            summary = "保存人工模型、來源版本、驅動程式及原始模組指紋，供清理後核對。"
        elif p.name == "cleanup-inventory.json.gz":
            summary = "保存刪除前逐項內容與指紋。"
        elif p.name == "changed-files.md":
            summary = "逐檔列出本輪交付範圍。"
        else:
            summary = "記錄 " + p.name + " 的計畫、審查、指紋或驗證結果。"
    rows.append("| [" + name + "](" + str(p) + ") | " + summary + " |")
if previous is None:
    assert not target.exists()
    with target.open("x") as handle:
        handle.write("\n".join(rows) + "\n")
else:
    assert target.read_bytes() == previous
    target.write_text("\n".join(rows) + "\n")
print(json.dumps({"delivered_paths": len(files), "protected_foreign_paths": len(foreign)}))
