    raise SystemExit("optimized Python cannot skip verification gates")

sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()

for p, digest in source.items():
    assert sha(R / p) == digest, p

    assert step["exit_code"] == 0 and not step["acceptance_errors"], step["name"]
    assert sha(E / step["log"]) == step["log_sha256"]

assert candidate["all_passed"]
unit = read(E / "engineering-unit.json")
race = read(E / "engineering-race.json")
assert unit["test_events"] == {"pass": 6040, "skip": 22}
assert sum(race["test_events"].values()) == 6062 and not race["test_events"].get("fail")
for x in [unit, race]:
    assert len(x["added_events"]) == 27
    assert all(v["before"] in ["pass", "skip"] and v["after"] in ["pass", "skip"] for v in x["changed_events"])
    assert all(v["action"] == "pass" or (v["action"] == "skip" and v["test"] == "TestZeroTauAdjointGPU") for v in x["added_events"])
    assert x["package_events"] == {"pass": 58, "skip": 3}
