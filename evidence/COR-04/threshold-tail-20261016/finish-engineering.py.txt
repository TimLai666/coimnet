"""Complete the requirement gate only after all final-source checks pass."""
import collections
import datetime
import gzip
import hashlib
import json
from pathlib import Path
import subprocess

if not __debug__:
    raise SystemExit('optimized Python cannot skip acceptance checks')
r = Path(__file__).resolve().parents[3]
e = Path(__file__).resolve().parent
w = r / 'runs/.tmp/threshold-tail-20261016'
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
source = read(e / 'formal-source.json')
assert len(source) == len(read(e / 'source-before.json')) + 3
paths = subprocess.check_output(['git', 'ls-files', '--cached', '--others', '--exclude-standard', '-z', '--', '*.go', 'go.mod', 'go.sum'], cwd=r).split(b'\0')
assert {p.decode() for p in paths if p} == set(source)
for path, digest in source.items():
    assert sha(r / path) == digest, path
engineering = read(e / 'engineering-verification.json')
assert engineering['all_passed'] and len(engineering['steps']) == 9
for step in engineering['steps']:
    assert step['exit_code'] == 0 and not step['acceptance_errors']
    assert sha(e / step['log']) == step['log_sha256']
candidate = read(e / 'candidate-verification.json')
assert candidate['all_passed'] and candidate['source_count'] == len(source)
new_count = candidate['regression_events_each_build']['pass']
assert new_count > 0
gain_checks = ['TestNeutralGainRejectsExcludedRange', 'TestNeutralGainRetainsValidRanges']
for name in ['unit', 'race']:
    result = read(e / ('engineering-' + name + '.json'))
    assert result['package_events'] == {'pass': 58, 'skip': 4}
    assert result['prior_event_count'] == 6136 and not result['changed_events']
    assert len(result['added_events']) == new_count
    assert all(x['action'] == 'pass' and x['test'].startswith('TestThresholdTail') for x in result['added_events'])
    events = [json.loads(line) for line in gzip.decompress((e / result['log']).read_bytes()).splitlines()]
    for test in gain_checks:
        assert any(x.get('Package') == 'github.com/TimLai666/coimnet/modulation' and x.get('Test') == test and x.get('Action') == 'pass' for x in events), test
for label in ['last-bit-final', 'last-bit-final-race']:
    receipt = read(e / (label + '.json'))
    assert receipt['exit_code'] == 0 and sha(e / receipt['log']) == receipt['log_sha256']
    assert [x['slope_bits'] for x in read(e / receipt['log'])] == [1, 1, 0, 0]
frozen = read(e / 'frozen-test.json')
for path, digest in frozen['files'].items():
    assert sha(r / path) == digest, path
unit, race = read(e / 'engineering-unit.json'), read(e / 'engineering-race.json')
result = dict(
    requirement=['COR-04', 'COR-09'], engineering_passed=True,
    verified_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),
    reproduction_command='Per-command receipts, baseline-controls.py.txt, candidate-commands.py.txt and engineering.py.txt retain exact commands; regression filter ^TestThresholdTail.',
    environment=read(e / 'environment.json'),
    input_fingerprints=dict(base_commit=read(e / 'baseline.json')['head'], formal_source='formal-source.json',
                            formal_source_sha256=sha(e / 'formal-source.json'), frozen_tests=frozen,
                            reference_files={p: sha(e / p) for p in ['reference-input.json', 'additional-reference.json', 'last-bit-reference.json', 'last-bit-probe.go.txt']}),
    observed_result=dict(source_count=len(source), formal_go_files=engineering['formal_go_files'],
                          ordinary_test_events=unit['test_events'], race_test_events=race['test_events'],
                          package_events_per_build=unit['package_events'], prior_terminal_events_unchanged_each_build=6136,
                          new_passing_regression_events_each_build=new_count,
                          normal_core_records_bytes_equal=32, old_core_state_new_process_continuations=32,
                          old_chemical_model_new_process_continuations=12, chemical_updates_after_resume_each=1,
                          public_cases=18, forward_mismatch=0, theta_gradient_mismatch=0,
                          last_bit_slope_bits_each_build=[1, 1, 0, 0],
                          approved_gain_range_checks_passed_each_build=gain_checks,
                          remaining_delivery=['final governance', 'owned temporary cleanup', 'commit and remote readback']),
    root_review='root-review.md', candidate_acceptance='candidate-verification.json', engineering_details='engineering-verification.json',
    test_log=['regression-before.log', 'regression-before-race.log', 'regression-candidate.log', 'regression-candidate-race.log',
              'engineering-unit.log.gz', 'engineering-race.log.gz', 'engineering-governance.log', 'last-bit-final.log', 'last-bit-final-race.log'],
    limits='macOS arm64 numerical and compatibility checks. No new real-task accuracy, biological completeness or absolute-zero-bug conclusion.')
with (e / 'verification.json').open('x') as f:
    json.dump(result, f, indent=2)
    f.write('\n')
p = r / 'docs/requirements-status.json'
raw = p.read_bytes()
status = json.loads(raw)
for item in status['requirements']:
    if item['id'] in result['requirement']:
        assert item['status'] == 'implemented_unverified'
        item['status'] = 'passed'
        item['evidence'].append('../evidence/COR-04/threshold-tail-20261016/verification.json')
assert p.read_bytes() == raw
p.write_text(json.dumps(status, ensure_ascii=False, indent=2) + '\n')
assert collections.Counter(x['status'] for x in status['requirements']) == {'passed': 91}
binary = w / 'final-root.test'
with (e / 'closure-binary.json').open('x') as f:
    json.dump(dict(path=binary.name, bytes=binary.stat().st_size, sha256=sha(binary)), f, indent=2)
    f.write('\n')
print(json.dumps(dict(engineering_passed=True, source_count=len(source), requirement_status='91/91 passed')))
