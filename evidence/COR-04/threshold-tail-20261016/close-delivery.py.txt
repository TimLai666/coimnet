# coding: utf-8
"""Record completed delivery after actual remote readback."""
import datetime
import hashlib
import json
from pathlib import Path

r = Path(__file__).resolve().parents[3]
e = Path(__file__).resolve().parent
read = lambda p: json.loads(p.read_text())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
source, primary, remote, cleanup = [read(e / p) for p in ['formal-source.json', 'committed-primary.json', 'remote-primary.json', 'cleanup.json']]
assert primary['primary_commit'] == remote['remote_commit'] and remote['matches']
assert cleanup['directory_removed'] and not (r / cleanup['owned_directory']).exists()
for path, digest in source.items():
    assert sha(r / path) == digest, path
for path, digest in read(e / 'foreign-before.json').items():
    assert sha(Path(path)) == digest, path
delivery = dict(
    requirement=['COR-04', 'COR-09'], delivered_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),
    reproduction_command='git push origin HEAD:refs/heads/codex/receptor-chemistry-training; git ls-remote --exit-code origin refs/heads/codex/receptor-chemistry-training; see remote-primary.json and committed-primary.json.',
    environment=read(e / 'environment.json'),
    input_fingerprints=dict(formal_source_sha256=sha(e / 'formal-source.json'), source_count=len(source),
                            fix_commit=primary['fix_commit'], primary_commit=primary['primary_commit']),
    observed_result=dict(remote=remote['remote_commit'], branch='codex/receptor-chemistry-training',
                          all_source_and_payload_blobs_match=True, unique_primary_blob_checks=primary['unique_committed_blob_checks'],
                          original_foreign_paths_preserved=32, owned_work_root_removed=True,
                          removed_bytes=cleanup['removed_bytes'], necessary_files_preserved=25,
                          normal_core_records_equal=32, old_neural_continuations_equal=32, old_chemical_continuations_equal=12),
    test_log=['remote-primary.json', 'committed-primary.json', 'cleanup.json', 'engineering-unit.log.gz', 'engineering-race.log.gz'],
    limits='Verified bounded numerical fix and delivery; no new task-quality, biology or absolute-zero-bug claim.')
with (e / 'delivery.json').open('x') as f:
    json.dump(delivery, f, indent=2)
    f.write('\n')

p = e / 'verification.json'
old = p.read_bytes()
v = json.loads(old)
v['observed_result']['remaining_delivery'] = []
v['observed_result']['delivery'] = dict(receipt='delivery.json', fix_commit=primary['fix_commit'], primary_commit=primary['primary_commit'], remote_readback=remote['remote_commit'])
assert p.read_bytes() == old
p.write_text(json.dumps(v, indent=2) + '\n')
