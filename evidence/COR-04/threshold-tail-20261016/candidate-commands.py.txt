"""Execute frozen candidate regressions and old-model controls."""
import hashlib
import json
from pathlib import Path
import subprocess

root = Path(__file__).resolve().parents[3]
evidence = Path(__file__).resolve().parent
work = root / 'runs/.tmp/threshold-tail-20261016'
source = json.loads((evidence / 'formal-source.json').read_text())
commands = [('reproduction-candidate', ['go', 'run', './runs/.tmp/threshold-tail-20261016/probe', str(evidence / 'reference-input.json')])]
for suffix, flags in [('', []), ('-race', ['-race'])]:
    commands.extend([
        ('regression-candidate' + suffix, ['go', 'test', *flags, '-json', './dynamics', './experiment', '-run', '^TestThresholdTail', '-count=1']),
        ('core-candidate' + suffix, ['go', 'run', *flags, './runs/.tmp/threshold-tail-20261016/core-compat']),
        ('core-state-resume-candidate' + suffix, ['go', 'run', *flags, './runs/.tmp/threshold-tail-20261016/core-compat', 'resume', str(evidence / ('core-before' + suffix + '.log'))]),
        ('chemical-resume-candidate' + suffix, ['go', 'run', *flags, './runs/.tmp/threshold-tail-20261016/chemical-compat', 'resume', str(work / ('chemical-models-race' if flags else 'chemical-models-ordinary'))]),
    ])
for label, command in commands:
    for path, digest in source.items():
        assert hashlib.sha256((root / path).read_bytes()).hexdigest() == digest, path
    subprocess.run(['python3', str(evidence / 'run-command.py.txt'), label, *command], cwd=root, check=True)
