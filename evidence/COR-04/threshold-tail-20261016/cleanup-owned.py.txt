import gzip
import hashlib
import json
import os
import pathlib
import shutil

if not __debug__:
    raise SystemExit("Optimized Python disables required checks; cleanup refused")

R = pathlib.Path(__file__).resolve().parents[3]
E = pathlib.Path(__file__).resolve().parent
W = R / "runs/.tmp/threshold-tail-20261016"
PROTECTED = {"62": 49680181, "real-task-quality-20261012": 50419696}
sha = lambda b: hashlib.sha256(b).hexdigest()
read = lambda p: json.loads(p.read_text())


def gate():
    source = read(E / "formal-source.json")
    assert len(source) == len(json.loads((E / 'source-before.json').read_text())) + 3
    for name, digest in source.items():
        assert sha((R / name).read_bytes()) == digest, name
    for name, digest in read(R / "evidence/MOD-04/lif-threshold-floor-20261015/protected-models.json").items():
        assert sha((R / name).read_bytes()) == digest, name
    for name, item in read(R / "evidence/MOD-04/lif-threshold-floor-20261015/persistent-data-preservation.json")["persistent_data_files"].items():
        b = (R / name).read_bytes()
        assert len(b) == item["bytes"] and sha(b) == item["sha256"], name
    for name, digest in read(E / "foreign-before.json").items():
        assert sha(pathlib.Path(name).read_bytes()) == digest, name
    for name, inode in PROTECTED.items():
        p = R / "runs/.tmp" / name
        assert not p.is_symlink() and p.stat().st_ino == inode


def inventory(root):
    result = {}
    for base, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            p = pathlib.Path(base) / name
            key = str(p.relative_to(root))
            if p.is_symlink():
                result[key] = {"symlink": os.readlink(p)}
            elif p.is_dir():
                result[key] = {"directory": True, "inode": p.stat().st_ino}
            else:
                assert p.is_file(), p
                b = p.read_bytes()
                result[key] = {"bytes": len(b), "sha256": sha(b)}
    return result

for name, digest in module_source.items():
    assert sha(pathlib.Path(name).read_bytes()) == digest, name

    handle.write(packed)
