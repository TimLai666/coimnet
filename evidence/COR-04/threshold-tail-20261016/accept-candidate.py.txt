"""Check public reproduction, frozen regressions and actual old-model resumes."""
import collections
import datetime
import hashlib
import json
import math
from pathlib import Path

r = Path(__file__).resolve().parents[3]
e = Path(__file__).resolve().parent
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
source = json.loads((e / 'formal-source.json').read_text())
assert len(source) == len(json.loads((e / 'source-before.json').read_text())) + 3
for path, digest in source.items():
    assert sha(r / path) == digest, path
frozen = json.loads((e / 'frozen-test.json').read_text())
for path, digest in frozen['files'].items():
    assert sha(r / path) == digest, path

def log(label):
    receipt = json.loads((e / (label + '.json')).read_text())
    p = e / receipt['log']
    assert receipt['exit_code'] == 0 and sha(p) == receipt['log_sha256']
    return p.read_bytes()

def numbers(raw):
    return json.loads(raw, parse_float=lambda s: ('number', s), parse_int=lambda s: ('number', s))

comparisons, counts = [], []
for suffix in ['', '-race']:
    events = [json.loads(line) for line in log('regression-candidate' + suffix).splitlines()]
    terminal = [x for x in events if x.get('Test') and x.get('Action') in ['pass', 'skip', 'fail']]
    count = collections.Counter(x['Action'] for x in terminal)
    assert set(count) == {'pass'} and count['pass'] > 0
    counts.append(dict(count))
    for kind, records in [('core', 16), ('chemical-resume', 6)]:
        before, after = log(kind + '-before' + suffix), log(kind + '-candidate' + suffix)
        assert before == after, (kind, suffix)
        assert len(json.loads(before)) == records
        comparisons.append(dict(before=kind + '-before' + suffix + '.log',
                                after=kind + '-candidate' + suffix + '.log',
                                records=records, bytes_equal=True))
    old = numbers(log('core-before' + suffix))
    resumed = numbers(log('core-state-resume-candidate' + suffix))
    assert resumed == [row['Continued'] for row in old]
assert counts[0] == counts[1]

public = json.loads(log('reproduction-candidate'))
assert len(public) == 18
for row in public:
    assert row['forward_matches'] and row['theta_gradient_matches'], row
    def close(got, want):
        if isinstance(want, list):
            assert len(got) == len(want)
            for a, b in zip(got, want):
                close(a, b)
        else:
            assert math.isfinite(got) and math.isclose(got, want, rel_tol=1e-12, abs_tol=0), (got, want)
    for key, expected in row['fixture']['expected_gradient'].items():
        close(row['gradient'][key], expected)

result = dict(all_passed=True, observed_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),
              source_manifest='formal-source.json', source_count=len(source), frozen_test='frozen-test.json',
              regression_events_each_build=counts[0], comparisons=comparisons,
              normal_core_records_bytes_equal=32, old_core_state_new_process_continuations=32,
              old_chemical_model_new_process_continuations=12, chemical_updates_after_resume_each=1,
              public_cases=18, forward_mismatch=0, theta_gradient_mismatch=0,
              all_public_gradient_fields_match=True,
              limits='CPU synthetic numerical and compatibility controls; no task-quality or biology claim.')
with (e / 'candidate-verification.json').open('x') as handle:
    json.dump(result, handle, indent=2)
    handle.write('\n')
p = r / 'docs/requirements-status.json'
raw = p.read_bytes()
status = json.loads(raw)
for item in status['requirements']:
    if item['id'] in ['COR-04', 'COR-09']:
        assert item['status'] == 'failed'
        item['status'] = 'implemented_unverified'
assert p.read_bytes() == raw
p.write_text(json.dumps(status, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(result))
