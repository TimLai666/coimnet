def update(path, substitutions, addition=''):
    p = r / path
    old = p.read_bytes()
    text = old.decode('utf-8')
    for before, after in substitutions:
        assert text.count(before) == 1, (path, before)
        text = text.replace(before, after)
    assert p.read_bytes() == old, path
    p.write_text(text + addition, encoding='utf-8')
