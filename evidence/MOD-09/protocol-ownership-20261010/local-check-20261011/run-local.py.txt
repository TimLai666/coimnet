import datetime
import hashlib
import json
import os
import platform
import subprocess
import time
from pathlib import Path

ROOT = Path('/Users/timlai/Developer/coimnet')
BASE = ROOT / 'runs/.tmp/bio-ownership-local-20261011'
EVIDENCE = ROOT / 'evidence/MOD-09/protocol-ownership-20261010/local-check-20261011'
GO = '/usr/local/go/bin/go'
EXPECTED_HEAD = 'cf008866cc0aa3fa9d2cab69c202930e51f6571c'


def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name, value):
    with (EVIDENCE / name).open('x') as out:
        json.dump(value, out, indent=2)
        out.write('\n')


def source():
    files = subprocess.check_output(['git', 'ls-files', '*.go'], cwd=ROOT, text=True).splitlines()
    return {'go_files': len(files),
            'source': [{'file': f, 'sha256': digest(ROOT / f)} for f in files],
            'modules': {f: digest(ROOT / f) for f in ['go.mod', 'go.sum']}}


def check(name, args):
    print(json.dumps({'check': name, 'state': 'running', 'utc': now()}), flush=True)
    started = now()
    tick = time.monotonic()
    with (EVIDENCE / (name + '.log')).open('xb') as out:
        result = subprocess.run(args, cwd=ROOT, env=ENV, stdout=out, stderr=subprocess.STDOUT)
    record = {'name': name, 'args': args, 'cwd': str(ROOT), 'started_utc': started,
              'finished_utc': now(), 'seconds': time.monotonic() - tick,
              'exit_code': result.returncode, 'log': name + '.log',
              'sha256': digest(EVIDENCE / (name + '.log'))}
    if name == 'format':
        record['unformatted_files'] = (EVIDENCE / (name + '.log')).read_text().splitlines()
        record['passed'] = result.returncode == 0 and not record['unformatted_files']
    else:
        record['passed'] = result.returncode == 0
    if name in ['unit-full', 'race-full']:
        lines = (EVIDENCE / (name + '.log')).read_text().splitlines()
        record['passed_packages'] = sum(line.startswith('ok ') for line in lines)
        record['without_test_packages'] = sum(line.startswith('? ') for line in lines)
        record['failed_package_lines'] = [line for line in lines if line.startswith('FAIL\t') or line.startswith('FAIL ')]
    save(name + '-execution.json', record)
    print(json.dumps(record), flush=True)
    return record
