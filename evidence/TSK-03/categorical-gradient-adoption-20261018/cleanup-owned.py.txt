"""Verify final acceptance and remove only ticket147's owned work directory."""
from pathlib import Path
import argparse, datetime, gzip, hashlib, json, os, shutil, stat, subprocess, tarfile

parser = argparse.ArgumentParser(description=__doc__)
mode = parser.add_mutually_exclusive_group()
mode.add_argument('--check', action='store_true', help='Verify completed acceptance, preserved inputs, original files and absence of open handles; never write or delete.')
mode.add_argument('--run', action='store_true', help='After the same checks, preserve an exact inventory and delete only runs/.tmp/categorical-gradient-candidate-20261018.')
args = parser.parse_args()
if not (args.check or args.run):
    parser.print_help()
    raise SystemExit(0)
if not __debug__:
    raise SystemExit('Optimized Python disables required checks; cleanup refused')
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
W = R / 'runs/.tmp/categorical-gradient-candidate-20261018'
P = R / 'evidence/LRN-08/aggregate-range-20261017'
PROTECTED = {'62': 49680181, 'real-task-quality-20261012': 50419696}
read = lambda p: json.loads(p.read_bytes())
sha = lambda b: hashlib.sha256(b).hexdigest()
required = ['runner-terminal.json', 'engineering-verification.json', 'verification.json',
            'metadata-governance.json', 'old-models-root-verification.json',
            'work-preservation.json', 'discarded-binaries.json']
missing = [n for n in required if not (E / n).is_file()]
if missing:
    print(json.dumps({'ready': False, 'missing': missing, 'deleted': False}))
    raise SystemExit(2)
terminal = read(E / 'runner-terminal.json')
assert terminal['session_id'] == 96150 and terminal['exit_code'] == 0
assert terminal['source_count'] == 895 and terminal['source_manifest'] == 'formal-source.json'
engineering = read(E / 'engineering-verification.json')
assert engineering['all_passed'] and engineering['source_count'] == 895
assert len(engineering['steps']) == 9
for step in engineering['steps']:
    assert step['exit_code'] == 0 and step['source_count'] == 895
    assert step['source_manifest'] == 'formal-source.json' and not step['acceptance_errors']
    raw = (E / step['log']).read_bytes()
    assert sha(raw) == step['log_sha256']
    if step['log'].endswith('.gz'):
        raw = gzip.decompress(raw)
        assert sha(raw) == step['raw_sha256'] and len(raw) == step['raw_bytes']
verified = read(E / 'verification.json')
assert verified['engineering_passed'] and verified['old_models_passed']
assert verified['source_count'] == 895 and verified['runner_terminal'] == 'runner-terminal.json'
source = read(E / 'formal-source.json')
assert sha((E / 'formal-source.json').read_bytes()) == verified['source_manifest_sha256']
old_models = read(E / 'old-models-root-verification.json')
assert old_models['old_model_load_and_next_training_passed']
assert old_models['original_candidate_formal_complete_outputs_byte_equal']
assert old_models['formal_source_manifest_sha256'] == verified['source_manifest_sha256']
governance = read(E / 'metadata-governance.json')
assert governance['exit_code'] == 0 and governance['passed_events'] == 31
assert governance['source_count'] == 895 and governance['source_manifest'] == 'formal-source.json'
assert sha((E / governance['log']).read_bytes()) == governance['log_sha256']
assert shutil.rmtree.avoids_symlink_attacks
assert not W.is_symlink() and W.resolve() == W and W.stat().st_ino == 61043512
anchor = P / 'preserved-work.tar.gz'
assert sha(anchor.read_bytes()) == '5104cdb4f97eaaacf951574b0162a008156fa112f4a65b856e47cd3c36c801c4'
with tarfile.open(anchor, 'r:gz') as handle:
    raw_modules = handle.extractfile('module-originals.json').read()
assert sha(raw_modules) == '34a6c7b8759c70527992dfc0967823ff93dda447d9c3c8e539342b4fae1238fc'
modules = json.loads(raw_modules)
models, foreign, data = (read(P / n) for n in ['protected-models.json', 'foreign-before.json', 'persistent-data.json'])
assert (len(source), len(models), len(foreign), len(data), len(modules)) == (895, 18, 32, 6, 15046)

def gate():
    paths = {p for p in subprocess.check_output(['git', 'ls-files', '-co', '--exclude-standard', '-z'], cwd=R).decode().split('\0') if p and (p.endswith('.go') or p in ['go.mod', 'go.sum'])}
    assert paths == set(source)
    for n, h in {**source, **models}.items():
        assert sha((R / n).read_bytes()) == h, n
    for n, h in {**foreign, **modules}.items():
        assert sha(Path(n).read_bytes()) == h, n
    for n, item in data.items():
        b = (R / n).read_bytes()
        assert len(b) == item['bytes'] and sha(b) == item['sha256'], n
    for n, inode in PROTECTED.items():
        p = R / 'runs/.tmp' / n
        assert not p.is_symlink() and p.stat().st_ino == inode

def inventory(root):
    result = {}
    for base, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            p = Path(base) / name
            assert not p.is_symlink(), p
            key = str(p.relative_to(root))
            if p.is_dir():
                result[key] = {'directory': True, 'inode': p.stat().st_ino}
            else:
                assert p.is_file(), p
                b = p.read_bytes()
                result[key] = {'bytes': len(b), 'sha256': sha(b)}
    return result

gate()
old_roots = {n: inventory(R / 'runs/.tmp' / n) for n in PROTECTED}
contents = inventory(W)
preserved = read(E / 'work-preservation.json')
archive = E / preserved['archive']
assert sha(archive.read_bytes()) == preserved['archive_sha256']
accounted = set()
with tarfile.open(archive, 'r:gz') as handle:
    members = handle.getmembers()
    assert len(members) == len(preserved['files'])
    for member in members:
        assert member.isfile() and member.name in preserved['files'] and member.name not in accounted
        assert not Path(member.name).is_absolute() and '..' not in Path(member.name).parts
        b = handle.extractfile(member).read()
        item = preserved['files'][member.name]
        assert len(b) == item['bytes'] and sha(b) == item['sha256']
        assert (W / member.name).read_bytes() == b
        accounted.add(member.name)
binaries = read(E / 'discarded-binaries.json')
assert set(binaries) == {'final-root.test'}
for n, item in binaries.items():
    b = (W / n).read_bytes()
    assert len(b) == item['bytes'] and sha(b) == item['sha256']
    accounted.add(n)
noncache = {n for n, item in contents.items() if n.split('/')[0] not in ['go-cache', 'go-mod-cache', 'go-tmp'] and not item.get('directory')}
assert noncache == accounted, ('Unaccounted inputs', noncache ^ accounted)
handles = subprocess.run(['lsof', '-t', '+D', str(W)], capture_output=True)
assert handles.returncode == 1 and not handles.stdout and not handles.stderr, handles
assert inventory(W) == contents
gate()
if args.check:
    print(json.dumps({'ready': True, 'entries': len(contents), 'noncache_files_preserved_or_discarded': len(accounted), 'no_open_handles': True, 'deleted': False}))
    raise SystemExit(0)
packed = gzip.compress(json.dumps(contents, sort_keys=True).encode(), mtime=0)
with (E / 'cleanup-inventory.json.gz').open('xb') as handle:
    handle.write(packed)
assert json.loads(gzip.decompress((E / 'cleanup-inventory.json.gz').read_bytes())) == contents
assert not W.is_symlink() and W.resolve() == W and W.stat().st_ino == 61043512
for base, dirs, files in os.walk(W, followlinks=False):
    p = Path(base)
    assert not p.is_symlink() and (p == W or W in p.parents)
    os.chmod(p, stat.S_IMODE(p.stat().st_mode) | 0o700)
shutil.rmtree(W)
assert not W.exists()
gate()
assert old_roots == {n: inventory(R / 'runs/.tmp' / n) for n in PROTECTED}
result = {'cleaned_at': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'owned_directory': str(W.relative_to(R)), 'owned_inode': 61043512,
          'removed_entries': len(contents), 'removed_bytes': sum(x.get('bytes', 0) for x in contents.values()),
          'inventory_archive': 'cleanup-inventory.json.gz', 'inventory_archive_sha256': sha(packed),
          'necessary_inputs_preserved': True, 'source_entries_preserved': 895, 'original_models_preserved': 18,
          'persistent_data_files_preserved': 6, 'foreign_files_preserved': 32, 'original_module_files_preserved': 15046,
          'protected_old_roots_full_inventory_preserved': True, 'no_open_handles': True, 'directory_removed': True}
with (E / 'cleanup.json').open('x') as handle:
    json.dump(result, handle, indent=2)
    handle.write('\n')
print(json.dumps(result), flush=True)
