def events(raw):
    tests, packages = {}, {}
    for line in raw.splitlines():
        try:
            row = json.loads(line)
        except ValueError:
            continue
        if row.get('Action') in ['pass', 'skip', 'fail']:
            if row.get('Test'):
                key = (row['Package'], row['Test'])
                assert key not in tests
                tests[key] = row['Action']
            else:
                packages[row['Package']] = row['Action']
    return tests, packages

def publish_new(name, raw):
    target = E / name
    temporary = E / (name + '.tmp')
    assert not target.exists() and not target.is_symlink()
    with temporary.open('xb') as handle:
        handle.write(raw)
        handle.flush()
        os.fsync(handle.fileno())
    os.link(temporary, target)
    temporary.unlink()
    fd = os.open(E, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)
