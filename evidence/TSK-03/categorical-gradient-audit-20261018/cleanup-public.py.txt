"""Remove only the completed gradient probe's copied tests and temporary files."""
from pathlib import Path
import argparse, datetime, hashlib, json, os, shutil, subprocess, tarfile

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--run', action='store_true', help='Require both completed red reproductions, preserve exact probe inputs, verify protected files, then remove only runs/.tmp/categorical-gradient-audit-20261018. The shared146cache is preserved.')
args = parser.parse_args()
if not args.run:
    parser.print_help()
    raise SystemExit(0)
if not __debug__:
    raise SystemExit('optimized Python cannot skip checks')
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
W = R / 'runs/.tmp/categorical-gradient-public-20261018'
P = R / 'evidence/LRN-08/aggregate-range-20261017'
read = lambda p: json.loads(p.read_bytes())
sha = lambda b: hashlib.sha256(b).hexdigest()
frozen = read(E / 'public-tests-frozen.json')
verified = read(E / 'public-reproduction-verification.json')
assert verified['public_update_difference_confirmed'] and not verified['candidate_fixed']
for build, session in [('unit', 7081), ('race', 52749)]:
    result = verified['results'][build]
    assert result['actual_terminal']['session_id'] == session and result['actual_terminal']['exit_code'] == 1 and result['test_events'] == {'fail': 2}
    assert sha((E / result['log']).read_bytes()) == result['log_sha256']
assert not W.is_symlink() and W.stat().st_ino == frozen['work_inode'] == 61026510
assert W.resolve() == W and shutil.rmtree.avoids_symlink_attacks
source = read(E / 'source.json')
foreign = read(P / 'foreign-before.json')
models = read(P / 'protected-models.json')
data = read(P / 'persistent-data.json')
archive = P / 'preserved-work.tar.gz'
assert sha(archive.read_bytes()) == '5104cdb4f97eaaacf951574b0162a008156fa112f4a65b856e47cd3c36c801c4'
with tarfile.open(archive, 'r:gz') as handle:
    module_bytes = handle.extractfile('module-originals.json').read()
assert sha(module_bytes) == '34a6c7b8759c70527992dfc0967823ff93dda447d9c3c8e539342b4fae1238fc'
modules = json.loads(module_bytes)
assert len(source) == 877 and len(modules) == 15046 and len(foreign) == 32 and len(models) == 18 and len(data) == 6

def gate():
    for name, digest in {**source, **models}.items():
        assert sha((R / name).read_bytes()) == digest, name
    for name, digest in {**foreign, **modules}.items():
        assert sha(Path(name).read_bytes()) == digest, name
    for name, item in data.items():
        b = (R / name).read_bytes()
        assert len(b) == item['bytes'] and sha(b) == item['sha256'], name
    for name, inode in {'62': 49680181, 'real-task-quality-20261012': 50419696}.items():
        p = R / 'runs/.tmp' / name
        assert not p.is_symlink() and p.stat().st_ino == inode

gate()
accounted = {}
for source_path, copied_path in frozen['overlay']['Replace'].items():
    target = Path(copied_path)
    b = target.read_bytes()
    matches = [name for name, digest in frozen['files'].items() if digest == sha(b)]
    assert len(matches) == 1 and (E / matches[0]).read_bytes() == b
    accounted[str(target.relative_to(W))] = matches[0]
overlay = (W / 'overlay.json').read_bytes()
assert json.loads(overlay) == frozen['overlay']
preserved_overlay = E / 'public-work-overlay.json.txt'
if preserved_overlay.exists():
    assert preserved_overlay.read_bytes() == overlay
else:
    with preserved_overlay.open('xb') as handle:
        handle.write(overlay)
accounted['overlay.json'] = 'public-work-overlay.json.txt'
cache = (W / 'go-tmp/xcrun_db').read_bytes()
assert len(cache) == 1276 and sha(cache) == 'fd76fdcab9f71bc3818d784671dc792781a5eba96e077bb99ebd72a9340c21fc'
accounted['go-tmp/xcrun_db'] = 'discarded-public-cache.json'
cache_record = {'file': 'go-tmp/xcrun_db', 'bytes': len(cache), 'sha256': sha(cache), 'reason': 'Disposable Xcode SDK and clang lookup cache generated inside this probe TMPDIR; contents inspected, no model or training data.'}
with (E / 'discarded-public-cache.json').open('x') as handle:
    json.dump(cache_record, handle, indent=2)
    handle.write('\n')
inventory = {}
for base, dirs, files in os.walk(W, followlinks=False):
    for name in dirs + files:
        p = Path(base) / name
        assert not p.is_symlink(), p
        inventory[str(p.relative_to(W))] = {'directory': True} if p.is_dir() else {'bytes': p.stat().st_size, 'sha256': sha(p.read_bytes())}
assert {name for name, item in inventory.items() if not item.get('directory')} == set(accounted)
handles = subprocess.run(['lsof', '-t', '+D', str(W)], capture_output=True)
assert handles.returncode == 1 and not handles.stdout and not handles.stderr, handles
for name, item in inventory.items():
    if not item.get('directory'):
        b = (W / name).read_bytes()
        assert len(b) == item['bytes'] and sha(b) == item['sha256']
with (E / 'public-cleanup-inventory.json').open('x') as handle:
    json.dump({'inventory': inventory, 'accounted': accounted}, handle, indent=2)
    handle.write('\n')
gate()
assert W.stat().st_ino == 61026510 and not W.is_symlink()
shutil.rmtree(W)
assert not W.exists()
gate()
result = {'cleaned_at': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'owned_directory': str(W.relative_to(R)), 'owned_inode': 61026510, 'removed_entries': len(inventory), 'removed_bytes': sum(item.get('bytes', 0) for item in inventory.values()), 'necessary_files_preserved': True, 'source_entries_preserved': 877, 'original_models_preserved': 18, 'persistent_data_files_preserved': 6, 'foreign_files_preserved': 32, 'original_module_files_preserved': 15046, 'shared146cache_preserved': True, 'directory_removed': True}
with (E / 'public-cleanup.json').open('x') as handle:
    json.dump(result, handle, indent=2)
    handle.write('\n')
print(json.dumps(result), flush=True)
