from pathlib import Path
import datetime, gzip, hashlib, json, os, shutil, stat, subprocess, sys, tarfile

r = Path('/Users/timlai/Developer/coimnet')
e = r / 'evidence/TSK-03/textgen-dimensions-20261015'
w = r / 'runs/.tmp/textgen-dimensions-20261015'
owned_inode = 55945854
protected = {'62': 49680181, 'real-task-quality-20261012': 50419696}
read = lambda p: json.loads(p.read_text())
sha = lambda raw: hashlib.sha256(raw).hexdigest()


def gate():
    sources = read(e / 'formal-source.json')
    assert len(sources) == 822
    for name, digest in sources.items():
        assert sha((r / name).read_bytes()) == digest, name
    for name, digest in read(e / 'protected-models.json').items():
        assert sha((r / name).read_bytes()) == digest, name
    for name, item in read(e / 'protected-preservation.json')['persistent_data_files'].items():
        raw = (r / name).read_bytes()
        assert len(raw) == item['bytes'] and sha(raw) == item['sha256'], name
    for name, inode in protected.items():
        p = r / 'runs/.tmp' / name
        assert not p.is_symlink() and p.stat().st_ino == inode, name


def inventory(root):
    result = {}
    for base, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            p = Path(base) / name
            key = str(p.relative_to(root))
            if p.is_symlink():
                result[key] = {'symlink': os.readlink(p)}
            elif p.is_dir():
                result[key] = {'directory': True, 'inode': p.stat().st_ino}
            else:
                assert p.is_file(), p
                raw = p.read_bytes()
                result[key] = {'bytes': len(raw), 'sha256': sha(raw)}
    return result


def archive_check(root, archive, members, expected_digest):
    assert sha(archive.read_bytes()) == expected_digest, archive
    with tarfile.open(archive, 'r:gz') as handle:
        assert sorted(handle.getnames()) == sorted(members), archive
        for name, item in members.items():
            raw = handle.extractfile(name).read()
            assert len(raw) == item['bytes'] and sha(raw) == item['sha256'], name
            assert raw == (root / name).read_bytes(), name


mode = sys.argv[1]
assert mode in ['preflight', 'clean']
gate()
assert not w.is_symlink() and w.stat().st_ino == owned_inode
assert read(e / 'engineering-verification.json')['all_passed']
accounted = set()
for item in read(e / 'compatibility-preservation.json')['archives']:

    accounted.update(str((root / name).relative_to(w)) for name in item['members'])

archive_check(w, e / work['archive'], work['files'], work['archive_sha256'])
accounted.update(work['files'])
binary = read(e / 'closure-binary.json')
raw = (w / binary['path']).read_bytes()
assert len(raw) == binary['bytes'] and sha(raw) == binary['sha256']
accounted.add(binary['path'])
before_protected = {name: inventory(r / 'runs/.tmp' / name) for name in protected}
contents = inventory(w)
for name, item in contents.items():
    if name.split('/')[0] not in ['go-cache', 'go-mod-cache', 'go-tmp'] and not item.get('directory'):
        assert name in accounted, ('unpreserved work', name)

    sys.exit(0)

handles = subprocess.run(['lsof', '-t', '+D', str(w)], capture_output=True)
assert handles.returncode == 1 and not handles.stdout and not handles.stderr, handles
assert inventory(w) == contents, 'owned contents changed before cleanup'
gate()
packed = gzip.compress(json.dumps(contents, sort_keys=True).encode(), mtime=0)
