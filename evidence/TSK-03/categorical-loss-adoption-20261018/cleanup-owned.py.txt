"""Check readiness or remove only this ticket's owned, verified work directory."""
from pathlib import Path
import argparse, datetime, gzip, hashlib, json, os, shutil, stat, subprocess, tarfile

parser = argparse.ArgumentParser(description=__doc__)
mode = parser.add_mutually_exclusive_group()
mode.add_argument('--check', action='store_true', help='Check completed validation, preserved inputs, original files and absence of open handles; do not delete or write anything.')
mode.add_argument('--run', action='store_true', help='After all checks pass, save an exact inventory and remove only runs/.tmp/categorical-loss-adoption-20261018. Preserve original modules, models, data and other work.')
args = parser.parse_args()
if not (args.check or args.run):
    parser.print_help()
    raise SystemExit(0)
if not __debug__:
    raise SystemExit('Optimized Python disables required checks; cleanup refused')
R = Path(__file__).resolve().parents[3]
E = Path(__file__).resolve().parent
W = R / 'runs/.tmp/categorical-loss-adoption-20261018'
P = R / 'evidence/LRN-08/aggregate-range-20261017'
PROTECTED = {'62': 49680181, 'real-task-quality-20261012': 50419696}
read = lambda p: json.loads(p.read_text())
sha = lambda b: hashlib.sha256(b).hexdigest()

required = ['runner-terminal.json', 'engineering-verification.json', 'verification.json', 'metadata-governance.json', 'discarded-binaries.json']
missing = [name for name in required if not (E / name).is_file()]
if missing:
    print(json.dumps({'ready': False, 'missing': missing, 'deleted': False}))
    raise SystemExit(2)
terminal = read(E / 'runner-terminal.json')
assert terminal['session_id'] == 36559 and terminal['exit_code'] == 0
assert terminal['source_manifest'] == 'formal-source-v2.json' and terminal['source_count'] == 877
assert terminal['label_prefix'] == 'engineering-final'
engineering = read(E / 'engineering-verification.json')
assert engineering['all_passed'] and engineering['source_count'] == 877
assert len(engineering['steps']) == 9
for step in engineering['steps']:
    assert step['exit_code'] == 0 and step['source_count'] == 877
    assert step['source_manifest'] == 'formal-source-v2.json' and not step['acceptance_errors']
    assert step['log'].startswith('engineering-final-')
    assert sha((E / step['log']).read_bytes()) == step['log_sha256']
verification = read(E / 'verification.json')
assert verification['engineering_passed'] and verification['runner_terminal'] == 'runner-terminal.json'
fingerprint = verification['input_fingerprints']
assert fingerprint['source_manifest'] == 'formal-source-v2.json' and fingerprint['source_entries'] == 877
assert fingerprint['source_manifest_sha256'] == sha((E / 'formal-source-v2.json').read_bytes())
assert read(E / 'source-state.json') == {'manifest': 'formal-source-v2.json', 'source_count': 877}
governance = read(E / 'metadata-governance.json')
assert governance['exit_code'] == 0 and governance['passed_events'] == 31
assert governance['source_manifest'] == 'formal-source-v2.json' and governance['source_count'] == 877
assert governance['source_manifest_sha256'] == fingerprint['source_manifest_sha256']
assert sha((E / governance['log']).read_bytes()) == governance['log_sha256']
assert not W.is_symlink() and W.stat().st_ino == 60930722 and W.resolve() == W
assert shutil.rmtree.avoids_symlink_attacks

def gate():
    source = read(E / 'formal-source-v2.json')
    assert len(source) == 877
    paths = {p for p in subprocess.check_output(['git', 'ls-files', '-co', '--exclude-standard', '-z'], cwd=R).decode().split('\0') if p and (p.endswith('.go') or p in ['go.mod', 'go.sum'])}
    assert paths == set(source)
    models = read(P / 'protected-models.json')
    data = read(P / 'persistent-data.json')
    foreign = read(P / 'foreign-before.json')
    assert len(models) == 18 and len(data) == 6 and len(foreign) == 32
    for name, digest in {**source, **models}.items():
        assert sha((R / name).read_bytes()) == digest, name
    for name, item in data.items():
        b = (R / name).read_bytes()
        assert len(b) == item['bytes'] and sha(b) == item['sha256'], name
    for name, digest in foreign.items():
        assert sha(Path(name).read_bytes()) == digest, name
    for name, inode in PROTECTED.items():
        p = R / 'runs/.tmp' / name
        assert not p.is_symlink() and p.stat().st_ino == inode

def inventory(root):
    result = {}
    for base, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            p = Path(base) / name
            key = str(p.relative_to(root))
            if p.is_symlink():
                result[key] = {'symlink': os.readlink(p)}
            elif p.is_dir():
                result[key] = {'directory': True, 'inode': p.stat().st_ino}
            else:
                assert p.is_file(), p
                b = p.read_bytes()
                result[key] = {'bytes': len(b), 'sha256': sha(b)}
    return result

preserved = read(E / 'work-preservation.json')
accounted = {}
anchor = preserved['reused_module_manifest']
archive = R / anchor['archive']
assert sha(archive.read_bytes()) == anchor['archive_sha256']
with tarfile.open(archive, 'r:gz') as handle:
    b = handle.extractfile(anchor['member']).read()
    assert sha(b) == anchor['member_sha256'] and (W / 'module-originals.json').read_bytes() == b
module_source = json.loads(b)
assert len(module_source) == 15046
for name, digest in module_source.items():
    assert sha(Path(name).read_bytes()) == digest, name
accounted['module-originals.json'] = anchor['archive'] + ':' + anchor['member']
preservation = read(E / 'cleanup-preservation.json')
assert sha((E / 'preserved-overlays.json.gz').read_bytes()) == preservation['overlay_archive_sha256']
overlays = json.loads(gzip.decompress((E / 'preserved-overlays.json.gz').read_bytes()))
assert len(overlays) == preservation['overlay_count']
for name, item in preservation['files'].items():
    target = R / item['path']
    if item['kind'] == 'file':
        b = target.read_bytes()
    elif item['kind'] == 'gzip':
        b = gzip.decompress(target.read_bytes())
    elif item['kind'] == 'tar':
        assert sha(target.read_bytes()) == item['archive_sha256']
        with tarfile.open(target, 'r:gz') as handle:
            b = handle.extractfile(item['member']).read()
    elif item['kind'] == 'overlay':
        b = overlays[item['member']].encode()
    else:
        raise AssertionError(item['kind'])
    assert len(b) == item['bytes'] and sha(b) == item['sha256'], name
    assert (W / name).read_bytes() == b, name
    accounted[name] = item
binaries = read(E / 'discarded-binaries.json')
assert set(binaries) == {'final-root.test'}
for name, item in binaries.items():
    b = (W / name).read_bytes()
    assert len(b) == item['bytes'] and sha(b) == item['sha256']
    accounted[name] = 'discarded-binaries.json'
gate()
old_roots = {name: inventory(R / 'runs/.tmp' / name) for name in PROTECTED}
contents = inventory(W)
noncache = {name for name, item in contents.items() if name.split('/')[0] not in ['go-cache', 'go-mod-cache', 'go-tmp'] and not item.get('directory')}
assert noncache == set(accounted), ('Unaccounted work files', noncache ^ set(accounted))
handles = subprocess.run(['lsof', '-t', '+D', str(W)], capture_output=True)
assert handles.returncode == 1 and not handles.stdout and not handles.stderr, handles
assert inventory(W) == contents, 'Owned contents changed before cleanup'
gate()
if args.check:
    print(json.dumps({'ready': True, 'entries': len(contents), 'noncache_files_accounted': len(accounted), 'no_open_handles': True, 'deleted': False}))
    raise SystemExit(0)
packed = gzip.compress(json.dumps(contents, sort_keys=True).encode(), mtime=0)
with (E / 'cleanup-inventory.json.gz').open('xb') as handle:
    handle.write(packed)
assert json.loads(gzip.decompress((E / 'cleanup-inventory.json.gz').read_bytes())) == contents
with (E / 'cleanup-accounting.json').open('x') as handle:
    json.dump(accounted, handle, indent=2)
    handle.write('\n')
assert not W.is_symlink() and W.stat().st_ino == 60930722 and W.resolve() == W
for base, dirs, files in os.walk(W, topdown=True, followlinks=False):
    directory = Path(base)
    assert not directory.is_symlink() and (directory == W or W in directory.parents)
    os.chmod(directory, stat.S_IMODE(directory.stat().st_mode) | 0o700)
shutil.rmtree(W)
assert not W.exists()
for name, digest in module_source.items():
    assert sha(Path(name).read_bytes()) == digest, name
gate()
assert old_roots == {name: inventory(R / 'runs/.tmp' / name) for name in PROTECTED}
result = {'cleaned_at': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'owned_directory': str(W.relative_to(R)), 'owned_inode': 60930722, 'removed_entries': len(contents), 'removed_bytes': sum(x.get('bytes', 0) for x in contents.values()), 'inventory_archive': 'cleanup-inventory.json.gz', 'inventory_archive_sha256': sha(packed), 'no_open_handles': True, 'necessary_files_preserved': True, 'source_entries_preserved': 877, 'original_model_files_preserved': 18, 'persistent_data_files_preserved': 6, 'foreign_files_preserved': 32, 'original_module_files_preserved': len(module_source), 'protected_old_roots_preserved': list(PROTECTED), 'directory_removed': True}
with (E / 'cleanup.json').open('x') as handle:
    json.dump(result, handle, indent=2)
    handle.write('\n')
print(json.dumps(result), flush=True)
