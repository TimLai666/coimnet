package main
import("encoding/json";"os";"math";"fmt")
func main(){for _,act:=range []string{"tanh","softplus"}{data,e:=os.ReadFile(os.Args[1]+"/coimnet-crosscpu-"+act+".json");if e!=nil{panic(e)};var d struct{Payload struct{Neural struct{Voltage []float64;History [][]float64}}};if e=json.Unmarshal(data,&d);e!=nil{panic(e)};mismatch:=0;var maxULP uint64;var maxAbs float64;samples:=[]string{};for j,v:=range d.Payload.Neural.Voltage{got:=math.Tanh(v);if act=="softplus"{got=math.Max(v,0)+math.Log1p(math.Exp(-math.Abs(v)))};saved:=d.Payload.Neural.History[0][j];if saved!=got{mismatch++;a,b:=math.Float64bits(saved),math.Float64bits(got);var ulp uint64;if a>b{ulp=a-b}else{ulp=b-a};if ulp>maxULP{maxULP=ulp};if diff:=math.Abs(saved-got);diff>maxAbs{maxAbs=diff};if len(samples)<3{samples=append(samples,fmt.Sprintf("index=%d voltage=%.17g saved=%.17g computed=%.17g ulp=%d",j,v,saved,got,ulp))}}};json.NewEncoder(os.Stdout).Encode(map[string]any{"activation":act,"values":len(d.Payload.Neural.Voltage),"mismatches":mismatch,"max_ulp":maxULP,"max_absolute_error":maxAbs,"samples":samples})}}
