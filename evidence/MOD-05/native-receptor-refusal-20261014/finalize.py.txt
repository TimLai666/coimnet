from pathlib import Path
import json, gzip, hashlib, collections

r = Path('/Users/timlai/Developer/coimnet')
e = r/'evidence/MOD-05/native-receptor-refusal-20261014'
archives = []
for p in list(e.glob('*.log')) + list(e.glob('normal-*-cases.json')):
    raw = p.read_bytes()
    if len(raw) <= 200000 and not p.name.endswith('-cases.json'):
        continue
    dest = p.with_name(p.name + '.gz')
    assert not dest.exists()
    dest.write_bytes(gzip.compress(raw, mtime=0))
    assert gzip.decompress(dest.read_bytes()) == raw
    entry = {'raw_path':p.name, 'archive':dest.name, 'raw_sha256':hashlib.sha256(raw).hexdigest(), 'archive_sha256':hashlib.sha256(dest.read_bytes()).hexdigest(), 'raw_bytes':len(raw), 'archive_bytes':dest.stat().st_size}
    archives.append(entry)
    receipt = e/(p.stem+'.json')
    if p.suffix == '.log' and receipt.exists():
        b = receipt.read_bytes()
        x = json.loads(b)
        assert x['log'] == p.name and x['log_sha256'] == entry['raw_sha256']
        x.update(log=dest.name, log_compression='gzip', compressed_log_sha256=entry['archive_sha256'])
        assert receipt.read_bytes() == b
        receipt.write_text(json.dumps(x, indent=2)+'\n')
    assert p.read_bytes() == raw
    p.unlink()
(e/'archive-terminal.json').write_text(json.dumps(archives, indent=2)+'\n')
p = e/'normal-compatibility.json'
b = p.read_bytes()
x = json.loads(b)

p.write_text(json.dumps(x, indent=2)+'\n')

    (e/(name+'-summary.json')).write_text(json.dumps(summary, indent=2)+'\n')
